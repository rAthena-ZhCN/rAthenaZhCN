function	script	ksjl	{
    set .@jl_max_num, 6;
    
    for(set .@i,1; .@i<=10; set .@i,.@i+1){
        //判断可以精炼装备
        if (getequipisenableref(.@i) && getequipisequiped(.@i)){
        
            .@equip_refine = getequiprefinerycnt(.@i);
            while (1) {
                if(.@equip_refine < .@jl_max_num){
                    set .@equip_refine, .@equip_refine+1;
                    successrefitem .@i;
                }
                else{
                    break;
                }
            }
        }
    }
    
    mes "[欧托莉柯]";
    mes "都精炼好了。";
    close;
}

function	script	add_Weight	{
    setarray .@use_item[0], 7776,1;                //消耗道具
    
    mes "[欧托莉柯]";
    mes "负重提升 需要消耗 ^FF8C8C" + getitemname(.@use_item[0]) + "×" + .@use_item[1] + "^000000。";
	
	
	set .@s_lv, getskilllv(681);
	if(.@s_lv < 10){
		mes "最多提升 10次(当前 ^fbb217" + .@s_lv + "^000000 次)";
		
		if (select("提升") == 1){
			clear;
			set .@item_num, countitem(.@use_item[0]);
			if(.@s_lv < 1){
				//未有技能
				if(.@item_num > 0){
					if(.@item_num < 11){
						mes "已提升 " + .@item_num + "次";
						delitem .@use_item[0], .@item_num;
						skill 681, .@item_num, 3;
					}
					else{
						mes "已提升 10次";
						delitem .@use_item[0], 10;
						skill 681, 10, 3;
					}
				}
				else{
					mes "没有 " + getitemname(.@use_item[0]);
				}
			}
			else{
				//已有技能
				if(.@item_num > 0){
					if(.@item_num < (10 - .@s_lv)){
						mes "已提升 " + .@item_num + "次";
						delitem .@use_item[0], .@item_num;
						skill 681, (.@s_lv + .@item_num), 3;
					}
					else{
						mes "已提升 " + (10 - .@s_lv) + "次";
						delitem .@use_item[0], (10 - .@s_lv);
						skill 681, (.@s_lv + (10 - .@s_lv)), 3;
					}
				}
				else{
					mes "没有 " + getitemname(.@use_item[0]);
				}
			}
		}
	}
	else{
		mes "最多提升 10次(已满级)";
	}

    close;
}
