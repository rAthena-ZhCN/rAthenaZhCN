	mes "请问您要前往哪一个环境？";
	next;

	// 提供玩家选择菜单
	switch(select("生物圈标本环境 - 草原:生物圈标本环境 - 火焰:生物圈标本环境 - 冰冻:生物圈标本环境 - 死灵:取消 (留在原地)")) {
		case 1:
			mes "[管家β]";
			mes "正在为您连接【草原】生态圈，请注意安全。";
			close2;
			warp "bl_grass",0,0; // 0,0 表示传送到该地图的随机安全坐标
			end;
			
		case 2:
			mes "[管家β]";
			mes "正在为您连接【火焰】生态圈，请注意高温。";
			close2;
			warp "bl_lava",0,0;
			end;
			
		case 3:
			mes "[管家β]";
			mes "正在为您连接【冰冻】生态圈，请做好防寒准备。";
			close2;
			warp "bl_ice",0,0;
			end;
			
		case 4:
			mes "[管家β]";
			mes "正在为您连接【死灵】生态圈，那里非常危险，请务必小心。";
			close2;
			warp "bl_death",0,0;
			end;
			
		case 5:
			mes "[管家β]";
			mes "好的，如果您准备好探索了，请随时来找我。";
			close;
	}
}
