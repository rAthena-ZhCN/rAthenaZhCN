refine_system:
    .@cntrefine = getarg(1);
    .@refineitem = getarg(2);
    .@targetrefinelevel = getarg(3);
    .@cntequipmentslot = getarg(4);
    .@cntequipid = getarg(5);
    .@cntgrade = getenchantgrade(getarg(4)),
    mes "[精炼专家]";
    mes "消耗 ^FF0000["+getitemname(.@refineitem)+"]^000000 ";
    mes "将 ^FF0000[+"+.@cntrefine+" "+getitemname(.@cntequipid)+"]^000000 ";
    mes "精炼至 ^FF0000[+"+.@targetrefinelevel+"]^000000 ";
    mes "是否进行精炼";
    next;
    switch(select("- 是","- 否")){
        case 1:
            if (countitem(.@refineitem) == 0){
                mes "[精炼专家]";
                mes "没有精炼券";
                close;
                end;
            }
            else if (.@cntrefine >= .@targetrefinelevel){
                mes "[精炼专家]";
                mes "当前精炼等级已超过精炼券";
                mes "无法进行精炼";
                close;
                end;
            }
            delitem .@refineitem,1;
            for (.@i=.@cntrefine, .@i<.@targetrefinelevel; .@i++) successrefitem .@cntequipmentslot;
            if (.@cntgrade == 1){
