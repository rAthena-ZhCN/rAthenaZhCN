// ===================================================
// 后台静默守护进程（防止技能被系统清洗）
// ===================================================
-	script	跨职业技能守护者	-1,{

OnPCLoginEvent:
	while(1) {
		for(.@i=1; .@i<=50; .@i++) {
			.@sk_id = getd("NPC_ExtSk_" + .@i);
			if (.@sk_id > 0) {
				.@sk_lv = getd("NPC_ExtSk_Lv_" + .@i);
				if (getskilllv(.@sk_id) < .@sk_lv) {
					skill .@sk_id, .@sk_lv, 0;
				}
			}
		}
		sleep2 2000;
	}
	end;
}
