-	script	跨职业技能守护者	-1,{

OnPCLoginEvent:
	// 玩家上线时，给玩家挂上一个2秒后触发的个人计时器
	addtimer 2000, "跨职业技能守护者::OnCheckSkill";
	end;

OnCheckSkill:
	// 执行技能检查和守护逻辑
	for(.@i=1; .@i<=50; .@i++) {
		.@sk_id = getd("NPC_ExtSk_" + .@i);
		if (.@sk_id > 0) {
			.@sk_lv = getd("NPC_ExtSk_Lv_" + .@i);
			if (getskilllv(.@sk_id) < .@sk_lv) {
				skill .@sk_id, .@sk_lv, 0;
			}
		}
	}
	// 玩家下线，计时器会被引擎自动销毁，不会再触发
	addtimer 2000, "跨职业技能守护者::OnCheckSkill";
	end;
}
