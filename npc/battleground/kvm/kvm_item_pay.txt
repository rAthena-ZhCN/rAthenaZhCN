	setarray .@prices[0],2000,840,630,580,1200,120,70,200,15,10,30;

	.@item_id = getarg(0);
	.@req_setting = getarg(1);
	.@item_type = getarg(2);

	mes "[后勤官]";

	if (!.@item_type)
		mes "您选择了 ^ff0000"+ getitemname(.@item_id) +"^000000。";
	else if (.@item_type == 1)
		mes "只有一种"+ callfunc("F_GetWeaponType",.@item_id) +": "+ getitemname(.@item_id) +".";
	else if (.@item_type == 2) {
		mes "您选择了'"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "等级 80 以上可使用的防具为：^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 3) {
		mes "您选择了'量产型"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "只有一种量产型"+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 4) {
		mes "您选择了'普及型"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "只有一种普及型"+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}

	if (!.@req_setting)
		mes "需要 ^0000ff2,000^000000 KVM点数，"+ .@name$ +"，您的积分为：^580080"+ kvm_point +"^000000。";
	else if (.@req_setting == 4)
		mes "需要 ^0000ff1,200^000000 KVM点数，"+ .@name$ +"，您的积分为：^580080"+ kvm_point +"^000000。";
	else
		mes "需要 ^0000ff"+ .@prices[.@req_setting] +"^000000 KVM点数，"+ .@name$ +"，您的积分为：^580080"+ kvm_point +"^000000。";

	mes "您确定要这件物品吗？";
	next;
	switch(select("不，我不购买。:是的，我要购买。")) {
	case 1:
		mes "[后勤官]";
		mes "您选择了'我不购买'。";
		mes "购买物品时请注意，兑换后不予退还。";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[后勤官]";
			mes "您购买了"+ getitemname(.@item_id) +".";
			if (!.@req_setting)
				mes "您的KVM点数减少了 ^0000ff"+ .@req_setting +"^000000 点，当前KVM点数为 ^580080"+kvm_point+"^000000。";
			if (.@req_setting == 4)
				mes "您的KVM点数减少了 ^0000ff1,200^000000 点，当前KVM点数为 ^580080"+kvm_point+"^000000。";
			else
				mes "您的KVM点数减少了 ^0000ff"+ .@prices[.@req_setting] +"^000000 点，当前KVM点数为 ^580080"+kvm_point+"^000000。";
		}
		else {
			mes "[后勤官]";
			if (!.@req_setting)
				mes "购买此物品需要 ^0000ff2,000^000000 KVM点数。";
			else if (.@req_setting == 4)
				mes "购买此物品需要 ^0000ff1,200^000000 KVM点数。";
			else
				mes "您需要 ^0000ff"+ .@prices[.@req_setting] +"^000000 KVM点数才能购买此物品。";
			mes "但您当前的KVM点数只有 ^580080"+ kvm_point +"^000000。";

			mes "等您攒够点数后，请再来。";

	}
	close;
