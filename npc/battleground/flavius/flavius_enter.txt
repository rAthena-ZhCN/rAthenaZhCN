bat_room,133,178,5	script	弗拉维乌斯军官#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 请稍等！！ -";
		mes "- 您目前携带了 -";
		mes "- 太多物品。 -";
		mes "- 请稍后再试 -";
		mes "- 减轻一些负重之后。 -";
		close;
	}
	mes "[纪尧姆军军官]";
	mes "让那些臭烘烘的克罗伊党徒见识纪尧姆军的实力！";
	next;
	switch(select("我愿意加入您的军队！:结束对话")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[纪尧姆军军官]";
			mes "很高兴您愿意加入我们对抗克罗伊军，但很抱歉：我不能让像您这样的新兵去残酷的战场上送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[纪尧姆军军官]";
				mes "很高兴您愿意加入我们对抗克罗伊军，但很抱歉：我不能让像您这样的新兵去残酷的战场上送死。";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[纪尧姆军军官]";
					mes "您似乎刚从战场回来。";
					mes "您现在回去还太早。去休息吧，敌人交给我们！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[纪尧姆军军官]";
						mes "我已收到报告，一支精锐部队已被派往与克罗伊军交战的战场。";
						mes "您何不去稍等一会儿？我建议您在那之前磨好武器、备好补给。";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[纪尧姆军军官]";
							mes "一支精锐部队已待命，即将被派往战场。";
							mes "您何不去稍等一会儿？我建议您在那之前磨好武器、备好补给。";
						}
						else {
							mes "[纪尧姆军军官]";
							mes "您看来确实已做好战斗准备！";
							mes "去让克罗伊军见识真正的恐惧！";
							mes "今天，我们的胜利呐喊将响彻整个战场！";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[纪尧姆军军官]";
		mes "今天，我们将获得胜利！";
		break;
	}
	close;
}
