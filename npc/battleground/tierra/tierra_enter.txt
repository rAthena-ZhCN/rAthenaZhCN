bat_room,124,178,5	script	提耶拉峡谷军官#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 请等一下！！ -";
		mes "- 您目前携带的 -";
		mes "- 物品太多了。 -";
		mes "- 请减轻负重后 -";
		mes "- 再行尝试。 -";
		close;
	}
	mes "[纪尧姆军军官]";
	mes "让那些臭烘烘的克罗伊人见识一下纪尧姆军的实力！";
	next;
	switch(select("我想加入你们的军队！:结束对话")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[纪尧姆军军官]";
			mes "很高兴您愿意加入我们，与克罗伊军作战，但很抱歉：我不能把您这样的新手送上残酷的战场送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[纪尧姆军军官]";
				mes "很高兴您愿意加入我们，与克罗伊军作战，但很抱歉：我不能把您这样的新手送上残酷的战场送死。";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[纪尧姆军军官]";
					mes "您似乎刚从战场回来。";
					mes "现在回去还太早。去休息吧，敌人交给我们！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("bat_a01");
					if (.@mapcount > 0) {
						mes "[纪尧姆军军官]";
						mes "我已收到报告，精锐部队已经派往与克罗伊军交战的战场。";
						mes "您何不先去等一会儿？建议您在这之前磨好武器、备齐补给。";
					}
					else {
						if ($@TierraBG1) {
							mes "[纪尧姆军军官]";
							mes "精锐部队已经待命，即将派往战场。";
							mes "您何不先去等一会儿？建议您在这之前磨好武器、备齐补给。";
						}
						else {
							mes "[纪尧姆军军官]";
							mes "您看起来确实已做好战斗准备！";
							mes "去让克罗伊人知道什么才是真正的恐惧！";
							mes "今天，我们的胜利呐喊将响彻整个战场！";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[纪尧姆军军官]";
		mes "今天，我们必将获胜！";
		break;
	}
	close;
}
