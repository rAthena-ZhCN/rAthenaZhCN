			mes "感谢您的兑换。";
			delitem .@cost, .@Items[.@i+1];

		}
		else mes "您的数量不足"+getitemname(.@cost)+"s.";
		break;
	}
	close;

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[阿克塞尔·罗斯]";
			mes "纪尧姆万岁！";
		} else if (.@team$ == "Croix") {
			mes "[斯万德里]";
			mes "克罗伊万岁！";
		}
		mes "让我们享受这光荣的胜利吧！";
		mes strcharinfo(0)+"，这正是胜利的象征。";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[阿克塞尔·罗斯]";
			mes "您虽败北，但您为这场战斗竭尽全力。";
			mes "这是纪尧姆·马罗洛为您的奉献所赐的奖励！";
			mes "把这次失败当作教训，下次您一定能赢。";
		} else if (.@team$ == "Croix") {
			mes "[斯万德里]";
			mes "哦，"+strcharinfo(0)+"别难过。";
			mes "虽然我们没能获胜，但已尽力而为。";
			mes "这是克罗伊赐予的皇家礼物，请不要忘记这场战斗。下一场我们会赢。";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
