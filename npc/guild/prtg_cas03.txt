// Guild Flags (Outside Castle)
//============================================================
-	script	Fadhgridh#1::OutsideFlagsPR3	722,{

	set .@GID, GetCastleData("prtg_cas03",1);

	if (.@GID == 0) {
		mes "[ 神圣卢恩-米德加尔特王国敕令 ]";
		mes " ";
		mes "1. 谨遵神圣卢恩-米德加尔特王国的敕令，";
		mes "本王国宣告";
		mes "此城并无正式城主。";
		mes " ";
		mes "2. 凡能";
		mes "历经一切试炼";
		mes "并摧毁华丽金属者，";
		mes "国王将赐予其";
		mes "此城的所有权。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ 回响之声 ]";
		mes "勇敢的人们……";
		mes "你愿意回到属于你的荣耀之地吗？";
		next;
		if(select("返回公会城堡:退出") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("prtg_cas03",1)) {
				warp "prtg_cas03",181,215;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 神圣卢恩-米德加尔特王国敕令 ]";
	mes " ";
	mes "1. 谨遵神圣卢恩-米德加尔特王国的敕令，";
	mes "本王国认可此地属于";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000公会的私有领地。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000公会的会长是";
	mes "^FF0000"+GetGuildMaster(.@GID)+"^000000";
	mes "若有任何人对此持有异议，";
	mes "便请手持钢刃，证明你的力量与荣誉。";
	close;

OnRecvCastlePt03:
	FlagEmblem GetCastleData("prtg_cas03",1);
	end;
}
