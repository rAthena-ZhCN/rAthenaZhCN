// Guild Flag (Outside Castle)
//============================================================
-	script	Eeyorbriggar#1::OutsideFlagsG2	722,{

	set .@GID, GetCastleData("gefg_cas02",1);

	if (.@GID == 0) {
		mes "[ 神圣卢恩-米德加尔特王国敕令 ]";
		mes " ";
		mes "1. 遵守神圣卢恩-米德加尔特王国的法令，";
		mes "兹宣告";
		mes "本城堡目前没有正式的主人。";
		mes " ";
		mes "2. 对于能够";
		mes "克服所有试炼";
		mes "并摧毁华丽金属者，";
		mes "国王将授予其";
		mes "本城堡的所有权。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ 回响之声 ]";
		mes "勇敢的人们……";
		mes "你们希望回到荣耀之地吗？";
		next;
		if(select("返回公会城堡。:离开。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("gefg_cas02",1)) {
				warp "gefg_cas02",178,43;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 神圣卢恩-米德加尔特王国敕令 ]";
	mes " ";
	mes "1. 遵守神圣卢恩-米德加尔特王国的法令，";
	mes "我们认可此处属于";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000公会的私有领地。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000公会的会长是";
	mes "^FF0000"+GetGuildMaster(.@GID)+"^000000";
	mes "若有人对此持有异议，";
	mes "请以手中钢刃证明自身的力量与荣誉。";
	close;

OnRecvCastleG02:
	FlagEmblem GetCastleData("gefg_cas02",1);
	end;
}
