xmas_in,100,96,4	script	圣诞老人::Santa2	718,{
	mes "[圣诞老人]";
	if(xmas_npc==0) set xmas_npc, 1; // For Lutie & Bard quest
	if(#event_xmas > 0 && #event_xmas < 30 ) goto L_Start;
	mes "圣诞快乐！";
	if(Class==0 || #event_xmas>=30 ) close; //anti exploit protection
	mes "我有个礼物要送给您！呵呵呵！";
	getitem rand(664,667),1; //gives one of 4 gift boxes
	set #event_xmas,#event_xmas+1;
	close;

L_Start:
	mes "我遇到了一点麻烦……";
	mes "您愿意听听吗？";
	next;
	menu "听圣诞老人说话。",M_0, "把证据交给圣诞老人。",M_1, "Cancel.",M_End;

	M_0:
		mes "[圣诞老人]";
		mes "我的麻烦是这样的。";
		mes "外面似乎有个人";
		mes "冒充我，四处散播";
		mes "恐惧，扰乱这片土地。";
		next;
		mes "[圣诞老人]";
		mes "就像传说中的格林奇，他夺走";
		mes "所有孩子的玩具，把它们";
		mes "据为己有。";
		next;
		mes "[圣诞老人]";
		mes "我在这里忙着制作明年的一批";
		mes "玩具，所以没办法亲自";
		mes "出去找他。";
		mes "所以我想请您出去";
		mes "替我消灭那个人。";
		next;
		mes "[圣诞老人]";
		mes "他手上有一件我的";
		mes "魔法袋子，所以每次";
		mes "您攻击他，他都会钻进袋子";
		mes "逃到别的地方。";
		next;
		mes "[圣诞老人]";
		mes "不过，";
		mes "他匆忙之间常常会掉东西。";
		mes "如果他碰巧掉了一只";
		mes "带洞的袜子，他就是用它偷";
		mes "穷孩子们的玩具，请把它捡起来。";
		next;
		mes "[圣诞老人]";
		mes "如果您收集 3 只，我会送给您";
		mes "一个试作的神秘箱子，";
		mes "我一直把它放在";
		mes "实验室里。它会随机吐出礼物";
		mes "省了我很多工作。";
		close;
	M_1:
		mes "[圣诞老人]";
		if(countitem(7034) < 3) goto L_NotEnuf;
		delitem 7034,3;
		mes "看来您做得";
		mes "很不错，已经打倒了那些";
		mes "假圣诞老人。继续加油！";
		next;
		getitem 644,1; //Gift Box Prototype
		set #event_xmas,#event_xmas+1;
		mes "[圣诞老人]";
		mes "这是您的报酬。";
		mes "如果您再拿到 3 只，我会再给您一个。";
		mes "希望您能拿到好道具。";
		close;
	
		L_NotEnuf:
			mes "作为证据的袜子还不够。";
			mes "去打倒那些邪恶的圣诞老人";
			mes "多带些回来给我，我会给您报酬。";
			close;
	M_End:
		mes "[圣诞老人]";
		mes "原来如此。那么，至少";
		mes "我们就在圣诞节的早晨再见吧。";
		close;

OnInit:
	disablenpc "圣诞老人";
	end;
}
