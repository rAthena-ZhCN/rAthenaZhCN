	end;

		mes "- 你发现了一堆覆着泥土的石头。-";
		close;
	}
	if (StPatrick2008 == 1) {
		mes "- 你发现了一堆覆着泥土的石头。-";
		next;
		if (countitem(12018) < 200) {
			mes "石头纹丝不动。";

			close;
		}
		else {
			if (select("使用鞭炮。:无视") == 2) {
				close;
			}
			mes "你在石堆下面埋了 200 个鞭炮。";
			mes "你点燃了引线。";
			next;
			mes "*轰！*";
			specialeffect2 EF_LORD;
			next;
			mes "待尘土与烟雾散去后，";
			mes "你在石缝与泥土之间发现了一个箱子。";
			mes "这箱子里一定装着奥莱利的宝贝。";
			mes "把箱子带去给奥莱利吧。";
			close2;
			set StPatrick2008,2;
			delitem 12018,200;
			getitem 7721,1;
		}
	}
	end;
}

		getitem 7915,1;
		end;

		getitem 7915,1;
		getitem 7916,1;

		end;
	case 3:
		getitem 7720,1;
		end;
	case 4:
		getitem 7915,1;
		getitem 7916,1;
		end;
	case 5:
		getitem 7916,2;
		end;
	case 6:
		getitem 7915,2;
		end;
	}
}
