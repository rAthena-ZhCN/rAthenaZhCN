		mes "-但他们的英名将与他们所捍卫的土地一样长存。-";

		}
		next;

		next;
		mes "-我或许没有去到我原定要去的地方，-";
		mes "-但我想，我最终来到了我该在的地方。-";
		mes "- 道格拉斯·亚当斯.-";

			getexp 93750,43750;
		}
		next;
		mes "-这是最后一段文字。-";
		next;
		mes "-真正的英雄主义异常朴素，毫不起眼。-";
		mes "-它不是不惜一切代价超越他人的冲动，-";
		mes "- 而是不惜一切代价服务他人的愿望。 -";
