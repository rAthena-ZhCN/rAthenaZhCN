payon,162,176,4	script	万圣节魔法师#iRO08	704,{
	if (Hallow08 < 1) {
		mes "[万圣节魔法师]";
		mes "嘿嘿嘿嘿嘿！";
		mes "这个万圣节我准备了一项特别活动，可以测试你的运气和敏捷。";
		mes "你有兴趣吗？";
		next;
		mes "[万圣节魔法师]";
		mes "来吧！别这么胆小！";
		mes "只要收集足够的票券，就能获得很棒的奖品！";
		mes "你觉得怎么样？";
		next;
		goto MainMenu;
	}
	if (Hallow08Kill == 1) {
		mes "[万圣节魔法师]";
		mes "你是个胆小鬼还是什么？！";
		mes "你知道你想再试一次……";
		mes "你知道规则吗？";
		next;
		set Hallow08Kill,0;
		if(select("知道。:不知道。") == 2) {
			goto Rules;
		}
		goto Participate;
	}
	if (Hallow08Kill == 2) {
		mes "[万圣节魔法师]";
		mes "哦，干得好！你挺厉害的！";
		mes "和腐尸一起玩不是很有趣吗？？";
		mes "你知道吗，腐尸生前也是人！";
		mes "哈！";
		mes "Kkkkkkk.";
		next;
		getitem 7941,1;
		set Hallow08Kill,0;
		mes "[万圣节魔法师]";
		mes "就像我承诺的那样";
		mes "你可以用万圣节票券换取很棒的物品。";
		close;
	}
	if (Hallow08 > 0) {
		mes "[万圣节魔法师]";
		mes "那么，你是想再听一遍规则，还是直接继续……";
		next;
		goto MainMenu2;
	}

		Rules:
			mes "[万圣节魔法师]";
			mes "这个村庄就像是虚拟的裴扬。";
			mes "周围有腐尸和绿腐尸游荡，南边有三个出口，但只有一个能走通。";
			mes "那就靠你自己去发现了。";
			next;
			mes "[万圣节魔法师]";
			mes "你不能使用任何技能杀死绿腐尸或腐尸。";
			mes "还有一件事……";
			mes "你可别忘了……";
			next;
			mes "[万圣节魔法师]";
			mes "所有参赛者都不能穿戴任何装备。";
			mes "把所有物品放进仓库，等负重为'0'时再回来这里。";
			next;
			mes "[万圣节魔法师]";
			mes "哦，还有一件事！";
			mes "你不能骑乘大嘴鸟，也不能携带手推车。";
			mes "如果你有，我会在你进入前帮你解除。";
			mes "明白了吗？";
			next;
			mes "[万圣节魔法师]";
			mes "记住，有三个出口，但只有一个会随机生效；在那里游荡的腐尸和绿腐尸无法被杀死，而且你不能穿戴任何装备。";
			next;
			if (Hallow08 > 0) {
				mes "[万圣节魔法师]";
				mes "Hey...";
				mes "准备好了就回来。";
				close;
			} else {
				goto MainMenu;
			}

		Participate:
			mes "[万圣节魔法师]";
			mes "好，你准备好了。";
			mes "让我检查一下你的负重。";
			next;
			if (Weight > 0) {
				mes "[万圣节魔法师]";
				mes "天啊！";
				mes "不管在哪里，总会有害群之马。";
				next;
				mes "[万圣节魔法师]";
				mes "你以为我会没注意到你的负重超过'0'吗？";
				mes "你超重了……";
				close;
			} else {
				mes "[万圣节魔法师]";
				mes "看来你可以出发了，负重刚刚好。";
				next;
				mes "[万圣节魔法师]";
				mes "我工作忙得不可开交，所以累坏了。";
				next;
				mes "[万圣节魔法师]";
				mes "我有时会忘记把你送过去……";
				next;
				mes "[万圣节魔法师]";
				mes "希望你平安归来。";
				close2;
				if (Hallow08 < 1) {
					set Hallow08,1;
				}
				set Hallow08Kill,1;
				set @Hallow08Warp, rand(1,3);
				percentheal -98,0;
				setriding 0;
				setcart 0;
				warp "evt_zombie",155,246;
				end;
			}

		TicketExchange:
			mes "[万圣节魔法师]";
			mes "你想用票券兑换奖品吗？";
			mes "干得好！嘿嘿嘿嘿！";
			next;
			mes "[万圣节魔法师]";
			mes "我来告诉你能兑换哪些物品。";
			next;
			mes "[万圣节魔法师]";
			mes "5张票券兑换甜蜜的南瓜派。";
			mes "20张票券兑换南瓜头饰。";
			mes "50张票券兑换神秘箱子。";
			mes "70张票券兑换神秘紫箱。";
			mes "200张票券兑换老旧卡册。";
			next;
			mes "[万圣节魔法师]";
			mes "你想兑换什么？";
			next;
			switch(select("甜蜜的南瓜派:南瓜头饰:神秘箱子:神秘紫箱:老旧卡册")) {
			case 1:
				if (countitem(7941) < 5) {
					goto NotEnough;
				} else {
					delitem 7941,5;
					getitem 12192,1;
					goto Enough;
				}
				break;
			case 2:
				if (countitem(7941) < 20) {
					goto NotEnough;
				} else {
					delitem 7941,20;
					getitem 5134,1;
					goto Enough;
				}
				break;
			case 3:
				if (countitem(7941) < 50) {
					goto NotEnough;
				} else {
					delitem 7941,50;
					getitem 603,1;
					goto Enough;
				}
				break;
			case 4:
				if (countitem(7941) < 70) {
					goto NotEnough;
				} else {
					delitem 7941,70;
					getitem 617,1;
					goto Enough;
				}
				break;
			case 5:
				if (countitem(7941) < 200) {
					goto NotEnough;
				} else {
					delitem 7941,200;
					getitem 616,1;
					goto Enough;
				}
			}

		NextTime:
			mes "[万圣节魔法师]";
			mes "好，到时候见。";
			mes "Kkkkkkkk.";
			close;

	MainMenu:
		menu "向我说明。",Rules,"我想参加。",Participate,"我想用票券兑换奖品。",TicketExchange,"I'll come back next time.",NextTime;

	MainMenu2:
		menu "现在送我回去！",Participate,"请告诉我规则",Rules,"我想用票券兑换奖品。",TicketExchange,"I'll come back next time.",NextTime;

	NotEnough:
		mes "[万圣节魔法师]";
		mes "你的票券不够！";
		mes "你连数都不会数吗？";
		mes "请带够票券再来。";
		close;

	Enough:
		mes "[万圣节魔法师]";
		mes "给你。";
		mes "你还需要……";
		mes "别的什么吗？";
		next;
		if (Hallow08 == 1) {
			goto MainMenu2;
		} else {
			goto MainMenu;
		}
}
