//============================================================
alberta,26,243,4	script	史蒂芬	58,{
	mes "[史蒂芬]";
	mes "猜猜我有什么？";
	mes "一种在米德加尔特不容易找到的美味……";
	next;
	mes "[史蒂芬]";
	mes "巧克力！";
	mes "没错，你不喜欢巧克力吗……我很喜欢。";
	mes "你运气不错，因为我卖得便宜，一块只要5,000 Zeny！";
	next;
	if(select("我想要巧克力！:不用了，谢谢。")==2) {
		mes "[史蒂芬]";
		mes "你一点巧克力都不想要？";
		mes "我跟你说！你会后悔的！";
		mes "最好现在就买一些……以后再也碰不到这样的巧克力了！";
		mes "好好考虑一下，改天再来找我吧。";
		close; }
	mes "[史蒂芬]";
	mes "哈！";
	mes "我就知道！";
	mes "不过我一次最多只能卖你5块……但如果你真的需要更多……";
	mes "你可以再来一趟。";
	mes "那么你想要几块？";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[史蒂芬]";
	if (.@i > 5)  {
		mes "抱歉，我不能给你那么多。";
		close; }
	if (Zeny < .@i*5000) {
		mes "抱歉，看来你买不起我这些巧克力。";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "给你！";
	mes "你可以把它当作礼物送给别人，也可以自己享用！";
	mes "嗯……甜甜的巧克力……";
	mes "随时来找我……！";
	close;
}
