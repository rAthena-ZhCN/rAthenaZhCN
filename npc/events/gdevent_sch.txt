schg_dun01,1,2,1	script	怪物控制员1#sch	81,{
OnEnable:
	set .@callwhere,rand(1,4);
	if (.@callwhere == 1) {
		monster "schg_dun01",164,236,"Kublin",1980,1,"怪物控制员1#sch::OnMyMobDead";
	}
	else if (.@callwhere == 2) {
		monster "schg_dun01",172,122,"Kublin",1980,1,"怪物控制员1#sch::OnMyMobDead";
	}
	else if (.@callwhere == 3) {
		monster "schg_dun01",247,159,"Kublin",1980,1,"怪物控制员1#sch::OnMyMobDead";
	}
	else {
		monster "schg_dun01",250,224,"Kublin",1980,1,"怪物控制员1#sch::OnMyMobDead";
	}
	end;

OnKill:
	killmonster "schg_dun01","怪物控制员1#sch::OnMyMObDead";
	end;

OnMyMobDead:
	if (mobcount("schg_dun01","Monster Controler1#sch::OnMyMObDead") == 0) {
		mapannounce "schg_dun01", "库布林：啊！",bc_map,"0x99CC00";
		mapannounce "schg_dun01", "摩尔斯通：我好像听到库布林在惨叫！！谁在那儿？库布林怎么了？喂，你！",bc_map,"0x99CC00";
		donpcevent "矮人#sch_gd::OnEnable";
	}
	end;
}
