//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	催眠教师	124,{
	set @npcname$,"[^D5A500Hypnotist^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "我已经告诉过你，这个活动只能完成一次。";
		close;
	}
	if(BaseLevel < 60) {
		mes "请在基础等级达到60级或以上后再来。";
		close;
	}
	if(SkillPoint != 0) {
		mes "如果你想让我继续，就必须把所有技能点用完。";
		mes "请尽快再来！";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "你想重置技能吗？";
		mes "抱歉，但是……";
		next;
		mes @npcname$;
		if(Weight) {
			mes "你无法重置技能，";
			mes "只要你身上";
			mes "还带着任何道具。";
		} else if(checkcart())
			mes "请先卸下你的手推车，我们再继续。";
		else if(checkfalcon())
			mes "请先放飞你的猎鹰，我们再继续。";
		else if(checkriding())
			mes "请先放走你的大嘴鸟，我们再继续。";
		next;
		mes @npcname$;
		mes "请尽快回来！";
		close;
	}
	mes "你好，我是技能重置员。";
	mes "你的名字是^0080FF"+strcharinfo(0)+"^000000。";
	mes "我能帮你什么？";
	next;
	menu "^009500关于技能重置的说明。^000000",L_Info,"^00B6FF重置技能。^000000",L_Reset,"^000088算了^000000",-;

	mes @npcname$;
	mes "你知道在哪里能找到我，";
	mes "如果你想重置技能的话！！"; 
	close;

L_Info:
	mes @npcname$;
	mes "技能重置并不是免费的！！";
	mes "技能重置费用为^D5A50020000 Zeny × 基础等级^000000。";
	mes "是的……每一点基础等级需要20000 Zeny才能重置技能。";
	next;
	mes @npcname$;
	mes "对了，还有一件事！";
	mes "你装备的手推车、猎鹰或大嘴鸟";
	mes "在重置技能后都会被移除。";
	next;
	mes @npcname$;
	mes "技能点只能再重置一次。";
	mes "今后请谨慎分配你的技能点。";
	close;
	
L_Reset:
	mes @npcname$;
	mes "在开始重置技能之前，";
	mes "你必须先告诉我你的基础等级。"; 
	next;
	mes "^D5A500[" + strcharinfo(0) + "]^000000";
	mes "我的基础等级是^AA00AA" + BaseLevel + "BaseLv.^000000"; 
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "技能重置所需的总金额为^529DFF" + @zeny + "Zeny^000000，用于技能重置服务。";
	next;
	if(Zeny < @zeny) {
		mes @npcname$;
		mes "看来你的钱不够。";
		mes "那就等下次机会吧。";
		emotion ET_SCRATCH;
		close;
	}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "谢谢。";
	emotion ET_THANKS;
//	logmes "SKILL RESET EVENT";
	close;
}
