mosk_in,21,246,5	script	品卡梅尼亚	71,{
	set .@n$,"^0000FF[Pinkamenia]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "你好，"+strcharinfo(0)+"，要";
		mes "买些东西吗？";
		mes "你得用爱情之花";
		mes "来换取，当然啦！";
		next;
		if(select("好，没问题！","不，不用了。") == 2) close;
		next;
		mes .@n$;
		mes "请选择物品。";
		mes "所需的爱情之花数量显示在括号中。";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00["+.@SweetsCost[.@i]+"]^000000"+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "想要那个的话，你需要更多爱情之花！";
			close;
		}
		mes "你确定要用 ^00aa00"+.@SweetsCost[.@choice]+"x 爱情之花 ^000000 换取 ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000?";
		if(select("不，我改变主意了。:好，交易吧！") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "好好享受你的物品吧！";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "你必须给我带来 15 朵爱情之花！";
			close;
		}
		mes "非常感谢你";
		mes "帮我们找回这些花！";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "请收下这份小小的";
		mes "‘谢意’。";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "如果你得到更多爱情";
		mes "之花，就可以在这里";
		mes "把它们换成一些甜点。";
		mes "回头见！";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "请带一顶 +8 蛋糕帽";
		mes "去找烘焙大师";
		mes "他就站在我旁边！";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "你好，"+strcharinfo(0)+"，今天是";
		mes "情人节，我们";
		mes "很乐意送上甜蜜的";
		mes "chocolates.";
		next;
		mes .@n$;
		mes "我们的问题是，我们";
		mes "需要一个强壮的人";
		mes "像你这样来帮";
		mes "我们，但首先你得";
		mes "带一顶 +8 蛋糕帽";
		mes "去找烘焙大师，他";
		mes "就站在我旁边！";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "你好，"+strcharinfo(0)+"！";
	close;

OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{ bonus bDex,1; bonus bMaxSP,80; bonus3 bAddMonsterDropItem,7864,RC_DemiHuman,500; }",0;
	end;
}

mosk_in,21,244,3	script	烘焙大师	410,{
	set .@n$,"^0000FF[Baker Extraordinaire]^000000";
	if(#V_QUE12==3) {
		mes .@n$;
		mes "去和品卡梅尼亚交换爱情之花！";
		close;
	} else if(#V_QUE12==2) {
		mes .@n$;
		mes "带 15 朵爱情之花给品卡梅尼亚！";
		close;
	} else if(#V_QUE12==1) { 
		mes .@n$;
		if (getequipid(EQI_HEAD_TOP) == 5024 && getequiprefinerycnt(EQI_HEAD_TOP) >= 8) {
			mes "啊，原来品卡梅尼亚让你";
			mes "把 +8 蛋糕";
			mes "帽带给我。现在我";
			mes "就用你的 +8 蛋糕";
			mes "帽换另一顶蛋糕帽。";
			next;
			mes .@n$;
			delitem 5024,1;
			getitem 5105,1;
			equip 5105;
			mes "现在，如果你戴上";
			mes "新的蛋糕帽，就";
			mes "有机会让";
			mes "亚人型魔物";
			mes "掉落爱情之花！";
			next;
			mes .@n$;
			mes "那些魔物偷走了我们的花，";
			mes "我们需要用它来制作";
			mes "我们的巧克力。现在去";
			mes "收集 15 朵爱情之花，";
			mes "带去给品卡梅尼亚！";
			set #V_QUE12,2;
			close;
		} else {
			mes "抱歉，你的";
			mes "+8 蛋糕帽呢？把它";
			mes "带来给我，记得";
			mes "要装备上！";
			close;
		}
	} else {
		mes .@n$;
		mes "你好，"+strcharinfo(0)+"！";
		close;
	}
}
