prontera,147,61,3	script	德里勒#pron	87,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你背包里的物品太多了。请使用卡普拉仓库后再试。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你负重过高，无法继续。";
		close;
	}
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[德里勒]";
		mes "我的工作是在 RWC 纪念饰品上钻出卡片插槽。";
		next;
		mes "[德里勒]";
		mes "很抱歉，你的右侧饰品栏没有装备任何物品。";
		close;
	}
	mes "[德里勒]";
	mes "我的工作是在 RWC 纪念饰品上钻出卡片插槽。而且我只处理 ^ff0000纯净的物品^000000，也就是尚未附魔过的物品。";
	next;
	set .@equip_id, getequipid(.@part);
	if (.@equip_id != 2966 && .@equip_id != 2968) {
		mes "[德里勒]";
		mes "不过，我看你右侧戴着的饰品无法处理。请装备 RWC 纪念饰品。";
		close;
	}
	mes "[德里勒]";
	mes "另外你要知道，钻开卡片插槽极其危险。^ff0000成功率约为 50%。^000000 你还要继续吗？";
	next;
	if(select("取消:开始吧！") == 1) {
		mes "[德里勒]";
		mes "那就再见了。";
		close;
	}
	if (.@equip_id == 2966) {
		set .@slotted, 2967; //RWC_2012_Ring_
		set .@name$,"RWC 2012 Memorial Ring";
		set .@str$,"ring";
	} else if (.@equip_id == 2968) {
		set .@slotted, 2969; //RWC_2012_Pendant_
		set .@name$,"RWC 2012 Memorial Pendant";
		set .@str$,"pendant";
	} else {
		mes "[德里勒]";
		mes "我无法鉴定你右手上戴的饰品。我不能处理它。";
		close;
	}
	if (getequipcardid(.@part,3) > 0) {
		mes "[德里勒]";
		mes "这件物品已经附过魔了。按规定我不能处理它。";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id))
		close;

	delequip .@part;
	if (rand(1,10) > 5) {
		getitem .@slotted,1;
		specialeffect2 EF_REPAIRWEAPON;
		mes "[德里勒]";
		mes "太好了！成功了！你的"+.@name$+"现在有了卡片插槽。去看看吧！";
		close;
	} else {
		specialeffect2 EF_LORD;
		mes "[德里勒]";
		mes "哎呀……真脆弱的"+.@str$+"... It broke during the procedure. I'm sorry.";
		close;
	}
}

prontera,147,59,3	script	戈德堡#pron	878,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你携带的物品太多了。请整理好背包后再来。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你负重过高，无法继续。";
		close;
	}
	mes "[戈德堡]";
	mes "你好！我负责用神秘的力量为 RWC 纪念饰品附魔。";
	next;
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[戈德堡]";
		mes "很抱歉，你的右侧饰品栏没有装备任何物品。";
		close;
	}
	set .@equip_id, getequipid(.@part);
	if (.@equip_id < 2966 || .@equip_id > 2969) {
		mes "[戈德堡]";
		mes "不过，我看你戴着的饰品不是我能够处理的。请装备 RWC 纪念饰品。";
		close;
	}
	set .@select, select("抱歉，我不感兴趣。:请为我的饰品赋予力量。:移除附魔。")-1;
	if (.@select == 0) {
		mes "[戈德堡]";
		mes "那么，下次再见……";
		close;
	}
	setarray .@equip_card[0], getequipcardid(.@part,0),getequipcardid(.@part,1),getequipcardid(.@part,2),getequipcardid(.@part,3);
	if (.@select == 1) {
		switch(getequipid(.@part)) {
		case 2966:
			setarray .@option[0],2,2,4,4;
			break;
		case 2967:
			setarray .@option[0],0,2,4,4;
			break;
		case 2968:
			setarray .@option[0],1,1,3,3;
			break;
		case 2969:
			setarray .@option[0],0,1,3,3;
			break;
		default:
			mes "[戈德堡]";
			mes "很抱歉，我无法处理你现在戴着的饰品。";
			mes "如果你把 RWC 纪念饰品装备在左侧，请试着换到右侧。";
			close;
		}
		for(set .@i,3; .@i>=0; set .@i,.@i-1) {
			if (.@equip_card[.@i] == 0) {
				set .@slot, .@i;
				set .@op_type, .@option[.@i];
				break;
			}
		}
		switch(.@op_type) {
		case 4:
			mes "[戈德堡]";
			mes "你想注入哪种附魔？";
			next;
			setarray .@enchant_select[0],1,2,3,4;
			set .@i, select("取消:提升斗志:ATK (%):最大HP:HP")-2;
			break;
		case 3:
			setarray .@enchant_select[0],5,6,7;
			set .@i, select("取消:魔法:MATK (%):SP")-2;
			break;
		case 2:
			mes "[戈德堡]";
			mes "^ff0000小心！附魔有约 25% 的几率失败。一旦失败，物品将会被破坏。^000000 你想注入哪种附魔？";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,14;
			set .@i, select("取消:STR:AGI:VIT:INT:DEX:LUK:SP")-2;
			break;
		case 1:
			mes "[戈德堡]";
			mes "^ff0000附魔有约 25% 的几率失败。一旦失败，物品将会被破坏。^000000 你想注入哪种附魔？";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,15,16;
			set .@i, select("取消:STR:AGI:VIT:INT:DEX:LUK:MHP:HP")-2;
			break;
		case 0:
			mes "[戈德堡]";
			mes "你的饰品已经接受了太多附魔，我几乎无法再处理了。";
			close;
		}
		if (.@i == -1) {
			mes "[戈德堡]";
			mes "那么，下次再见。";
			close;
		}
		mes "[戈德堡]";
		mes "附魔的力量将随机决定。^ff0000附魔一旦注入，就无法移除。^000000 要继续吗？";
		next;
		if(select("不，请停下。:是，请继续。") == 1) {
			mes "[戈德堡]";
			mes "那么，下次再见……";
			close;
		}
		set .@enchant_type, .@enchant_select[.@i];
		if (!getequipisequiped(.@part)) {
			mes "[戈德堡]";
			mes "我工作的时候请不要脱下装备，好吗？";
			close;
		}
		switch(.@enchant_type) {
		case 1:
			setarray .@enc[0],4811,4810,4809; //Fighting_Spirit1,Fighting_Spirit2,Fighting_Spirit3
			break;
		case 2:
			setarray .@enc[0],4819,4766,4767; //Atk1,Atk2,Atk3
			break;
		case 3:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 4:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		case 5:
			setarray .@enc[0],4760,4761,4806; //Matk1,Matk2,Matk3
			break;
		case 6:
			setarray .@enc[0],4815,4814,4813; //Spell1,Spell2,Spell3
			break;
		case 7:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 8:
			setarray .@enc[0],4700,4701,4702; //Strength1,Strength2,Strength3
			break;
		case 9:
			setarray .@enc[0],4730,4731,4732; //Agility1,Agility2,Agility3
			break;
		case 10:
			setarray .@enc[0],4740,4741,4742; //Vitality1,Vitality2,Vitality3
			break;
		case 11:
			setarray .@enc[0],4710,4711,4712; //Inteligence1,Inteligence2,Inteligence3
			break;
		case 12:
			setarray .@enc[0],4720,4721,4722; //Dexterity1,Dexterity2,Dexterity3
			break;
		case 13:
			setarray .@enc[0],4750,4751,4752; //Luck1,Luck2,Luck3
			break;
		case 14:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 15:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 16:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		default:
			mes "[戈德堡]";
			mes "出了问题呢，让我检查一下。";
			close;
		}

		if (.@enchant_type < 8)
			set .@i, rand(1,300); // 0% break chance.
		else
			set .@i, rand(1,400); // 25% break chance.

		     if (.@i < 151) set .@enchant, .@enc[0];
		else if (.@i < 251) set .@enchant, .@enc[1];
		else if (.@i < 301) set .@enchant, .@enc[2];
		else set .@enchant,9;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		    callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]))
			close;

		set .@equip_card[.@slot], .@enchant;
		if (.@slot == 2 && .@enchant == 0) {
			set .@equip_card[3],0;
		} else if (.@slot == 1 && .@enchant == 0) {
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		} else if (.@slot == 0 && .@enchant == 0) {
			set .@equip_card[1],0;
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		}

		set .@equip_refine, getequiprefinerycnt(.@part);
		delequip .@part;
		if (.@enchant == 9) {
			mes "[戈德堡]";
			mes "哦，天哪！";
 			mes "这件物品承受不住附魔的力量，因此被破坏了。很抱歉。";
			specialeffect2 EF_LORD;
			close;
		}
		if (.@enchant == 0) { // Should never happen.
			mes "[戈德堡]";
			mes "哦……看来注入的各种力量之间出现了某种不稳定。这导致所有附魔都消失了。真遗憾，不过请再试一次吧！";
		} else {
			mes "[戈德堡]";
			mes "太好了！";
			mes "附魔成功了！它将附加在插槽编号^990000"+(.@slot+1)+"^000000。";
			specialeffect2 EF_REPAIRWEAPON;
		}

//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];

		close;
	} else if (.@select == 2) {
		mes "[戈德堡]";
		mes "我只会重置附魔选项，不会对已插卡的卡片做任何处理。你想继续吗？";
		next;
		if(select("我要停下。:好，请继续。") == 1) {
			mes "[戈德堡]";
			mes "改变主意了再来吧。";
			close;
		}
		if (countitem(6665) == 0) {
			mes "[戈德堡]";
			mes "很抱歉。你没有 RWC 初始化券。可以检查一下背包吗？";
			close;
		}
		if (getiteminfo(.@equip_card[3], ITEMINFO_SUBTYPE) != CARD_ENCHANT) {
			mes "[戈德堡]";
			mes "嗯……这件装备干干净净。什么都没有，我可没法初始化！再检查一下吧。";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[戈德堡]";
		mes "你物品上的附魔选项将被初始化。";
		delitem 6665,1; //RWC_Inicializer
		delequip .@part;
	
//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		for ( .@i = getiteminfo(.@equip_id, ITEMINFO_SLOT); .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];
		
		close;
	}
}
