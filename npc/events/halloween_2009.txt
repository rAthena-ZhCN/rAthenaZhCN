prontera,152,192,5	script	南瓜帽研究员	712,{
	if(BaseLevel < 45) {
		mes "[南瓜帽研究员]";
		mes "走开，我不需要小孩。走开！我不和初心者说话。";
		next;
		mes "[南瓜帽研究员]";
		mes "等你达到能与更强魔物战斗的等级再回来。";
		close;
	}
	mes "[南瓜帽研究员]";
	mes "对了，你喜欢甜蜜的南瓜派吗？";
	next;
	mes "[南瓜帽研究员]";
	mes "我是南瓜帽研究员，你为什么不听听我的故事呢？";
	next;
	while(1) {
		switch(select("听故事。:询问南瓜帽。:获得南瓜帽。:结束对话。")) {
			case 1:
				mes "[南瓜帽研究员]";
				mes "我一直在研究升级版的南瓜帽。";
				mes "我发现过程非常简单。";
				next;
				mes "[南瓜帽研究员]";
				mes "过程相当简单。";
				mes "如果你带^4a4aff20杰克南瓜^000000来给我，我就示范给你看。";
				mes "这个提议是不是很诱人？";
				next;
				break;
			case 2:
				mes "[南瓜帽研究员]";
				mes "这个升级版南瓜帽可是很厉害的东西！";
				next;
				mes "[南瓜帽研究员]";
				mes "戴上它的人可以利用凝聚的能量，做出回复大量HP与SP的甜蜜的南瓜派。";
				next;
				mes "[南瓜帽研究员]";
				mes "你只需要^4a4aff20杰克南瓜^000000。";
				next;
				break;
			case 3:
				mes "[南瓜帽研究员]";
				mes "你想要甜蜜的南瓜派吗？好，让我数数你带来的杰克南瓜。";
				next;
				if(countitem(1062) < 20) {
					mes "[南瓜帽研究员]";
					mes "我需要^4a4aff20杰克南瓜^000000。";
					mes "我不是炼金术士，也不是巫师，没法凭空创造东西。";
					next;
					mes "[南瓜帽研究员]";
					mes "好，去狩猎魔物再回来吧。";
					mes "我会在这里待一阵子，你慢慢来。";
					next;
					break;
				}else{
					mes "[南瓜帽研究员]";
					mes "希望这对你有用。战斗时别忘了戴上它，这样才能获得甜蜜的南瓜派。";
					delitem 1062,20;	// Jack o' Pumpkin
					getitem 5668,1;		// Weird Pumpkin Hat
					next;
					mes "[南瓜帽研究员]";
					mes "我想我该回去继续研究了。";
					close;
				}
			case 4:
				mes "[南瓜帽研究员]";
				mes "再见，后会有期。";
				mes "祝你好运……";
				close;
		}
	}
}

-	script	Halloween Wizard#iRO09::09HallowWiz	737,{
	mes "[万圣节巫师]";
	mes "...";
	mes "你想捉弄别人吗？";
	next;
	while(1) {
		switch(select("什么把戏？:当然:不要。")) {
			case 1:
				mes "[万圣节巫师]";
				mes "只要用少量材料，我就能在世界其他地方召唤魔物。";
				next;
				mes "[万圣节巫师]";
				mes "听起来很有趣吧？";
				next;
				mes "[万圣节巫师]";
				mes "如果你带透明之布、杰克南瓜、破旧的布或破掉的南瓜头来，我就能召唤魔物。";
				next;
				break;
			case 2:
				mes "[万圣节巫师]";
				mes "你想捉弄哪个城镇？";
				next;
				getmapxy(.@mapname$,.@mapx,.@mapy,BL_PC,""+strcharinfo(0)+"");
				if (.@mapname$ == "prontera") {
					switch(select("吉芬:裴扬:阿尔贝塔:艾尔帕兰")) {
						case 1:
							set .@HallowTown,3;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "payon") {
					switch(select("普隆德拉:吉芬:阿尔贝塔:艾尔帕兰")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "geffen") {
					switch(select("普隆德拉:裴扬:阿尔贝塔:艾尔帕兰")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "alberta") {
					switch(select("普隆德拉:吉芬:裴扬:艾尔帕兰")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "aldebaran") {
					switch(select("普隆德拉:吉芬:裴扬:阿尔贝塔")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,4;
							break;
					}
				}
				setarray .@HallowTowns$[1],"prontera","payon","geffen","alberta","aldebaran";
				mes "[万圣节巫师]";
				mes "好，那我们就进行下一步。";
				next;
				mes "[万圣节巫师]";
				mes "你想用多少透明之布或杰克南瓜？不要超过100，因为那是我能使用的上限。";
				next;
				input .@input;
				if (.@input == 0) {
					mes "[万圣节巫师]";
					mes "你还没想好。";
					mes "这没什么大不了的。";
					mes "想好了告诉我。";
					next;
					break;
				}
				else if (.@input > 100) {
					mes "[万圣节巫师]";
					mes "我告诉过你，数量必须在1到100之间！";
					mes "你根本没认真听！";
					next;
					break;
				}
				else {
					set .@fabric,countitem(1059);
					set .@jack,countitem(1062);
					set .@worn,countitem(6299);
					set .@crushed,countitem(6298);
					set .@whispers,0;
					set .@darklords,0;

					set .@total,.@fabric + .@jack + .@worn + .@crushed;

					if(.@total < .@input) {
						mes "[万圣节巫师]";
						mes "重新数一遍你拥有的物品数量，然后告诉我总数。";
						mes "Huhuhuhuhuhu...";
						next;
						break;
					}

					if(.@fabric > 0) {
						if(.@fabric >= .@input) {
							delitem 1059,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 1059,.@fabric;
							set .@input,.@input - .@fabric;
							set .@whispers,.@whispers+.@fabric;
						}
					}
					if(.@worn > 0 && .@input != 0) {
						if(.@worn >= .@input) {
							delitem 6299,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 6299,.@worn;
							set .@input,.@input - .@worn;
							set .@whispers,.@whispers+.@worn;
						}
					}
					if(.@jack > 0 && .@input != 0) {
						if(.@jack >= .@input) {
							delitem 1062,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 1062,.@jack;
							set .@input,.@input - .@jack;
							set .@darklords,.@darklords+.@jack;
						}
					}
					if(.@crushed > 0 && .@input != 0) {
						if(.@crushed >= .@input) {
							delitem 6298,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 6298,.@crushed;
							set .@input,.@input - .@crushed;
							set .@darklords,.@darklords+.@crushed;
						}
					}
					if (.@input > 0) {
						mes "有个问题。";
						close;
					}
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Whisper",3014,.@whispers;
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Dark Lord",3015,.@darklords;
					mes "[万圣节巫师]";
					mes "这是你想要的东西。";
					mes "想象一下其他村庄的人会怎么想？";
					mes "哇哈哈哈哈";
					close;
				}
			case 3:
				mes "[万圣节巫师]";
				mes "如果你改变主意，就回来这里……";
				mes "我会在这里待一阵子……";
				mes "Kkkk...";
				close;
		}
	}
}
