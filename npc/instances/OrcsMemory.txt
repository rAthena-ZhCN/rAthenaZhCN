gef_fild10,242,202,0	script	次元峡谷碎片	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Orc's Memory";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "只有等级 ^ff000030 ~ 80^000000 的玩家才能进入此副本。";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "队伍状态已确认。您要预约进入"+.@md_name$+"吗？";
			next;
			switch(select("预约进入"+.@md_name$+":进入副本:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "队伍名称："+ getpartyname(.@party_id);
					mes "队长："+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 - 预约失败。";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000- 正在尝试预约入口";
				mes "完成预约后，若要进入"+.@md_name$+".";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name$ + "进入回忆副本:取消") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "副本生成后即可进入。";
		next;
		if(select("进入副本"+.@md_name$+":取消") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ff与兽人记忆副本相关的所有记录和后续影响均已删除。现在可以重新生成或再次进入副本。^000000";
		erasequest 12059;
		close;
	}
	mes "要生成副本，您必须是队长，且队伍中至少有 2 名成员。";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "发生了未知错误。";
		close;
	case IE_NOINSTANCE:
		mes "不存在名为兽人记忆的回忆副本。";
		mes "回忆副本已被队长销毁，或因超过时间限制而消失。请在 2 小时后再试。";
		close;
	case IE_NOMEMBER:
		mes "只有队伍成员才能进入回忆副本。";
		close;
	case IE_OK:
		mapannounce "gef_fild10", getpartyname( getcharid(1) ) +"的队伍成员"+strcharinfo(0)+"已进入兽人记忆。",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}

1@orcs,180,30,0	script	#Resurrect Monsters1	844,{
	end;

OnInstanceInit:
	disablenpc instance_npcname("#Resurrect Monsters1");
	end;

OnDisable:
	stopnpctimer;
	killmonster instance_mapname("1@orcs"),instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

OnEnable:
	enablenpc instance_npcname("#Resurrect Monsters1");
	monster instance_mapname("1@orcs"),0,0,"Orc Warrior",1023,30,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

OnMyMobDead:
	set .@map$, instance_mapname("1@orcs");
	set .@mob_dead_num, 30 - mobcount(.@map$,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead");
	if (.@mob_dead_num > 0) {
		set .@mob_ran,rand(1,30);
		if (.@mob_ran > 29) {
			monster .@map$,0,0,"Orc Warrior",1023,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
		}
		else if ((.@mob_ran > 28) && (.@mob_ran < 30)) {
			monster .@map$,0,0,"High Orc",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "高等兽人：我们需要更多防御！再叫些人过来！",bc_map,"0xff4444";
			}
		}
		else if ((.@mob_ran > 26) && (.@mob_ran < 29)) {
			areamonster .@map$,41,91,51,81,"High Orc",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "高等兽人在哪！？叫他们去挡住敌人！",bc_map,"0xff4444";
			}
		}
		else {
			areamonster .@map$,17,187,27,177,"High Orc",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,5) == 3) {
				mapannounce .@map$, "注意：敌军开始向第 4 区集结。",bc_map,"0x77ff77";
			}
			if (rand(1,100) == 50) {
				initnpctimer;
			}
		}
	}
	end;

OnTimer10:
	mapannounce instance_mapname("1@orcs"), "守护兽人首领的喊声：看来这比预想的要更久。召唤钟乳石巨人！",bc_map,"0xff4444";
	end;

OnTimer4010:
	set .@map$, instance_mapname("1@orcs");
	mapannounce .@map$, "钟乳石巨人正从地下深处掘土而出。",bc_map,"0x77ff77";
	areamonster .@map$,17,187,27,177,"Stalactic Golem",1278,20,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	stopnpctimer;
	end;
}
