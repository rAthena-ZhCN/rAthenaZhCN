payon,173,238,5	script	杰德#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FF等一下！你现在携带的物品太多了。请先把一些东西放进卡普拉仓库，再回来吧。^000000";
		close;
	}
	mes "[杰德]";
	mes "给我拿两个";
	mes "颜色相同的";
	mes "魔力矿石来，我就";
	mes "帮你换成另一种";
	mes "颜色的魔力矿石。";
	next;
	switch(select("蓝色魔力矿石换成红色！:红色魔力矿石换成黄色！:黄色魔力矿石换成蓝色！")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[杰德]";
		mes "哈……！";
		mes "你在开玩笑吧？";
		mes "这项服务，我没办法";
		mes "提供给你，除非";
		mes "你至少给我";
		mes "2 "+ getitemname(.@item_req) +"！";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[杰德]";
		mes "我想我能做出";
		mes "总共"+ .@gems +" "+ getitemname(.@item_id) +"，";
		mes "用的是你现有的"+ getitemname(.@item_req) +"。";
		mes "这些你身上就有。";
		mes "你想怎么做？";
		next;
		switch(select("能换多少就换多少。:我想指定数量。:我放弃。")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[杰德]";
			mes "那么，你想要";
			mes "多少个？";
			mes "最多可以";
			mes "输入 100 个。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[杰德]";
					mes "一个都不要？";
					mes "看来你是";
					mes "改变主意了……";
					close;
				}
				else if (.@input > 100) {
					mes "[杰德]";
					mes "Errm...";
					mes "我说过，请输入";
					mes "不超过";
					mes "100 的数量，记得吗……？";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[杰德]";
					mes "Errm...";
					mes "你没有那么多";
					mes "宝石可以交换……";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[杰德]";
			mes "当然，没问题。";
			mes "随时欢迎再来。";
			close;
		}
		mes "[杰德]";
		mes "给你。";
		mes "随时欢迎你";
		mes "再来光临。";
		mes "嗯，你那是什么眼神？";
		mes "我脸上有什么东西吗？";
		close;
	}
}
