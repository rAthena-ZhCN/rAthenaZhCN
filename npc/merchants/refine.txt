//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "我就是武器锻造师。";
	mes "各种武器、防具和装备我都能精炼，所以";
	mes "请告诉我您想精炼什么。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + callfunc("F_EquipNameCN", getequipid(.@indices[.@i])) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "您身上的物品，我恐怕一件也精炼不了……";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "那个部位您没有";
		mes "穿戴任何";
		mes "我能精炼的东西。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我恐怕";
		mes "完全没法精炼这件物品……";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "这件东西我";
		mes "没法再精炼了。这已经";
		mes "是精炼的极限了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "要精炼这个，我需要";
		mes "一个 ^003366"+getitemname(.@material)+"^000000 以及";
		mes "手续费" + .@price + "Zeny。";
		mes "您真的要继续吗？";
		next;
		if(select("是:否") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "Yeah...";
			mes "不用";
			mes "着急，慢慢来。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "不好！如果我继续";
			mes "精炼下去，它有可能";
			switch(.@material) {
			case 985:
				mes "被破坏！也就是说，^FF0000这件装备^000000，以及附加在这件防具上的 ^FF0000所有卡片^000000 或特殊属性，^FF0000都会消失^000000。";
				break;
			default:
				mes "被破坏，您会^FF0000失去这把武器^000000，以及武器上的^FF0000所有卡片^000000，";
				mes "还有附加的特殊属性。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "我已经说得再清楚不过了。";
			mes "武器一旦被破坏，";
			mes "就再也拿不回来了。";
			mes "您确实有可能";
			mes "^FF0000永远失去这把武器^000000。";
			mes "您还想精炼吗？";
			next;
			if(select("是:否") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "我完全同意……";
				mes "我或许是个了不起的精炼师，但有时候连我也会出错。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "您似乎没有";
			mes "足够的 Zeny 或"+getitemname(.@material)+"...";
			mes "再去多弄一些吧。我整天";
			mes "都在这儿，需要就来找我。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "您以为我傻吗？！";
			mes "您趁我不注意把东西换了！给我滚出去！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "天哪！";
				mes "可恶！又来一次！";
				mes "真是对不起，不过您也知道，熟能生巧嘛。";
				mes "嗯，对吧？嘿嘿……";
			} else if(.@lose == 2) {
				mes "不——要——啊！";
				mes "它坏了！";
				mes "对、对不起！";
			} else {
				mes "糟了！";
				mes "它承受不住";
				mes "更多的淬炼了！";
				mes "这事真是抱歉……";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完美！";
			mes "嘿嘿！";
			mes "又一次，";
			mes "无懈可击的手艺";
			mes "出自大师之手~";
		} else if(.@win == 2) {
			mes "成功……！";
			mes "我的非凡";
			mes "才华今天";
			mes "又一次大放异彩。";
		} else {
			mes "嘿嘿！";
			mes "完成了。";
			mes "毫无疑问，我的手艺";
			mes "包您满意。";
			mes "完美无瑕~";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以把它精炼到安全上限，也可以按您想要的次数来。由您决定。";
		next;
		.@menu2 = select("请精炼到安全上限。","次数由我来定。","我改变主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "您想让我精炼几次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "这件物品没法精炼那么多次。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "这将尝试把装备精炼" + .@refinecheck + "次，超出安全上限。您的装备可能会被破坏……可以吗？";
			next;
			if(select("Yes...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "是您说的……那就这样吧。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "是您说的……那就这样吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "这要花费您" + .@refinecnt + " " + getitemname(.@material) + "与" + .@fullprice + "Zeny。可以吗？";
	next;
	if(select("是","No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "是您说的……那就这样吧。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "您就这点钱吗？很遗憾，我没办法降价为您效劳。请设身处地替我想想。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "听好了……您身上什么都没穿戴……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "铛……不，您以为我会那么傻吗？！";
			mes "您把它换了……";
			mes "趁我还没用锤子把您敲晕，快滚！！";
			close;
		} 
		mes "铛，铛！！！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "哇啊啊啊！！！真是对不起……我警告过您可能会这样……";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "这是没用完的 Zeny 和材料，还给您……";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "都弄好了……欢迎再来。";
	close;
}

		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "其他金属？";

		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "那么您要买几个？";
	mes "如果什么都不想要，请输入数字“0”。";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "交易";
			mes "已取消。";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";

			mes "明白吗？很好。";
			next;
		}
		else {
			break;
		}
	}

	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "Err...";
		mes "您没有";
		mes "足够的 Zeny 来购买";
		mes ""+ .@input +" of them.";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "嘿，你好！";
	mes "您想让我";
	mes "修理什么物品吗？";
	mes "修理物品";
	mes "就交给我吧！";
	next;
	switch(select("其实我有几件要修……:暂时没有。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "哦，真了不起！";
			mes "您对自己的东西真是爱惜有加。您的物品一件都没坏！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "要是大家都像您这样，我就要失业了！！哈哈~！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "Hmm...";
		mes "我看看……";
		mes "在您所有物品中，";
		mes "" + .@checkitem + "已经损坏。";
		mes "要修理吗？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "每修一件需要" + .@repairprice + "Zeny。修好您所有损坏的物品总共需要" + .@totalcost + "Zeny！要修理这些物品吗？";
		next;
		switch(select("是:否")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "喂喂……";
				mes "收到修理账单之前，先看看您的钱包吧！您的 Zeny 不够，我什么也修不了。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "好！都修好了。以后稍微爱惜一点。物品也是有生命的，懂吗。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "嗯？有点不对劲。等等……请把要修理的物品穿戴在身上，再来找我。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "这事本来跟我无关，不过让物品一直坏着可不好。您最好尽快修好它们！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Hohoho...";
		mes "如果您没有";
		mes "要修的东西，";
		mes "那就没有";
		mes "别的事找我了。";
		close;
	}
}
