function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[赛亚布勒姆]";
	mes "您想为"+ getitemname(.@item_id) +"增加插槽吗？";
	mes "好的，请为我带来^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000，"+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000") : "") + "以及我的 ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 Zeny^000000 服务费。";
	mes "对了，别忘了带上"+ getitemname(.@item_id) +"！";
	next;
	mes "[赛亚布勒姆]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "如果您带齐了所需物品和Zeny，我现在就可以尝试增加插槽。";
		mes "不过您要知道，我有可能会失败。";
		mes "所以我必须事先提醒您……";
		next;
		mes "[赛亚布勒姆]";
		mes "如果武器增加插槽失败，那么^FF0000武器^000000与^FF0000其上镶嵌的所有卡片都会被销毁^000000。";
	}
	else // armor
	{
		mes "如果您备齐了所需材料、我的Zeny服务费以及那件防具，我们就可以开始增加插槽。";
		mes "但在这之前，我必须提醒您其中的风险。";
		next;
		mes "[赛亚布勒姆]";
		mes "如果防具增加插槽失败，那么^FF0000防具^000000、^FF0000其精炼等级^000000 ^FF0000以及其上镶嵌的所有卡片都会被销毁^000000。";
	}
	mes "您还想尝试增加插槽吗？";
	next;
	switch(select("尝试增加插槽:取消"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[赛亚布勒姆]";
				mes "那么，开始吧！";
				mes "您最好祈祷能成功。";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[赛亚布勒姆]";
					mes "太好了，看起来成功了。";
					mes "完成得很漂亮。恭喜您！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[赛亚布勒姆]";
					mes "再会，伙计！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[赛亚布勒姆]";
					mes "哇！……非常抱歉，失败了。";
					mes "不过，这完全不是我的错。";
					mes "这是您的运气，是上天注定的，好吗？";
					mes "别太失望，下次再试试吧。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[赛亚布勒姆]";
					mes "祝您下次好运！";
					close;
				}
			}
			else
			{
				mes "[赛亚布勒姆]";
				mes "我想继续这次增加插槽的尝试，但您还缺少一些东西。";
				mes "您确定装备、所需材料和Zeny都齐全吗？";
				close;
			}
		case 2:
			mes "[赛亚布勒姆]";
			mes "需要时间考虑一下，是吗？";
			mes "好的，我理解。";
			mes "请记住，总是小心翼翼，人生就没什么乐趣了~";
			close;
	}
}
