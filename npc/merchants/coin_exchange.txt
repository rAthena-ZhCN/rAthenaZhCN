-	script	::merchant_13_2	-1,{
	set .@n$, "["+strnpcinfo(1)+"]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "看来你带的东西太多了。";
			mes "何不先把一些物品放进仓库再回来？";
			close;
		}
		mes "Hello.";
		mes "你在找什么？";
		next;
		if (strnpcinfo(1) == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("查看物品说明:马努克的机遇:马努克的勇气:马努克的信仰:取消");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("查看物品说明:购买捕虫堇的果实酱:购买萤火虫的蜂蜜酱:购买山茱萸的眼泪:什么也不做");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "这是物品说明。";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "我可以卖给你 3 个"+getitemname(.@Items[.@i])+"，只要 ^3131FF1 枚硬币^000000。";
			next;
			switch(select("购买:不购买")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "感谢光临。";
					delitem .@Coin,1; 
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "你的硬币不够。";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "改变主意的话再来吧。";
		close;
	}
	else {
		if (strnpcinfo(1) == "Merchant of Manuk") {
			mes "Rtt od d";
			mes "Qwo hd is d irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
man_in01,286,16,1	duplicate(merchant_13_2)	Merchant of Manuk	454
