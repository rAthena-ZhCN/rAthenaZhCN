		mes "在皇家之龙";

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "欢迎来到";
	mes getarg(1) + ".";
	mes "有什么可以为您效劳的？";
	next;
	switch(select("记录:休息 -> 5000 Zeny:取消")) {
	case 1:
		mes .@npc_name$;
		mes "您的复活点";
		mes "已记录。";
		mes "谢谢您，";
		mes "欢迎再次光临。";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "很抱歉，服务费是 5,000 Zeny。下次请务必确认带够入住的钱，好吗？";
			close;
		}
		mes "谢谢。";
		mes "祝您";
		mes "好好休息~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
