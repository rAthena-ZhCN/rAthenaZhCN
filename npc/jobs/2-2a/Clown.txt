valkyrie,53,54,3	script	宫廷乐师#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[宫廷乐师]";
			mes "恭喜您。";
			mes "向勇士们致敬！";
			close;
		}
		mes "[宫廷乐师]";
		mes "你是否想要";
		mes "和我一起唱首歌呢？";
		mes "Sha la la la la~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[宫廷乐师]";
		mes "这个沉闷的世界";
		mes "凡人们需要";
		mes "更多欢乐的歌声。";
		mes "你愿意将歌声带给他们";
		mes "并在对抗邪恶的";
		mes "战斗中扭转局势吗？";
		next;
		mes "[宫廷乐师]";
		mes "你愿意为了米德加尔特...";
		mes "作为一名宫廷乐师这么做吗？";
		mes "作为一名宫廷乐师？";
		next;
		if (select("不：是。") == 1) {
			mes "[宫廷乐师]";
			mes "当你准备好了，";
			mes "随时欢迎回来。";
			mes "向勇士们致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[宫廷乐师]";
			mes "你仍然可以学习更多技能。请使用";
			mes "所有剩余的技能点后再回来找我。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[宫廷乐师]";
		mes "恭喜！";
		mes "作为一名宫廷乐师，你的";
		mes "歌声将为盟友带来";
		mes "希望，而给敌人带来";
		mes "绝望。";
		close;
	}
	mes "[宫廷乐师]";
	mes "欢迎";
	mes "来到巴哈拉宫，";
	mes "荣耀大厅。";
	next;
	mes "[宫廷乐师]";
	mes "请随意";
	mes "让自己感到舒适";
	mes "在你逗留期间。";
	mes "向勇士们致敬！";
	close;
}
