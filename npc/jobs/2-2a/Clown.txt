valkyrie,53,54,3	script	宫廷乐师#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[宫廷乐师]";
			mes "Congratulations.";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[宫廷乐师]";
		mes "要和我";
		mes "一起唱首歌吗？";
		mes "沙啦啦啦啦～";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[宫廷乐师]";
		mes "阴郁的";
		mes "凡人世界，需要";
		mes "更多欢快的歌声。";
		mes "你愿意将歌声";
		mes "带给他们，扭转";
		mes "对抗邪恶的战局吗？";
		next;
		mes "[宫廷乐师]";
		mes "你愿意";
		mes "为米德加尔特……";
		mes "以宫廷乐师的身份吗？";
		next;
		if (select("否:是") == 1) {
			mes "[宫廷乐师]";
			mes "等你准备好，";
			mes "随时回来。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[宫廷乐师]";
			mes "你仍可以学习更多技能。请用完";
			mes "所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[宫廷乐师]";
		mes "恭喜！";
		mes "作为宫廷乐师，";
		mes "你的歌声将";
		mes "为同伴带来希望，";
		mes "为敌人带来绝望。";
		close;
	}
	mes "[宫廷乐师]";
	mes "欢迎";
	mes "前往巴哈拉宫，";
	mes "荣誉殿堂。";
	next;
	mes "[宫廷乐师]";
	mes "请在这里";
	mes "放松一下";
	mes "不必拘束。";
	mes "荣耀归于战士们！";
	close;
}
