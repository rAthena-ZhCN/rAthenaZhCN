valkyrie,53,56,3	script	冷艳舞姬#Valkyrie	101,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[冷艳舞姬]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[冷艳舞姬]";
		mes "向左移动，";
		mes "再向右~！";
		mes "然后踏步...！";
		mes "舞蹈可以是";
		mes "不仅仅是一种爱好。";
		mes "对我来说，它是一种生活方式~";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[冷艳舞姬]";
		mes "米德加尔特大陆";
		mes "需要有才华的女性";
		mes "来巧妙地改变";
		mes "善与恶之间的平衡。";
		mes "这场战斗中的平衡。";
		next;
		mes "[冷艳舞姬]";
		mes "你准备好";
		mes "承担这个角色，";
		mes "成为一名冷艳舞姬了吗？";
		next;
		if (select("不:是。") == 1) {
			mes "[冷艳舞姬]";
			mes "当你准备好了，";
			mes "随时欢迎你回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[冷艳舞姬]";
			mes "你仍然可以学习更多的技能。请在回到我这里之前";
			mes "使用掉所有剩余的技能点。";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[冷艳舞姬]";
		mes "恭喜！";
		mes "作为一名冷艳舞姬，我知道";
		mes "你的表演";
		mes "能够打动所有";
		mes "观看者的心...";
		close;
	}
	mes "[冷艳舞姬]";
	mes "欢迎";
	mes "来到巴哈拉宫，";
	mes "荣誉之殿。";
	next;
	mes "[冷艳舞姬]";
	mes "请在这里";
	mes "尽情放松";
	mes "享受这里的时光。";
	mes "向勇士致敬！";
	close;
}
