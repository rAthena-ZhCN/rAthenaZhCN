valkyrie,53,42,3	script	武术宗师#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[武术宗师]";
			mes "恭喜你。";
			mes "荣耀归于勇士！";
			close;
		}
		mes "[武术宗师]";
		mes "技巧。";
		mes "速度。";
		mes "英雄之力。";
		mes "敏捷。";
		next;
		mes "[武术宗师]";
		mes "武术宗师可以";
		mes "从所有这些特质中受益。";
		mes "但人生有限，";
		mes "无人能精通";
		mes "世间万象……";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[武术宗师]";
		mes "时机已至。";
		mes "是时候让伟大的英雄们";
		mes "挺身而出，对抗";
		mes "肆虐米德加尔特世界的";
		mes "邪恶势力了！";
		next;
		mes "[武术宗师]";
		mes "你是否愿意";
		mes "成为武术宗师？";
		next;
		if (select("否:是。") == 1) {
			mes "[武术宗师]";
			mes "当你准备好了，";
			mes "随时欢迎回来。";
			mes "荣耀归于勇士！";
			close;
		}
		if (SkillPoint) {
			mes "[武术宗师]";
			mes "你仍可学习更多技能。请先将";
			mes "剩余的所有技能点数用完，再回到我这里。";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[武术宗师]";
		mes "恭喜！";
		mes "以武术宗师之姿生存，";
		mes "用你双拳的力量，";
		mes "为这个世界";
		mes "带来光明。";
		close;
	}
	mes "[武术宗师]";
	mes "欢迎来到";
	mes "巴哈拉宫，";
	mes "荣耀之殿。";
	next;
	mes "[武术宗师]";
	mes "请在此处";
	mes "放松身心，";
	mes "自在休憩。";
	mes "荣耀归于勇士！";
	close;
}
