valkyrie,53,58,3	script	神行太保#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神行太保]";
			mes "恭喜你！";
			mes "向勇士们致敬！";
			close;
		}
		mes "[神行太保]";
		mes "呵……";
		mes "当英雄可不容易，";
		mes "还得干些见不得光的勾当——";
		mes "不可靠、";
		mes "鬼祟……";
		mes "狡黠……";
		next;
		mes "[神行太保]";
		mes "但每当局势艰险，";
		mes "我的伙伴们都知道，";
		mes "他们一定能指望上我。";
		mes "我需要他们，";
		mes "他们也需要我。";
		mes "they need me.";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[神行太保]";
		mes "这个世界需要更多";
		mes "敢于游走于秩序与混乱之间";
		mes "的英雄。";
		mes "在法度与无序的夹缝中前行。";
		next;
		mes "[神行太保]";
		mes "你准备好加入这支";
		mes "最隐秘、最迅捷的";
		mes "战士行列了吗？你准备好";
		mes "成为一名神行太保了吗？";
		mes "不。是。";
		next;
		if (select("不。是。") == 1) {
			mes "[神行太保]";
			mes "等你准备好了，";
			mes "随时欢迎回来。";
			mes "向勇士们致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[神行太保]";
			mes "你仍有机会学习更多技能，请在回到我面前之前，";
			mes "务必用完所有剩余的技能点数。";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[神行太保]";
		mes "恭喜你！";
		mes "身为一名神行太保，我希望";
		mes "你刺向背后的，都是该刺之人。";
		mes "以彼之道，还施彼身，";
		mes "用恶徒自己的卑劣手段";
		mes "铲除奸邪！";
		close;
	}
	mes "[神行太保]";
	mes "欢迎来到";
	mes "巴哈拉宫，";
	mes "英烈之殿堂。";
	next;
	mes "[神行太保]";
	mes "请在此期间";
	mes "随意放松，";
	mes "自在休憩。";
	mes "向勇士们致敬！";
	close;
}
