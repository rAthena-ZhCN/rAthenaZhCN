valkyrie,53,39,3	script	圣殿十字军#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[圣殿十字军]";
			mes "恭喜你。";
			mes "荣耀归于勇士！";
			close;
		}
		mes "[圣殿十字军]";
		mes "切勿忘记";
		mes "神圣之战";
		mes "即将来临！";
		mes "我们必须做好准备！";
		mes "愿正义之光";
		mes "永远照亮我们的前路！";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[圣殿十字军]";
		mes "神圣之战将";
		mes "在我们尚未察觉时";
		mes "降临。比以往任何时候都更需要";
		mes "坚强的男儿与巾帼";
		mes "为正义与良善";
		mes "而战。";
		next;
		mes "[圣殿十字军]";
		mes "你愿意以圣殿十字军之名，";
		mes "站在正义的一方";
		mes "战斗吗？";
		next;
		if (select("否:是。") == 1) {
			mes "[圣殿十字军]";
			mes "当你准备好了，";
			mes "随时欢迎回来。";
			mes "荣耀归于勇士！";
			close;
		}
		if (SkillPoint) {
			mes "[圣殿十字军]";
			mes "你仍可学习更多技能。请先使用";
			mes "所有剩余的技能点数，再回到我这里。";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[圣殿十字军]";
		mes "恭喜你。";
		mes "身为圣殿十字军，我希望";
		mes "你能保护弱者，";
		mes "并为我们赢得即将到来的";
		mes "善恶之战";
		mes "的胜利。";
		close;
	}
	mes "[圣殿十字军]";
	mes "欢迎来到";
	mes "巴哈拉宫，";
	mes "荣耀之殿。";
	next;
	mes "[圣殿十字军]";
	mes "请在此";
	mes "安心休憩，";
	mes "不必拘束。";
	mes "荣耀归于勇士！";
	close;
}
