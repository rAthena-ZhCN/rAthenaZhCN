		if (.@karma_d > 4) {
			mes "[议员]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[议员]";
		mes "啊……";
		mes "工艺的巅峰，";
		mes "这就是议员的作品。";
		mes "这是";
		mes "议员的工作。";
		next;
		mes "[议员]";
		mes "一旦你成为";
		mes "最顶尖的人，";
		mes "你就会开始以";
		mes "发现新的更好的";
		mes "制作方法为目标……";
		mes "进行锻造……";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[议员]";
		mes "时机已到！";
		mes "我们的世界需要勇敢、";
		mes "勤劳的冒险者";
		mes "像你这样的人……";
		next;
		mes "[议员]";
		mes "你想成为";
		mes "议员吗？";
		next;
		if (select("不:是。") == 1) {
			mes "[议员]";
			mes "当你准备好了，";
			mes "随时可以回来。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[议员]";
			mes "你仍然有可能学习更多的技能。请在回到我这里之前使用";
			mes "所有剩余的技能点。";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[议员]";
		mes "恭喜！";
		mes "作为议员，";
		mes "我希望你能为米德加尔特";
		mes "开辟一条通向更光明未来的道路。";
		mes "未来。";

	}

	mes "在你停留期间。";
	mes "向勇士致敬！";
