valkyrie,44,47,5	script	超魔导师#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[超魔导师]";
			mes "恭喜你。";
			mes "向勇士们致敬！";
			close;
		}
		mes "[超魔导师]";
		mes "我们超魔导师有";
		mes "责任";
		mes "正确使用我们的毁灭性魔法";
		mes "为正义的目的服务。";
		next;
		mes "[超魔导师]";
		mes "需要一生的修炼";
		mes "才能成为";
		mes "一名超魔导师。你能想象";
		mes "如果我们的力量";
		mes "落入了错误的手中会发生什么吗？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[超魔导师]";
		mes "是时候了。";
		mes "米德加尔特";
		mes "需要那些能够";
		mes "驾驭最强魔法的人……";
		next;
		mes "[超魔导师]";
		mes "你想";
		mes "成为一名超魔导师吗？";
		next;
		if (select("否：是") == 1) {
			mes "[超魔导师]";
			mes "当你准备好了，";
			mes "随时欢迎回来。";
			mes "向勇士们致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[超魔导师]";
			mes "你仍然有可能学习更多技能。请使用";
			mes "你剩余的所有技能点后再回来找我。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[超魔导师]";
		mes "恭喜你。";
		mes "作为一名超魔导师，";
		mes "我希望你能用";
		mes "你的力量为";
		mes "受压迫者带来和平。";
		close;
	}
	mes "[超魔导师]";
	mes "欢迎";
	mes "来到巴哈拉宫，";
	mes "荣誉殿堂。";
	next;
	mes "[超魔导师]";
	mes "请";
	mes "在这里放松自己，";
	mes "享受你的时光。";
	mes "向勇士们致敬！";
	close;
}
