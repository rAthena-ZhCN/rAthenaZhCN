valkyrie,44,47,5	script	超魔导师#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[超魔导师]";
			mes "Congratulations.";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[超魔导师]";
		mes "我们超魔导师";
		mes "有责任";
		mes "将毁灭性的魔法";
		mes "用于正途。";
		next;
		mes "[超魔导师]";
		mes "一生的修行";
		mes "是成为超魔导师";
		mes "所必需的。你能想象";
		mes "如果我们的力量";
		mes "落入恶人之手会怎样吗？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[超魔导师]";
		mes "时候到了。";
		mes "而米德加尔特";
		mes "需要能够";
		mes "驾驭最强魔法的人……";
		next;
		mes "[超魔导师]";
		mes "您想搭乘";
		mes "成为超魔导师吗？";
		next;
		if (select("否:是") == 1) {
			mes "[超魔导师]";
			mes "等你准备好，";
			mes "随时回来。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[超魔导师]";
			mes "你仍可以学习更多技能。请用完";
			mes "所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[超魔导师]";
		mes "Congratulations.";
		mes "作为超魔导师，";
		mes "我希望你能运用";
		mes "自己的力量，为受压迫者";
		mes "带来和平。";
		close;
	}
	mes "[超魔导师]";
	mes "欢迎";
	mes "前往巴哈拉宫，";
	mes "荣誉殿堂。";
	next;
	mes "[超魔导师]";
	mes "请在这里";
	mes "放松一下";
	mes "不必拘束。";
	mes "荣耀归于战士们！";
	close;
}
