valkyrie,44,39,5	script	骑士领主#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[骑士领主]";
			mes "恭喜你。";
			mes "荣耀归于勇士！";
			close;
		}
		mes "[骑士领主]";
		mes "我们骑士肩负着";
		mes "无比重大的责任……";
		mes "为服务与守护而战。";
		next;
		mes "[骑士领主]";
		mes "即使付出";
		mes "生命的代价，";
		mes "我们也必须保护";
		mes "同伴们的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[骑士领主]";
		mes "你的时刻到了！";
		mes "世界仍需要你。";
		mes "请以全新的姿态，";
		mes "继续作为英雄前行。";
		next;
		mes "[骑士领主]";
		mes "你是否愿意";
		mes "成为骑士领主？";
		next;
		if (select("否:是。") == 1) {
			mes "[骑士领主]";
			mes "当你准备好了，";
			mes "随时欢迎回来。";
			mes "荣耀归于勇士！";
			close;
		}
		if (SkillPoint) {
			mes "[骑士领主]";
			mes "你仍有学习更多技能的可能。请在回到我这里之前，";
			mes "将所有剩余的技能点数全部使用完毕。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[骑士领主]";
		mes "恭喜！";
		mes "身为骑士领主，";
		mes "我希望你能始终";
		mes "冲锋在战斗最前线，";
		mes "带领盟友赢得胜利！";
		close;
	}
	mes "[骑士领主]";
	mes "欢迎来到";
	mes "巴哈拉宫，";
	mes "荣耀之殿。";
	next;
	mes "[骑士领主]";
	mes "请在此地";
	mes "放松身心，";
	mes "好好休憩。";
	mes "荣耀归于勇士！";
	close;
}
