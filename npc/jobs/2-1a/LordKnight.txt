valkyrie,44,39,5	script	骑士领主#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[骑士领主]";
			mes "Congratulations.";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[骑士领主]";
		mes "我们骑士肩负着";
		mes "重大的责任……";
		mes "效忠与守护。";
		next;
		mes "[骑士领主]";
		mes "哪怕付出";
		mes "自己的生命，";
		mes "我们也必须守护";
		mes "同伴的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[骑士领主]";
		mes "你的时刻到了！";
		mes "世界仍然需要你。";
		mes "请继续你的人生，";
		mes "以英雄的全新姿态。";
		next;
		mes "[骑士领主]";
		mes "你想";
		mes "成为骑士领主吗？";
		next;
		if (select("否:是") == 1) {
			mes "[骑士领主]";
			mes "等你准备好，";
			mes "随时回来。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[骑士领主]";
			mes "你仍可以学习更多技能。请用完";
			mes "所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[骑士领主]";
		mes "恭喜！";
		mes "作为骑士领主，";
		mes "我希望你能";
		mes "站在战斗的最前线，";
		mes "带领同伴走向胜利！";
		close;
	}
	mes "[骑士领主]";
	mes "欢迎";
	mes "前往巴哈拉宫，";
	mes "荣誉殿堂。";
	next;
	mes "[骑士领主]";
	mes "请在这里";
	mes "放松一下";
	mes "不必拘束。";
	mes "荣耀归于战士们！";
	close;
}
