valkyrie,44,55,5	script	神射手#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神射手]";
			mes "Congratulations.";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[神射手]";
		mes "一箭。";
		mes "一命。";
		mes "并不难做到";
		mes "只要你培养出";
		mes "那种战斗风格的";
		mes "的眼光。";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[神射手]";
		mes "世界";
		mes "需要强大的弓箭手";
		mes "像你一样。你准备好承担";
		mes "这份重大的责任了吗？";
		next;
		mes "[神射手]";
		mes "你愿意";
		mes "迈出下一步，";
		mes "成为神射手吗？";
		next;
		if (select("否:是") == 1) {
			mes "[神射手]";
			mes "等你准备好，";
			mes "随时回来。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[神射手]";
			mes "你仍可以学习更多技能。请用完";
			mes "所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[神射手]";
		mes "恭喜！";
		mes "作为神射手，我希望";
		mes "邪恶的爪牙";
		mes "永远不得安宁，";
		mes "只要他们出现在";
		mes "你的视野中！";
		close;
	}
	mes "[神射手]";
	mes "欢迎";
	mes "前往巴哈拉宫，";
	mes "荣誉殿堂。";
	next;
	mes "[神射手]";
	mes "请在这里";
	mes "放松一下";
	mes "不必拘束。";
	mes "荣耀归于战士们！";
	close;
}
