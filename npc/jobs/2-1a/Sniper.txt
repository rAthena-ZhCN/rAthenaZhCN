valkyrie,44,55,5	script	神射手#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神射手]";
			mes "恭喜你！";
			mes "向勇士们致敬！";
			close;
		}
		mes "[神射手]";
		mes "一击，";
		mes "一杀。";
		mes "只要你掌握了";
		mes "这种战斗风格的";
		mes "洞察力，";
		mes "便不再困难。";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[神射手]";
		mes "这个世界正亟需";
		mes "如你这般强大的弓箭手。你是否已准备好";
		mes "承担这份非凡的责任？";
		mes "承担这份非凡的责任？";
		next;
		mes "[神射手]";
		mes "你是否愿意";
		mes "迈出下一步，";
		mes "转职为神射手？";
		next;
		if (select("否：是。") == 1) {
			mes "[神射手]";
			mes "当你准备好了，";
			mes "随时欢迎回来。";
			mes "向勇士们致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[神射手]";
			mes "你仍可学习更多技能，请在返回我这里之前，";
			mes "务必用完所有剩余的技能点数。";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[神射手]";
		mes "恭喜你！";
		mes "身为一名神射手，我希望";
		mes "邪恶的爪牙";
		mes "永无宁日——";
		mes "只要他们尚在";
		mes "你的火狩之下！";
		close;
	}
	mes "[神射手]";
	mes "欢迎来到";
	mes "巴哈拉宫，";
	mes "荣耀之殿堂。";
	next;
	mes "[神射手]";
	mes "请在此期间";
	mes "放松身心，";
	mes "自在休憩。";
	mes "向勇士们致敬！";
	close;
}
