//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_卡普拉	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display 卡普拉 Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[卡普拉职员]";
		mes "欢迎来到";
		mes "卡普拉公司。";
		mes "卡普拉服务";
		mes "永远伴您左右。";
		mes "请问需要什么帮助？";
		break;
	case 1:// Niflheim specific message
		mes "[卡普拉职员]";
		mes "^666666欢-欢迎-迎光临";
		mes "来-来到卡-卡普拉";
		mes "公-公司……^000000";
		break;
	case 2:// Guild Castle 卡普拉 message
		mes "[卡普拉职员]";
		mes "欢迎。^ff0000" + getguildname( getcharid(2) ) + "^000000 公会成员。";
		mes "无论您走到哪里，卡普拉公司都会陪伴在您身边。";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[卡普拉职员]";
		mes "那么，您是从遥远的国度来学习我们的文化，还是只是来观光呢？";
		mes "无论哪种，何不多停留一会儿呢？";
		mes "空气中永远弥漫着";
		mes "怡人野花的芬芳。";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[卡普拉职员]";
		mes "我们众多的卡普拉";
		mes "服务网点，让您永远不会";
		mes "远离家乡。";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild 卡普拉 Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "使用仓库", "使用公会仓库", "租用手推车", "使用传送服务", "取消";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"存档","使用仓库","取消"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"使用仓库","取消"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"存档","使用仓库","租用手推车","查看其他信息","取消"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"使用仓库","租用手推车","查看其他信息","取消"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"使用仓库","查看其他信息","取消"; break;
			// Save, Storage, and Pushcart only (卡普拉 Warehouse)
			case 7:	setarray .@K_Menu0$[0],"存档","使用仓库","租用手推车","取消"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"存档","使用仓库","查看其他信息","取消"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"使用仓库","租用手推车","使用传送服务","查看其他信息","取消"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"使用仓库","存档","租用手推车","取消"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"存档","使用仓库","使用传送服务","租用手推车","查看其他信息","取消"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "存档")
			return;
		else if (.@K_Menu0$[.@j] == "使用仓库") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "使用传送服务") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[卡普拉职员]";
				mes "由于 ^FF0000限制";
				mes "运输协定^000000，";
				mes "卡普拉公司无法";
				mes "提供瞬间移动服务";
				mes "在黑森林共和国。";
				next;
				mes "[卡普拉职员]";
				mes "请您";
				mes "改用飞空艇服务";
				mes "。感谢您的";
				mes "理解与配合。";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "租用手推车") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "查看其他信息") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "取消") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "使用公会仓库") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[卡普拉职员]";
			mes "很抱歉，现在有另一位公会成员正在使用公会仓库";
			mes "。请等那位成员使用完毕。";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[卡普拉职员]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666对-对-对不起，";
			mes "您-您-您-是";
			mes "初-初心者……需-需要";
			mes "基-基本技-技能、等-等级6……^000000";
			return;
		}
		mes "很抱歉，你";
		mes "需要初心者的";
		mes "基本技能等级6才能";
		mes "使用仓库服务。";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for 卡普拉 Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[卡普拉职员]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Ze-zeny……";
					mes "更-更多 zeny……！";
					mes "需-需要150……zeny……";
					mes "呃！要-要抽-抽血了~！^000000";
					return;
				}
				// Standard Message
				mes "抱歉，您的钱";
				mes "不足以使用";
				mes "仓库服务。我们的";
				mes "仓库使用费为" + .@fee + "zeny。";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[卡普拉职员]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666谢……谢您……使用……^000000";
		}
		// Normal message
		else {
			mes "[卡普拉职员]";
			mes "来，我这就为您打开";
			mes "您的仓库。";
			mes "感谢您使用";
			mes "卡普拉服务。";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[卡普拉职员]";
	mes "请选择";
	mes "您的目的地。";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "取消")
		return;
	// Consume "Free Ticket for 卡普拉 Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[卡普拉职员]";
			mes "很抱歉，您没有";
			mes "足够的 zeny 支付瞬间移动";
			mes "服务的费用。";
			mes "传送至" + @wrpD$[.@j] + "的费用为" + @wrpP[.@j] + "zeny。";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "艾尔帕兰") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "阿尔贝塔") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "科莫多") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "科莫多灯塔") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "吉芬") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "伊斯鲁德") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "朱诺") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "雷神之锤死亡矿井") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "梦罗克") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "兽人地下城") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "斐扬") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "普隆德拉") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "温巴拉") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[卡普拉职员]";
		mes "很抱歉，";
		mes "手推车出租服务";
		if (checkre(0) == true)
			mes "仅限商人系职业使用。";
		else {
			mes "仅限商人、";
			mes "铁匠、神工匠、";
			mes "炼金术师与创造者使用。";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[卡普拉职员]";
		mes "您已经装备了";
		mes "一辆手推车。";
		mes "很遗憾，我们无法";
		mes "同时向同一位顾客";
		mes "出租多辆手推车。";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[卡普拉职员]";
		mes "只有学会“手推车”技能后才能租用手推车。";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[卡普拉职员]";
		mes "手推车出租";
		mes "费用为" + .@rental_fee + "zeny。您";
		mes "要租用手推车吗？";
		next;
		if (select("租用手推车:取消") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[卡普拉职员]";
			mes "很抱歉，你";
			mes "没有足够的";
			mes "足够的 zeny 支付手推车";
			mes "出租费用，费用为" + .@rental_fee + "zeny。";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

				viewpoint 2,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			break;
		default:
			return;
		}
	}
}

	} else if (.@type == 2) {// RockRidge Save
		mes "Saved.";
		mes "感谢您的惠顾。";
	}
	close3;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "请输入仓库密码：";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Wrong storage password.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + "为您隆重推出新服务：";
		mes "为仓库加装密码保护的附加服务。";
		next;
		if (select( "设置新密码 -> 5000z", "取消" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "您的仓库已设置密码保护。您要怎么做？";
	next;
	switch( select( "更改旧密码 -> 5000z", "解除仓库密码 -> 1000z", "取消" ) ) {
	case 1:
		mes .@npc_name$;
		mes "首先，请输入您的^0000FF旧密码^000000。";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "密码错误。无法设置新密码。";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "解除密码前，请先输入您的密码。";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "密码尚未解除。";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "您的 zeny 不足。";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "仓库密码已成功解除。";
			mes "感谢您使用" + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "密码错误。1000z 将不予退还。";
			mes "下次请输入正确的密码。";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "现在请输入^FF0000新密码^000000，以保护您的仓库免受盗贼侵扰。";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "密码尚未更改。";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "您的 zeny 不足。";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "您的仓库已设置密码保护。";
	mes "感谢您使用" + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "请输入 1000~10000000 之间的数字：";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "不能使用这么大的密码。";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "不能使用这么短的密码。";
		return 0;
	}
	return .@code_;
}
