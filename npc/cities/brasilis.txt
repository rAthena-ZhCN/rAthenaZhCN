-	script	::Crewman_bra2	-1,{
	mes "[船员]";
	mes "嘿，你听说过一个叫巴西的地方吗？";
	mes "那是一座热带城市，像沙漠一样炎热，却又多雨。是个非常神秘的地方。";
	next;
	mes "[船员]";
	mes "我们最近发现了一条新的海上航线，可以轻松到达那里。";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "往返只要 10,000 zeny，VIP 只要 1,000！那么，你想去吗？"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "往返只要 10,000 zeny！那么，你想去吗？";
	}
	next;
	switch(select("带我去巴西！:我留在这里.")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[船员]";
			mes "我说的是"+callfunc("F_InsertComma",.@cost)+" zeny.";
			close;
		}
		else {
			mes "[船员]";
			mes "太棒了~！！我们走吧~！";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[船员]";
		mes "如果你以后有兴趣，告诉我一声，我就带你去。";
		close;
	}
}

	}
}

brasilis,137,77,5	script	冰淇淋小贩	85,{
	mes "[冰淇淋小贩]";
	mes "来呀~来呀~";
	mes "冰淇淋是炎热日子里最棒的零食~";
	mes "只要 ^3355FF100 Zeny^000000~";
	mes "冰淇淋~";
	mes "快来买冰淇淋！";
	next;
	switch(select("给我一支！:冰淇淋？:取消.")) {
	case 1:
		mes "[冰淇淋小贩]";
		mes "因为想买冰凉冰淇淋的人太多，一次只能订购 5 支。";
		mes "那么，你要几支？";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[冰淇淋小贩]";
				mes "一支都不要？";
				mes "行吧，别挡道，我还有客人要招呼。";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[冰淇淋小贩]";
				mes "Wow.";
				mes "你订得太多了。";
				mes "吃超过 5 支，你可能得跟肚子里的怪物打一架。冷静点，老兄。";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[冰淇淋小贩]";
			mes "喂~！你的钱不够。";
			mes "只要 ^3355FF100 Zeny^000000~ 真的！";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[冰淇淋小贩]";
			mes "你的东西好像太多了。";
			mes "买之前请先减轻背包重量。";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[冰淇淋小贩]";
		mes "“冰淇淋是……";
		mes "等等，你难道不知道";
		mes "什么是冰淇淋吗？";
		mes "你是从哪块";
		mes "石头底下钻出来的？";
		next;
		mes "[冰淇淋小贩]";
		mes "这话听起来有多怪，我都不想说。";
		mes "总之，趁冰淇淋还冰凉，就在这里买吧。";
		close;
	case 3:
		mes "[冰淇淋小贩]";
		mes "别错过品尝全大陆最棒冰淇淋的机会~！";
		close;
	}
}
