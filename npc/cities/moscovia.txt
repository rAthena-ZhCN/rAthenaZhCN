			mes "我会帮你享受一段愉快的旅程";
			mes "前往莫斯科比亚！";

			mes "[莫斯科比亚宣传官]";
			mes "那么，我们现在就开始吧。";
			mes "你需要付我 10,000 zeny";
			mes "才能前往莫斯科比亚。";

				close;
			} else {
				mes "[莫斯科比亚宣传官]";
				mes "那么，我们这就出发了。";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[莫斯科比亚宣传官]";
			mes "如果你现在太忙，";
			mes "随时再来告诉我。";
			mes "我随时都愿意带任何人去莫斯科比亚。";
