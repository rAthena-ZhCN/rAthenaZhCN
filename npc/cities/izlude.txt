-	script	::Sailor_izlude	-1,{
	mes "[水手]";
	mes "嘿，大家听好！";
	mes "注意，注意！";
	mes "快来乘风";
	mes "搭乘这艘迷人的船！";
	mes "快，快！";
	next;
	switch(select("拜伦岛 -> 150 Zeny.:阿尔贝塔码头  -> 500 Zeny.:取消")) {
	case 1:
		if (Zeny < 150) {
			mes "[水手]";
			mes "150 Zeny！";
			mes "只要 150 Zeny 就能乘坐！";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[水手]";
			mes "500 Zeny！";
			mes "只要 500 Zeny 就能乘坐！";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
