airport,143,43,5	script	机场职员#airport1a::airport1	90,{
	mes "[机场职员]";
	mes "欢迎来到";
	mes "艾音布罗克机场，";
	mes "我们提供直达";
	mes "以下城市的航班：";
	mes "朱诺、里希塔乐镇和休格尔。";
	next;
	if (select("搭乘飞空艇:取消") == 1) {
		mes "[机场职员]";
		mes "飞空艇搭乘费";
		mes "为1,200 Zeny，但如果您";
		mes "持有飞空艇免费搭乘券，";
		mes "便可免费。您要";
		mes "搭乘飞空艇吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[机场职员]";
			mes "很抱歉，您没有";
			mes "飞空艇免费";
			mes "搭乘券，也没有";
			mes "足够的Zeny支付";
			mes "搭乘费。请记住，";
			mes "搭乘费为1,200 Zeny。";
			close;
		}
	}
	mes "[机场职员]";
	mes "谢谢，";
	mes "祝您愉快。";
	close;
}
