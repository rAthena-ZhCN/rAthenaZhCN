-	script	::Airship_Staff_izlude	-1,{
	mes "[飞空艇职员]";
	mes "欢迎！您要";
	mes "搭乘飞空艇吗？它";
	mes "的航班会经停";
	mes "朱诺和拉赫。";
	next;
	if (select("搭乘飞空艇前往朱诺/拉赫。:取消。") == 1) {
		mes "[飞空艇职员]";
		mes "搭乘费为";
		mes "1,200 Zeny。不过，";
		mes "若使用飞空艇免费搭乘券，";
		mes "这笔费用可免。那么，";
		mes "您还想搭乘吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[飞空艇职员]";
			mes "很抱歉，您没有";
			mes "没有1,200 Zeny支付";
			mes "搭乘费。";
			close;
		}
	}
	mes "[飞空艇职员]";
	mes "谢谢，";
	mes "欢迎再来。";
	mes "祝您愉快~";
	close;
}
