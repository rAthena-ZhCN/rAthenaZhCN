y_airport,143,43,5	script	机场职员#y_air1a::y_airport1	90,{
	mes "[机场职员]";
	mes "欢迎来到朱诺机场，我们提供飞往艾音布罗克、里希塔乐镇和休格尔的国内航班，";
	mes "以及飞往伊斯鲁得和拉赫的国际航班。";
	mes "请问有什么可以为您服务？";
	next;
	if (select("搭乘飞空艇。:取消。") == 1) {
		mes "[机场职员]";
		mes "所有航班的";
		mes "登机费为1,200 Zeny。若您";
		mes "使用飞空艇免费搭乘券，";
		mes "登机费将予以免除。那么您要出发吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[机场职员]";
			mes "很抱歉，您";
			mes "没有1,200 Zeny";
			mes "来支付登机费。";
			close;
		}
	}
	mes "[机场职员]";
	mes "谢谢，祝您";
	mes "今天愉快。";
	close;
}
