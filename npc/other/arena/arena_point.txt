//= Point exchanger: Turbo track points -> Arena Points

//============================================================

prt_are_in,103,11,3	script	竞技场点数管理员	732,{
	mes "[竞技场点数管理员]";
	mes "希望你在竞技场中";
	mes "玩得愉快。如果你在";
	mes "艾尔帕兰的竞速赛道";
	mes "获得了竞速点数，我可以";
	mes "把它们兑换成竞技场点数。";
	next;
	switch(select("点数查询:兑换点数:^660000兑换说明^000000")) {
	case 1:
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + "，";
		mes "你目前拥有";
		mes "" + arena_point + "点竞技场点数";
		mes "以及" + tt_point + "点竞速点数。";
		close;
	case 2:
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + "，";
		mes "你目前拥有";
		mes "" + arena_point + "点竞技场点数";
		mes "以及" + tt_point + "点竞速点数。";
		next;
		mes "[竞技场点数管理员]";
		mes "请从以下";
		mes "竞速点数兑换竞技场点数的";
		mes "项目中选择。请注意，";
		mes "一次兑换超过 10 点";
		mes "竞速点数时，只能按 ^4D4DFF10 的倍数^000000进行兑换。";
		next;
		switch(select("2 竞速点 -> 1 竞技场点:4 竞速点 -> 2 竞技场点:6 竞速点 -> 3 竞技场点:8 竞速点 -> 4 竞技场点:10 竞速点及以上:取消")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[竞技场点数管理员]";
			mes "请输入你要兑换的次数，";
			mes "也就是要把多少份";
			mes "10 点竞速点数换成";
			mes "竞技场点数。最大";
			mes "可输入值为 20。";
			mes "若要取消，请输入''^3355FF0^000000''。";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[竞技场点数管理员]";
				mes "你已";
				mes "取消";
				mes "本项服务。";
				close;
			}
			else if (.@input > 20) {
				mes "[竞技场点数管理员]";
				mes "你的请求超出";
				mes "了上限。请输入";
				mes "不大于 20 的数值。";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[竞技场点数管理员]";
					mes "如果继续此次兑换，";
					mes "你的竞技场点数将";
					mes "超过上限。你最多";
					mes "只能拥有 29,000 点";
					mes "竞技场点数。请先";
					mes "消耗一些竞技场点数。";
					next;
					mes "[竞技场点数管理员]";
					mes "请先消耗一些";
					mes "竞技场点数，再";
					mes "使用本服务。";
					mes "感谢您的惠顾。";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[竞技场点数管理员]";
					mes "很抱歉，";
					mes "你的竞速点数";
					mes "不足以进行";
					mes "本次竞技场点数";
					mes "兑换。";
					close;
				}
				else {
					mes "[竞技场点数管理员]";
					mes "你已将";
					mes "10 点竞速点数";
					mes "兑换成 5 点竞技场点数" + .@input + " times.";
					mes "共" + .@want_point1 + "点竞速点数已兑换成";
					mes "" + .@want_point +"点竞技场点数。";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[竞技场点数管理员]";
					mes "" + strcharinfo(0) + "，";
					mes "你现在拥有";
					mes "^4682B4" + arena_point + "^000000 点竞技场点数";
					mes "以及 ^00688B" + tt_point + "^000000 点竞速点数。";
					mes "感谢您的惠顾。";
					close;
				}
			}
		case 6:
			mes "[竞技场点数管理员]";
			mes "你已";
			mes "取消";
			mes "本项服务。";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[竞技场点数管理员]";
		mes "如果继续此次兑换，";
		mes "你的竞技场点数将";
		mes "超过上限。你最多";
		mes "只能拥有 29,000 点";
		mes "竞技场点数。请先";
		mes "消耗一些竞技场点数。";
		next;
		mes "[竞技场点数管理员]";
		mes "请先消耗一些";
		mes "竞技场点数，再";
		mes "使用本服务。";
		mes "感谢您的惠顾。";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + "，";
		mes "你已兑换了"+ .@track_point +"点竞速";
		mes "点数，换得"+ .@arena_point +"点竞技场点数。";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[竞技场点数管理员]";
		mes "" + strcharinfo(0) + "，";
		mes "你现在拥有";
		mes "^4682B4" + arena_point + "^000000 点竞技场点数";
		mes "以及 ^00688B" + tt_point + "^000000 点竞速点数。";
		mes "感谢您的惠顾。";
		close;
	}
	else {
		mes "[竞技场点数管理员]";
		mes "很抱歉，你的竞速点数";
		mes "不足。";
		mes "本兑换服务至少需要 2 点竞速点数。";
		close;
	}
}
