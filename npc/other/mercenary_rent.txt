-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "[佣兵用品商人]";
	mes "你好，我卖的是";
	mes "佣兵们可以";
	mes "使用的用品。请问你";
	mes "有没有特别";
	mes "想要的东西？";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[佣兵用品商人]";
	mes getitemname(.@item[.@m]);
	mes "每个售价" + .@str_cost$ + "Zeny。";
	mes "你要买多少个？";
	next;
	input .@input;
	if (!.@input) {
		mes "[佣兵用品商人]";
		mes "你改变主意了？";
		mes "好吧，随时欢迎";
		mes "你回来";
		mes "购买佣兵专用药水。";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[佣兵用品商人]";
		mes "很抱歉，你";
		mes "每次最多";
		mes "只能购买10,000个";
		mes "这种药水。请输入";
		mes "1到10,000之间的数字。";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[佣兵用品商人]";
		mes "抱歉，您的钱";
		mes "没有足够的Zeny买下";
		mes "这么多药水。不过，我会";
		mes "在你准备好";
		mes "购买佣兵用品时";
		mes "一直在这里等你。";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[佣兵用品商人]";
		mes "如果我给你那么多";
		mes "药水，你也";
		mes "无法把它们全部带走。";
		mes "请在清出";
		mes "一些物品栏空间后";
		mes "再来找我。";
		close;
	}
	mes "[佣兵用品商人]";
	if (.@input == 1) 
		mes "这是你的" + getitemname(.@item[.@m]) + ".";
	else {
		mes "给，这里正好是";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "谢谢，欢迎你";
	mes "在需要更多";
	mes "佣兵药水时再来。";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

// Mercenary Switch NPCs
//============================================================
-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[检查员]";
	mes "请输入密码。";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[检查员]";
	if (.@i == -2)
		mes "Error.";
	else if (.@i == 0)
		mes "号码错误";
	else if (.@i == 1) {
		mes "请选择。";
		next;
		switch( select( "关闭佣兵NPC","开启佣兵NPC" ) ) {
		case 1:
			mes "NPC已关闭。";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "NPC已开启。";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
