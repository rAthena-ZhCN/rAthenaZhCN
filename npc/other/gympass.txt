payon,173,141,4	script	猛男卡布斯#GymPass	899,{
	mes "[猛男]";
	mes "嘿，你好。现在的人";
	mes "身体活动量已经不像";
	mes "以前那么大了。就算你靠";
	mes "战斗为生，身体的某些";
	mes "部位也可能变得松弛";
	mes "无力。你懂我的意思吗？";
	next;
	mes "[猛男]";
	mes "嘿，跟我一起锻炼，我可以";
	mes "保证你能";
	mes "举起并携带更多";
	mes "东西。只要每次给我";
	mes "^FF0000健身房会员券^000000，";
	mes "就可以开始了。";
	next;
	mes "[猛男]";
	mes "但别太兴奋：";
	mes "不管我带你";
	mes "做多少训练，都";
	mes "可能练过头。你听说过";
	mes "有谁练得太壮吗？";
	mes "那是因为他们都死了。懂了吧？";
	next;
	mes "[猛男]";
	mes "我想说，你可以";
	mes "放心地认真跟我";
	mes "一起锻炼，把物品携带";
	mes "量提升^FF000010倍^000000。";
	mes "那么……准备好流汗了吗？";
	next;
	switch(select("是:否:呃，我的锻炼效果没了。")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[猛男]";
				mes "哦，太好了，我看你";
				mes "带了健身房会员券。";
				mes "好，就照着我的动作做，";
				mes "试着感受肌肉的灼烧感。";
				mes "准备好了吗？开始吧。";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[猛男]";
				mes "好了，你现在应该";
				mes "能携带更多东西了。";
				mes "我看看，我们还能";
				mes "把你的物品携带量再提升";
				mes "^FF00000" + .@remain_carry + "^000000次，只要继续这样";
				mes "一起训练下去。";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[猛男]";
				mes "老兄，我不是跟你说过吗？";
				mes "你得把";
				mes "^FF0000健身房会员券^000000带来，";
				mes "才能锻炼，练出";
				mes "搬运物品的肌肉。";
				close;
			}
		}
		else {
			mes "[猛男]";
			mes "老兄，我觉得我们";
			mes "没法再把你的物品搬运";
			mes "肌肉练得更强了。";
			mes "就算勉强尝试，";
			mes "对你的身体也太危险了！拜托，";
			mes "我跟你说过极限在哪。";
			close;
		}
	case 2:
		mes "[猛男]";
		mes "唉，那太可惜了。";
		mes "好吧，如果你";
		mes "改变主意就回来。跟";
		mes "你的朋友也说说我：";
		mes "要是他们身体松弛，我会帮";
		mes "他们练出好身材。";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "[猛男]";
			mes "怎么了？";
			mes "你的物品搬运";
			mes "肌肉萎缩了？算你";
			mes "走运，还有肌肉记忆";
			mes "这回事。重新练回来";
			mes "不会花太久……";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[猛男]";
			mes "怎么样？";
			mes "你的物品搬运";
			mes "肌肉长回来了，";
			mes "就这么简单！别再";
			mes "半途而废了，好吗？";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[猛男]";
			mes "Uhh...";
			mes "我们以前没有";
			mes "一起锻炼过。";
			mes "这点我很确定。";
			close;
		}
	}
}
