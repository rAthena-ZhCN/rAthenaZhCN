iz_dun04,130,234,5	script	守门人#iz_dun	413,{
	mes "[守门人]";
	mes "站住！";
	mes "冒险家，你来这里有什么事？";
	mes "我们发现了这座海底城市的一个新楼层。";
	next;
	mes "[守门人]";
	mes "你认为自己能迎接这个挑战吗？";
	mes "这里的怪物等级在110到124之间。";
	mes "而且你必须提防魔鬼大乌贼！";
	next;
	switch(select("放出魔鬼大乌贼！:我留在这里。")) {
	case 1:
		mes "[守门人]";
		mes "务必小心，祝你好运。";
		mes "如果遇到魔鬼大乌贼，一定要有朋友相助，否则就像我们一样赶快逃吧。";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[守门人]";
		mes "探索新区域之前，最好先做好准备。";
		mes "准备好出发后再回来吧。";
		close;
	}
}
