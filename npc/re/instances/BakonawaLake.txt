ma_scene01,174,179,4	script	塔霍	541,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你携带的道具太多了。请卸下一些后再来。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你的负重过重了。请减轻重量后再来。";
		close;
	}
	if (BaseLevel < 140) {
		mes "[塔霍]";
		mes "喂。这地方对你来说太危险了。你保护不了自己。马上离开这里，回村子里去！";
		close;
	}
	if (malaya_bakona1 < 7 && malaya_bakona2 < 15) {
		mes "[塔霍]";
		mes "我们都会被那巴库暴龙杀掉的。唉，必须杀掉它，可惜我们还没准备好。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$, "Bakonawa Lake";

	set .@baku_time, checkquest(12278,PLAYTIME);
	if (.@baku_time == -1) {
		if (checkquest(12279) == -1) setquest 12279;
	}
	if (!.@party_id) {
		mes "[塔霍]";
		mes "没有能和你同行的人吗？就算你想独自前往，也请先组好队伍再来。";
		close;
	}
	if (.@baku_time == -1) {
		if (is_party_leader() == true) {
			mes "[塔霍]";
			mes "你就是讨伐巴库暴龙的队长吗？路况不好，我们得用绳子从悬崖下去，可以吗？";
			next;
			set .@i, select("请编一条绳子。:我现在就下去。:取消。");
		} else {
			mes "[塔霍]";
			mes "你们是来讨伐巴库暴龙的吧。队长得帮我编好下去用的绳子。";
			next;
			set .@i, select(":我现在就下去。:取消。");
		}
		switch(.@i) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "[塔霍]";
				mes "啊，绳子掉下去了。我得重新编一条。";
				close;
			}
			mes "[塔霍]";
			mes "我正在编，编好后你就可以下去了。";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "[塔霍]";
				mes "发生了未知错误。";
				close;
			case IE_NOINSTANCE:
				mes "[塔霍]";
				mes "绳子还没编好。请稍等一下。";
				close;
			case IE_NOMEMBER:
				mes "[塔霍]";
				mes "平民不可进入。";
				close;
			case IE_OK:
				mapannounce "ma_scene01","队伍成员，"+strcharinfo(0)+" of the party "+getpartyname(.@party_id)+" is entering the dungeon, "+.@md_name$+".",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
				setquest 12278;
				//warp "1@ma_b",64,51;
				close;
			}
		case 3:
			close;
		}
	} else if (.@baku_time == 0 || .@baku_time == 1) {
		mes "[塔霍]";
		mes "通往湖边的绳子已经断了。我们正在清理周围，请过一段时间再来。";
		close;
	} else if (.@baku_time == 2) {
		mes "[塔霍]";
		mes "这周围已经清理干净了。现在你拿到绳子后就可以再下去了。";
		erasequest 12278;
		close;
	}
}
