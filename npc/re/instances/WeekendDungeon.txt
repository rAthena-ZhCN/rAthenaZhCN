		close;

	case -1:
		break;
	case 0:
	case 1:

		case -1:
		case 2:
			mes "[玛丽·杰]";
			mes "抱歉，进入" + .@md_name$ + "的期限已过。";
			mes "请休息一下，等待副本再次开放。";

		case 0:
		case 1:
			break;
		}
		break;

		erasequest 12378;

		mes "冷却时间已结束。";
		mes "您可以重新进入记忆副本。";

		end;

		mes "[玛丽·杰]";
		mes "您还没有准备吗？";

		close;

		mes "您携带的物品种类过多。请整理您的背包。";
		close;

		mes "[礼物包装员]";
		mes "有需要包装服务时随时再来！";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
