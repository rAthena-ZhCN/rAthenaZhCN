moro_cav,50,64,5	script	黄色种子#bios	CLEAR_NPC,{
	if (BaseLevel < 160) {
		mes "^ff0000等级低于160的角色无法进入此副本。^000000";
		close;
	}
	if (isbegin_quest(15006) == 0) {
		mes "^ff0000你必须先与流浪老人见面，才能进入此副本。^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000你必须先组建或加入至少1人的队伍，才能进入此副本。^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 5 mins to re-enter
			mes "^ff0000上次进入后23小时内无法再次进入此记忆副本。^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ff上次进入后23小时内无法再次进入此记忆副本。^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// Memorial Dungeon: Isle of Bios
		erasequest 15007;// Memorial Dungeon: Isle of Bios
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// Memorial Dungeon: Isle of Bios
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "Create Memorial dungeon.";
	switch( select( .@create$, "进入拜欧斯岛。", "Cancel." ) ) {
	case 1:
		instance_create("Isle of Bios");
		mes "^ff0000已创建记忆副本。^000000";
		mes "^ff0000请点击并再次按回车键。^000000";
		close;
	case 2:
		switch( instance_enter("Isle of Bios") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000请注意，上次进入后23小时内无法再次进入此记忆副本。^000000";
				next;
			}
			mes "^ff0000你的队长尚未创建时空裂缝。^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", "" + strcharinfo(0) + "的队伍，" + getpartyname( getcharid(1) ) + ", is entering Isle of Bios.",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// Memorial Dungeon: Isle of Bios
				setquest 15007;// Memorial Dungeon: Isle of Bios
			}
			end;
		}
	case 3:
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
