dali02,99,148,3	script	次元装置#XX1	PORTAL,{
	if (BaseLevel < 145) {
		mes "- 必须达到145级以上";
		mes "看来可以进入。 -";
		close;
	}
	switch( checkquest(9336,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "- 次元装置已关闭。";
		mes "看来无法进入。 -";
		close;
	case 2:
		break;
	}
	if (isbegin_quest(9335) > 0) {
		mes "- 次元装置已关闭。";
		mes "看来无法进入。 -";
		close;
	}
	if (select( "不进入装置。", "进入时间装置" ) == 1) {
		if (getcharid(1) > 0) {
			mes "- 次元装置似乎";
			mes "即将扭曲时空 -";
			close;
		}
		mes "- 这里有一股潮湿的气息。 -";
		close;
	}
	.@md_name$ = "Fenrir and Sarah";
	switch( instance_enter(.@md_name$) ) {
	case IE_OTHER:
		mes "发生了未知错误。";
		close;
	case IE_NOINSTANCE:
		mes "回忆副本" + .@md_name$ + "不存在。";
		mes "队长尚未创建回忆副本。";
		close;
	case IE_NOMEMBER:
		mes "只有队伍成员才能进入回忆副本。";
		close;
	case IE_OK:
		mapannounce "dali02", "队伍成员" + strcharinfo(0) + "所在队伍" + getpartyname( getcharid(1) ) + "进入了房间" + .@md_name$ + "", bc_map,0xFF99,FW_NORMAL,12;
		setquest 9335;// Search for shards of Gigantes
		if (isbegin_quest(9336) > 0)
			erasequest 9336;
		// warp "1@glast",367,304;
		end;
	}
}

dali02,93,146,6	script	助理教授#a1	4_M_REPAIR,{
	if (sarah_fenrir == 0)
		end;
	disable_items;
	mes "[首席助手]";
	mes "我从教授那里听说过你。";
	mes "教授吩咐过我，要为你冒着生命危险带回基勘特斯碎片一事给予足够的报酬。";
	next;
	switch( select( "为耳环附魔。", "购买耳环", "结束对话" ) ) {
	case 1:
		mes "[首席助手]";
		mes "我们在遗迹发掘出的莎拉耳环，一定蕴藏着巨大的潜力。";
		next;
		switch( select( "为莎拉耳环附魔", "初始化莎拉耳环的附魔", "结束对话" ) ) {
		case 1:
			mes "[首席助手]";
			mes "一只耳环可以进行2次附魔。";
			mes "你想为哪只耳环附魔？";
			next;
			switch( select( "莎拉的左耳环", "莎拉的右耳环", "退出" ) ) {
			case 1:
				.@sarah_item_id = 28310;
				.@sarah_earring$ = "left";
				break;
			case 2:
				.@sarah_item_id = 28311;
				.@sarah_earring$ = "right";
				break;
			case 3:
				mes "[首席助手]";
				mes "如果你想在";
				mes "莎拉耳环上进行附魔，";

			mes "必须装备该物品才能进行附魔。";
			mes "你想为哪个栏位附魔？";
			next;
			if (select( "左栏", "右栏" ) == 1) {
				.@part = EQI_ACC_L;
				.@column$ = "left";
				.@bonus_size = 4;		// 4 possibilities
				.@bonus_chance = 20;	// 20% per possibility
			}
			else {
				.@part = EQI_ACC_R;
				.@column$ = "right";
				.@bonus_size = 6;		// 6 possibilities
				.@bonus_chance = 10;	// 10% per possibility
			}
			.@equip_id = getequipid(.@part);
			if (.@equip_id != .@sarah_item_id) {
				mes "[首席助手]";
				mes "若要附魔，你必须装备莎拉的" + .@sarah_earring$ + "耳环。";
				close;

			setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
			if (.@card[2] > 0) {// 2 enchants
				mes "[首席助手]";
				mes "已完成全部附魔的莎拉耳环无法继续附魔。";
				close;
			}
			if (.@card[3] > 0) {// 1 enchant
				mes "[首席助手]";
				mes "莎拉耳环目前处于第4栏位已附魔的状态。";
				.@slot = 3;
			}
			else {// none enchant
				mes "[首席助手]";
				mes "莎拉耳环目前处于未附魔的状态。";
				.@slot = 4;
			}
			next;
			mes "[首席助手]";
			mes "你想附魔什么能力，在第" + .@slot + "栏位？";
			next;
			switch( select( "CRI或暴击", "神射手或出血", "省魔或MATK", "攻击延迟或技能延迟" ) ) {
			case 1:
				setarray .@bonus[0],
					4863,	// Fatal1
					4864,	// Fatal2
					4939,	// Critical2
					4940,	// Critical3
					4941,	// Critical4
					4865;	// Fatal3
				mes "[首席助手]";
				mes "已选择CRI或暴击附魔，栏位" + .@slot + ".";
				break;
			case 2:
				setarray .@bonus[0],
					4832,	// Expert_Archer1
					4833,	// Expert_Archer2
					4942,	// Dodge1
					4943,	// Dodge2
					4944,	// Dodge3
					4834;	// Expert_Archer3
				mes "[首席助手]";
				mes "已选择神射手或出血附魔，栏位" + .@slot + ".";
				break;
			case 3:
				setarray .@bonus[0],
					4945,	// Thrift1
					4946,	// Thrift2
					4897,	// Matk3p
					4898,	// Matk4p
					4947,	// Thrift3
					4899;	// Matk5p

				mes "已选择省魔或MATK附魔，栏位" + .@slot + ".";

					4869,	// Attack_Delay_1

			next;
			mes "[首席助手]";
			mes "一次附魔需要4个基勘特斯碎片。";
			next;
			mes "[首席助手]";
			mes "附魔是高风险的过程，莎拉耳环有可能会被破坏。";
			mes "你还是要进行吗？";

			if (select( "否", "是" ) == 1) {

				mes "请下定决心后再来找我！";
				close;
			}
			if (countitem(6803) < 4) {
				mes "[首席助手]";
				mes "基勘特斯的碎片不足。";

			}
			mes "[首席助手]";
			mes "很好。开始附魔！";
			mes "祝你好运～！";
			next;

			mes "啊啊啊！！";
