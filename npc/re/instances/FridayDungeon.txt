		'rand_chest++;

		donpcevent instance_npcname("#fd_box" + 'rand_chest) + "::OnStart";
	}
	sleep 10000;
	monster 'map_md_gef$,0,0, getarg(1), getarg(0),1, instance_npcname("md_gef_mobs_spawn") + "::" + getarg(2);
	end;

	'step = 0;
	'hard = 0;
	'count = 0;
	'rand_chest = 0;
	'freestones = 0;
	'boss_actv = 0;
	'map_md_gef$ = instance_mapname("1@md_gef");
	end;

				end;
			if (countitem(25235) < 10) {
				mes "[业余收藏家]";
				mes "请带给我10个" + mesitemlink(25235, false) + ".";
				close;
			}
			if (Zeny < 100000) {
				mes "[业余收藏家]";
				mes "我需要100,000 Zeny！";
				close;
			}
			delitem 25235,10; //Catalyst
			Zeny -= 100000;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@slot, .@equip_id) || callfunc("F_IsEquipCardHack", .@slot, .@card[0], .@card[1], .@card[2], .@card[3]))
				end;

			delequip .@slot;
			.@enchant = F_enchant(1);
			specialeffect2 EF_REPAIRWEAPON;
			mes "[业余收藏家]";
			mes "该" + mesitemlink(.@equip_id, false) + "已附上" + mesitemlink(.@enchant, false) + ".";
			getitem2 .@equip_id,1,1,0,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@enchant;
			close;
		}
		end;
	}
	else {	// reset
		if (.@equip_id == 28483) {
			.@pos = 2;
			.@success = 80;	// success royal
		} else {
			.@pos = 3;
			.@success = 20;	// success normal
		}
		if (.@equip_card[.@pos] == 0) {
			mes "[业余收藏家]";
			mes "该" + mesitemlink(.@equip_id, false) + "尚未达到最大附魔。";
			mes "我无法重置这件饰品。";
			close;
		}
		mes "[业余收藏家]";
		mes "将移除" + mesitemlink(.@equip_id, false) + "上的所有附魔效果吗？";
		mes "^FF0000有" + .@success + "%的成功率。如果重置失败，饰品将被破坏！^000000";
		next;
		if (select( "我稍后再来", "请继续") == 1)
			end;
		if (countitem(25235) < 10) {
			mes "[业余收藏家]";
			mes "请带给我10个" + mesitemlink(25235, false) + "";
			close;
		}
		if (Zeny < 100000) {
			mes "[业余收藏家]";
			mes "我需要100,000 Zeny！";
			close;
		}
		delitem 25235,10; //Catalyst
		Zeny -= 100000;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@slot, .@equip_id) || callfunc("F_IsEquipCardHack", .@slot, .@card[0], .@card[1], .@card[2], .@card[3]))
			end;

		delequip .@slot;
		if (rand(100) > .@success) {
			specialeffect2 EF_SUI_EXPLOSION;
			mes "[业余收藏家]";
			mes "可恶……";
			mes "我失败了，饰品已经被破坏。非常抱歉。";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[业余收藏家]";
		mes "我成功了。";
		mes "该" + mesitemlink(.@equip_id, false) + "已初始化。";
		getitem2 .@equip_id,1,1,0,0,.@equip_card[0],.@equip_card[1],0,0;
		close;
	}
	end;

function	F_equip_menu {
	.@l$ = getitemname(getequipid(EQI_ACC_L));
	.@r$ = getitemname(getequipid(EQI_ACC_R));
	.@acc_l$ = .@l$ == "null" ? "" : "Left Acc - " + .@l$;
	.@acc_r$ = .@r$ == "null" ? "" : "Right Acc - " + .@r$;

	switch( select( .@acc_l$, .@acc_r$ ) ) {
	case 1:
		.@slot = EQI_ACC_L;
		break;
	case 2:
		.@slot = EQI_ACC_R;
		break;
	}
	return .@slot;
}
