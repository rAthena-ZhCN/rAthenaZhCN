ma_fild02,312,317,5	script	守卫#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[守卫]";
		mes "^ff0000等级130^000000以下的人";
		mes "不能进入此处。";
		mes "这里很危险。请回去吧。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Buwaya Cave";

	if (!.@party_id) {
		mes "[守卫]";
		mes "宝箱巨鳄很危险。";
		mes "请先与其他";
		mes "的人组好队再来。";
		close;
	}
	if (is_party_leader() == true) {
		mes "[守卫]";
		mes "这里很危险。请回去吧。";
		mes "如果没有事，请回去吧。";
		next;
		while(1) {
			switch(select("为什么这么危险？:我是来讨伐宝箱巨鳄的。:结束对话。")) {
			case 1:
				mes "[守卫]";
				mes "最近发生了";
				mes "村民失踪的事件。";
				mes "追查足迹后，";
				mes "我们发现巨型怪物宝箱巨鳄";
				mes "就是元凶。";
				next;
				mes "[守卫]";
				mes "士兵和佣兵已被派去";
				mes "讨伐宝箱巨鳄，但从未";
				mes "完全成功。";
				next;
				mes "[守卫]";
				mes "宝箱巨鳄至今仍在";
				mes "绑架村民。";
				mes "请召集一些人手来除掉宝箱巨鳄。";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[守卫]";
					mes "宝箱巨鳄还躲着。";
					mes "就算你现在进去，宝箱巨鳄也不会出来。";
					mes "请稍后再来。";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[守卫]";
					mes "队伍名称是……"+getpartyname(.@party_id)+".";
					mes "队长是……"+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 现在无法开启。";
					mes "请稍后再试。";
					close;
				}
				mes "[守卫]";
				mes "我将开启通往 ^0000ff"+.@md_name$+"^000000。";
				mes "祝你好运。";
				close;
			case 3:
				mes "[守卫]";
				mes "这里很危险。你不能进入。";
				close;
			}
		}
	} else {
		mes "[守卫]";
		mes "让我和队长谈谈。";
		mes "请叫队长出来。";
		close;
	}
}

ma_fild02,315,323,0	script	洞穴入口#buwaya	45,2,2,{
	end;
OnTouch:
	if (BaseLevel < 130) {
		mes "[守卫]";
		mes "^ff0000等级130^000000以下的人";
		mes "不能进入此处。";
		mes "这里很危险。请回去吧。";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[守卫]";
		mes "宝箱巨鳄还躲着。";
		mes "就算你现在进去，宝箱巨鳄也不会出来。";
		mes "请稍后再来。";
		close;
	} else {
		erasequest 4229;
		// fall through
	}
	switch(select("进入。:返回。")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[守卫]";
			mes "哦，现在时机不太好。";
			mes "请稍后再试。";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[守卫]";
			mes "这里很危险。请不要进入。";
			close;
		case IE_OK:
			mapannounce "ma_fild02",getpartyname(getcharid(1))+"队伍的"+strcharinfo(0)+"成员开始在宝箱巨鳄洞穴讨伐宝箱巨鳄。",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "——因未知力量而无法进入。——";
			close;
		}
	case 2:
		mes "[守卫]";
		mes "你要珍惜自己的性命。";
		mes "行动前最好三思。";
		close;
	}
}
