// Main NPC :: mal_jerun
//============================================================
malangdo,221,174,6	script	霍林克#mal_cash	559,{
	disable_items;
	mes "[霍林克]";
	mes "我是喵~铁匠霍林克~";
	mes "精炼大师霍林克~";
	mes "我是在梦罗克学艺的特别猫咪霍林克~";
	mes "我女儿可为我骄傲了，霍林克~";
	mes "霍林克~今天要精炼什么呢？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? callfunc("F_EquipNameCN", getequipid(.@indices[.@i])) : F_getpositionname(.@indices[.@i]) +"-[未装备物品]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) {
		mes "[霍林克]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "我的老师阿拉甘说过喵~";
			mes "愚蠢是治不好的……";
			break;
		case EQI_ARMOR:
			mes "这里没什么好看的喵！！";
			break;
		case EQI_HAND_L:
			mes "喵？您想让我对这只左手做什么……？";
			break;
		case EQI_HAND_R:
			mes "喵？您想让我对这只右手做什么……？";
			break;
		case EQI_GARMENT:
			mes "喵？您身上什么都没装备。";
			break;
		case EQI_SHOES:
			mes "喵嗷~！别弄乱我敏感的嗅觉喵~。";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "喵？饰品在哪儿？";
			break;
		case EQI_HEAD_MID:
		case EQI_HEAD_LOW:
			mes "喵？您是在说其他头部装备吗喵~？";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[霍林克]";
		mes "这种东西连阿拉甘也精炼不了喵。";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[霍林克]";
		mes "喵~完美的精炼。这是阿拉甘做的吗喵~？";
		close;
	}
	mes "[霍林克]";
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
// Holink has different hardcoded prices
//	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "armor";
		mes "您选的是一件防具喵~";

		switch(.@equip_lv) {
			case 1:
				.@price = 15000;
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "weapon";

		switch( .@equip_lv ){
			case 1: // Level 1 Weapon
				.@price = 500;
				mes "1级武器……？";
				break;
			case 2: // Level 2 Weapon
				.@price = 2000;
				mes "喵，2级武器……？";
				break;
			case 3: // Level 3 Weapon
				.@price = 20000;
				mes "喵喵~~3级武器~~";
				break;
			case 4: // Level 4 Weapon
				.@price = 50000;
				mes "喵、喵！……4级武器……！";
				mes "我至今只见过两次，";
				mes "那是在跟阿拉甘学习的时候……喵、喵！！";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}

	mes "您需要^ff9999"+getitemname(.@material)+"^000000 以及^ff9999"+.@price+"^000000 Zeny 才能进行这次精炼喵~";
	mes "要继续吗喵~？";
	next;
	if(select("是！！:否！！") == 2) {
		mes "[霍林克]";
		mes "喵嗷！！";
		mes "您不相信精炼大师霍林克吗喵~？";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[霍林克]";
		mes "喵！！";
		if (.@type$ == "armor")
			mes "这件防具已经精炼过太多次了喵。";
		else {
			mes "危险。危险~";
			mes "这件武器已经精炼过很多次了喵~";
			next;
			mes "[霍林克]";
		}
		mes "如果再继续";
		mes "尝试进一步精炼这件物品，可能会把它弄坏喵。";
		next;
		mes "[霍林克]";
		mes "一旦"+.@type$+"损坏，您就";
		mes "再也无法使用它了喵。更不用说……当前所有的";
		mes "^ff0000卡片与附魔效果必定会消失^000000。";
		mes "您还想试试吗喵~？";
		next;
		if(select("是的，我要试！！:算了！！") == 2) {
			mes "[霍林克]";
			mes "喵！明智的选择喵。";
			mes "但是！！";
			mes "看到您怀疑精炼大师霍林克，我很不高兴喵~";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[霍林克]";
		mes "您没有所需的材料。";
		mes "您需要^ff9999"+getitemname(.@material)+"^000000 以及^ff9999"+.@price+"^000000 Zeny 喵~";
		mes "去取来吧喵~";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[霍林克]";
		emotion ET_FRET;
		mes "等一下……";
		mes "您以为我傻吗？！";
		mes "您趁我不注意把东西换掉了！给我出去！";
		close;
	}

	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		mes "[霍林克]";
		mes "喵~喵~喵！精炼真好玩~";
		next;
		emotion ET_CHUP;
		mes "[霍林克]";
		mes "完美！！完美喵！！";
		mes "我是精炼巫师阿拉甘的弟子~";
		mes "霍林克！！";
		mes "今天又是精炼成功的一天喵！！";
		close;
	}
	failedrefitem .@part;
	mes "[霍林克]";
	mes "喵~喵~喵嗷！！";
	next;
	switch(rand(1,5)) {
		case 1: emotion ET_CRY; break;
		case 2: emotion ET_PROFUSELY_SWEAT; break;
		case 3: emotion ET_KEK; break;
		case 4: emotion ET_SCRATCH; break;
		case 5: emotion ET_BIGTHROB; break;
	}
	mes "[霍林克]";
	mes "喵！！啊啊啊~~！！！！";
	mes "喵嗷！！我失败了喵！！";
	next;
	mes "[霍林克]";
	mes "......";
	mes "......";
	mes "全部~都~坏掉了喵……";
	next;
	mes "[霍林克]";
	mes "喵……阿拉甘老师曾经说过，";
	mes "要从失败中学习……";
	mes "人类，这一次失败会成为你未来成功的起点。";
	close;
}
