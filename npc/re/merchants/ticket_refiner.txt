//= 1.1 Do not refine above ticket level. [Euphy]

// Main NPC :: safety_Ref_NPC
//============================================================
prontera,184,177,6	script	精炼大师	851,{
	disable_items;
	setarray .@cert_weapon[0],
		6864,	// Guarantee_Weapon_19Up
		6875,	// Guarantee_Weapon_18Up
		6874,	// Guarantee_Weapon_17Up
		6873,	// Guarantee_Weapon_16Up
		6872,	// Guarantee_Weapon_15Up
		6871,	// Guarantee_Weapon_14Up
		6870,	// Guarantee_Weapon_13Up
		6584,	// Guarantee_Weapon_12Up
		6238,	// Guarantee_Weapon_11Up
		6228,	// Guarantee_Weapon_9Up
		6229,	// Guarantee_Weapon_8Up
		6230,	// Guarantee_Weapon_7Up
		6231,	// Guarantee_Weapon_6Up
		6456;	// Guarantee_Weapon_5Up
	setarray .@cert_armor[0],
		6865,	// Guarantee_Armor_19Up
		6881,	// Guarantee_Armor_18Up
		6880,	// Guarantee_Armor_17Up
		6879,	// Guarantee_Armor_16Up
		6878,	// Guarantee_Armor_15Up
		6877,	// Guarantee_Armor_14Up
		6876,	// Guarantee_Armor_13Up
		6585,	// Guarantee_Armor_12Up
		6239,	// Guarantee_Armor_11Up
		6232,	// Guarantee_Armor_9Up
		6233,	// Guarantee_Armor_8Up
		6234,	// Guarantee_Armor_7Up
		6235,	// Guarantee_Armor_6Up
		6457;	// Guarantee_Armor_5Up
	setarray .@cert_level[0],19,18,17,16,15,14,13,12,11,9,8,7,6,5;
	.@size_cert = getarraysize(.@cert_weapon);

	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@cert_weapon[.@i]) > 0 || countitem(.@cert_armor[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}

	if (.@check == 0) {
		mes "[精炼大师]";
		mes "你好！";
		mes "怎么了？";
		mes "我是专门";
		mes "精炼道具的专家，";
		mes "但我已经不再工作了。";
		next;
		switch(select("我还是先走吧。:嗯……这倒让我有点好奇。")) {
		case 1:
			mes "[精炼大师]";
			mes "保重，冒险者。";
			close;
		case 2:
			mes "[精炼大师]";
			mes "其实，我有时会用 ^006400精炼券^000000 为冒险者提供精炼服务……";
			mes "再见~！";
			close;
		}
	}
	emotion ET_SURPRISE;
	mes "[精炼大师]";
	mes "你好！";
	mes "我可以将道具精炼到 ^006400与你精炼券相同的等级^000000。";
	mes "不必担心！绝不会有损坏道具的风险。";
	next;
	if(select("我稍后再来。:用精炼券精炼道具。") == 1) {
		mes "[精炼大师]";
		mes "Okay.";
		mes "你可以稍后再来。";
		close;
	}
	mes "[精炼大师]";
	mes "你想精炼哪件装备？";
	next;
	setarray .@position$[1],"Head upper","Armor","Left hand","Right hand","Robe","Shoes","Accessory 1","Accessory 2","Head middle","Head lower";
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ += (getequipisequiped(.@indices[.@i]) ? callfunc("F_EquipNameCN", getequipid(.@indices[.@i])) : .@position$[.@i] + "- [空]") + ":";

	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[精炼大师]";
		mes "你必须先装备想要精炼的道具。";
		close;
	}
	if (!getequipisenableref(.@part)) {
		emotion ET_OTL;
		mes "[精炼大师]";
		mes "哦，抱歉。";
		mes "这件道具无法精炼。";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_WEAPON ){
		switch( getequipweaponlv( .@part ) ){
			case 1:
			case 2:
			case 3:
			case 4:
				copyarray .@tickets[0], .@cert_weapon[0], .@size_cert;
				.@type$ = "Weapon";
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_ARMOR ){
		switch( getequiparmorlv( .@part ) ){
			case 1:
				copyarray .@tickets[0], .@cert_armor[0], .@size_cert;
				.@type$ = "Armor";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	.@check = 0;
	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@tickets[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}
	if (.@check == 0) {
		emotion ET_THINK;
		mes "[精炼大师]";
		mes "如果你想精炼这件 ^006400"+.@type$+"^000000，请带上 ^006400"+.@type$+"精炼券^000000。";
		mes "再会！";
		close;
	}
	mes "[精炼大师]";
	mes "请选择你想要使用的 ^006400"+.@type$+"精炼券^000000。";
	next;
	.@menu$ = "";
	for ( .@i = 0; .@i < .@size_cert; ++.@i )
		.@menu$ += getitemname(.@tickets[.@i]) + ":";
	.@select = select(.@menu$)-1;
	.@ticket_lv = .@cert_level[.@select];
	.@ticket_id = .@tickets[.@select];

	if (countitem(.@ticket_id) == 0) {
		emotion ET_QUESTION;
		mes "[精炼大师]";
		mes getitemname(.@ticket_id)+"不在你的背包中。你是否把它放进仓库了？";
		mes "请再确认一次。";
		mes "再会！";
		close;
	}
	if (getequiprefinerycnt(.@part) >= .@ticket_lv) {
		emotion ET_PROFUSELY_SWEAT;
		mes "[精炼大师]";
		mes "^8B4513这件道具的精炼等级已经与你的精炼券相同。^000000";
		mes "请带来精炼等级低于你精炼券的道具。";
		close;
	}
	mes "[精炼大师]";
	mes "我将把 ^006400"+callfunc("F_EquipNameCN", getequipid(.@part))+"^8B4513 精炼至 +"+.@ticket_lv+"级^000000，使用 ^006400"+getitemname(.@ticket_id)+"^000000。";
	mes "可以开始吗？";
	next;
	if(select("否。:是。") == 1) {
		emotion ET_THINK;
		mes "[精炼大师]";
		mes "哦，你改变主意了。";
		mes "Ok.";
		mes "你可以稍后再来。";
		close;
	}
	mes "[精炼大师]";
	mes "Great.";
	mes "如你所愿！";
	mes "我自有独特的精炼方法……";
	mes ".......ka boom!";
	specialeffect EF_SUI_EXPLOSION;
	if (countitem(.@ticket_id) < 1) {
		next;
		mes "错误！";
		mes "请将此情况上报。";
		close;
	}
	delitem .@ticket_id,1;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes "[精炼大师]";
		emotion ET_FRET;
		mes "等一下……";
		mes "你以为我傻吗？！";
		mes "你竟趁我不注意把道具换掉了！给我出去！";
		close;
	}
	successrefitem .@part, .@ticket_lv - getequiprefinerycnt(.@part);
	next;
	emotion ET_DELIGHT;
	mes "[精炼大师]";
	mes "好了，就是这个~";
	mes "我说，^0000FF"+strcharinfo(0)+"^000000！";
	mes "祝贺你闪耀的"+.@type$+".";
	mes "你看起来棒极了！";
	mes "再会~！";
	close;
}
