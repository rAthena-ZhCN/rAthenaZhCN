S_Kvasir:

	.@fromname$ = getarg(2);

	.@toname$ = getarg(4);

		mes "您想把它换成" + .@toname$ + "?";
		next;
		mes "[咔哒咔哒喵喵-GX]";
		switch( select( "是！:离开" ) ){
			case 1:
				// Recheck the items once more
				if( countitem( .@fromid ) > 0 && countitem( 6615 ) > 9 ){
					mes "您已将" + .@fromname$ + "换成了" + .@toname$ + ".";
					delitem .@fromid, 1;

					close;
				}

				mes "您决定查看其他道具。";
				return;
		}
