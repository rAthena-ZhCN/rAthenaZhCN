//===== rAthena Script =======================================
//= Insurance Exchanger
//===== Description: =========================================
//= Exchange items for New_Insurance item.
//===== Changelogs: ==========================================
//= 1.0 Initial release. [Balfear]
//============================================================

prontera,82,108,5	script	心灵商人#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "由于你携带的物品过多，无法继续对话。";
		mes "请整理好物品后再试一次。";
		close;
	}
	mes "[心灵商人]";
	mes "你好！我看得出你是一位全心投入旅行、乘船横渡大陆、并享受危险刺激的冒险者！";
	next;
	if(!countitem(.@old_item)){
		mes "[心灵商人]";
		mes "你的仓库里有" + mesitemlink( .@old_item ) + "吗？";
		next;
		mes "[心灵商人]";
		mes "如果你带给我" + .@old_item_name$ + "，我就把它换成" + mesitemlink( .@new_item ) + ". What do you think? Do you have any idea what this item is?";
	} else {
		mes "[心灵商人]";
		mes "你有没有想过把你的" + mesitemlink( .@old_item ) + "换成更好的东西？比如……" + mesitemlink( .@new_item ) + ". What do you think?";
	}
	next;
	switch(select("那是什么？",(countitem(.@old_item)?"请帮我兑换！":""),"我不感兴趣")) {
	case 1:
		mes "[心灵商人]";
		mes mesitemlink( .@old_item ) + "是一种道具，使用后 30 分钟内死亡时，可免除一次经验值损失。";
		next;
		mes "[心灵商人]";
		mes "但很多人对时限感到遗憾，对吧？所以我研究了一番，准备了" + mesitemlink( .@new_item ) + ".";
		next;
		mes "[心灵商人]";
		mes mesitemlink( .@new_item ) + "是一种道具，只要持有它，^EE0000无论时限如何^000000，都能在死亡时免除一次经验值损失。当然，死亡之后，"+.@new_item_name$+"会被消耗。";
		next;
		select("你为什么要换掉它？");
		mes "[心灵商人]";
		mes "你好奇这个吗？我是商人。无论与人的生死是否相关，买卖东西都是我的宿命。你觉得我会错过赚钱的机会吗？我赚到钱，而活着的人也能从死亡的痛苦中得到一丝缓解。这不是很划算吗？";
		next;
		mes "[心灵商人]";
		mes "兑换比例是 5 张" + mesitemlink( .@old_item ) + "换 1 个" + mesitemlink( .@new_item ) + ". Do you have any questions?";
		next;
		mes "[心灵商人]";
		mes "嗯～你觉得如何？这不是很公平的交易吗？你想一直为时间担心吗？对于喜欢惊险冒险的冒险者来说，这可是必备之物。";
		close;
		
	case 2:	
		mes "[心灵商人]";
		mes "对对，好决定。换成更好的东西总是更划算，不是吗？";
		next;
		mes "[心灵商人]";
		mes "我可以用 ^0000cd5 张"+.@old_item_name$+"换 1 个"+.@new_item_name$+".";
		next;
		mes "[心灵商人]";
		mes "不过，^EE0000[非卖品]" + .@old_item_name$ + " is not eligible for exchange^000000, so it is useless to bring it here and ask for it to be exchanged.";
		next;
		mes "[心灵商人]";
		mes "现在要兑换吗？";
		next;
		if (select("请帮我兑换！:我不想兑换。") == 2)
			break;
		mes "[心灵商人]";
		mes "你想兑换多少？^0000cd一次最多可兑换 10 个"+.@new_item_name$+"^000000。";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[心灵商人]";
			mes "嗯？你不感兴趣吗？这可是好东西，对你这样的人来说真的很合适。不过如果你不想保住性命，我也不勉强你。";
			close;
		}
		if (.@amount > 10) {
			mes "[心灵商人]";
			mes "我说过了，^0000cd一次最多只能兑换 10 个^000000。请再确认一下数量。";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[心灵商人]";
			mes "呃，不好意思……你的" +.@old_item_name$+ "不够吧？是 5 张" + mesitemlink( .@old_item ) + "换 1 个" + mesitemlink( .@new_item ) + ". Please check the quantity.";
			close;
		}
		mes "[心灵商人]";
		mes "好了，这样我就把你的 ^0000cd" + .@old_item_name$ + "^000000 换成了 ^EE0000" + .@new_item_name$ + "^000000!! That was a good deal, wasn't it? The next time you get your hands on a "+.@old_item_name$+", come see me again!";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[心灵商人]";
	mes "这买卖太划算了！冒险者，对于像你这样在危险之地闯荡的人来说，这可是必备品！";
	next;
	mes "[心灵商人]";
	mes "如果你对它感兴趣了，一定要回来兑换。好吗？";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "你手上有好东西。要不要把你的" + getitemname(12209) + "换成更好的？", "", bc_self;
	end;
}
