job3_rune01,88,62,5	script	龙驯养师	105,{
	mes "[龙驯养师]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "欢迎。您要租借龙吗？";
		next;
		switch(select("是:否")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[龙驯养师]";
				mes "请先学会骑乘龙。";
				close;
			}
			else if (checkdragon()) {
				mes "[龙驯养师]";
				mes "您已经拥有一头龙了。";
				close;
			}
			else if (ismounting()) {
				mes "[龙驯养师]";
				mes "请先卸下您的商城坐骑。";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[龙驯养师]";
			mes "我明白了。祝您愉快。";
			close;
		}
	}
	mes "你在这里做什么？";
	mes "只有符文骑士才能租借龙。";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[魔导机甲整备师]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "欢迎，机匠。";
		mes "您要租用手推车，";
		mes "还是驾驶魔导机甲？";
		next;
		switch(select("租用手推车:驾驶魔导机甲:购买应急魔导机甲:升级冷却装置:取消")) {
		case 1:
			if (checkcart()) {
				mes "[魔导机甲整备师]";
				mes "抱歉，您已经";
				mes "拥有手推车了。";
				close;
			}
			setcart;
			mes "[魔导机甲整备师]";
			mes "给您！";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[魔导机甲整备师]";
				mes "抱歉，您已经";
				mes "在驾驶魔导机甲了。";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[魔导机甲整备师]";
				mes "请先学会技能取得魔导机甲执照。";
				close;
			}
			else if(ismounting()) {
				mes "[魔导机甲整备师]";
				mes "请先卸下您的商城坐骑。";
				close;
			}
			setmadogear;
			mes "[魔导机甲整备师]";
			mes "祝您玩得开心，欢迎再来！";
			close;
		case 3:
			mes "[魔导机甲整备师]";
			mes "应急魔导机甲在紧急情况下非常有用，售价为1,000,000 Zeny。";
			next;
			if (select("购买:取消") == 2) {
				mes "[魔导机甲整备师]";
				mes "我明白了。如果您改变了主意，";
				mes "请随时来找我。";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[魔导机甲整备师]";
				mes "抱歉，您已经拥有一台应急魔导机甲了。";
				close;
			}
			if (Zeny < 1000000) {
				mes "[魔导机甲整备师]";
				mes "抱歉，您的Zeny不足以购买应急魔导机甲。";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[魔导机甲整备师]";
			mes "给您！";
			close;
		case 4:
			mes "[魔导机甲整备师]";
			mes "您想升级哪个装置？";
			next;
			if (select("冷却装置:高品质冷却器") == 1) {
				mes "[魔导机甲整备师]";
				mes "将冷却装置升级为高品质冷却器需要1个冷却装置和2,000,000 Zeny。";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[魔导机甲整备师]";
				mes "将高品质冷却器升级为特殊冷却器需要1个高品质冷却器和4,000,000 Zeny。";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("升级:取消") == 2) {
				mes "[魔导机甲整备师]";
				mes "我明白了。如果您改变了主意，";
				mes "请随时来找我。";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[魔导机甲整备师]";
				mes "抱歉，您没有" + getitemname(.@itemid) + ".";
				close;
			}
			if (Zeny < .@cost) {
				mes "[魔导机甲整备师]";
				mes "抱歉，您的Zeny不足以升级该装置。";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[魔导机甲整备师]";
			mes "给您！这是属于您的" + getitemname(.@itemid) + ".";
			close;
		case 5:
			close;
		}
	}
	mes "有什么可以为您效劳的？";
	mes "魔导机甲仅供机匠使用。";
	close;
}
