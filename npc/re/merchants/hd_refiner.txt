//===== rAthena Script =======================================
//= HD Refiners
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use HD ores to refine equipment. Upon
//= failure, the equipment is not destroyed; rather, its
//= refine level decreases by 1. The success rate is identical
//= to that for Enriched ores.
//= - "Blacksmith Mighty Hammer" only refines from +7~9.
//= - "Basta" only refines from +10 and up.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Blacksmith Mighty Hammer (+7~9) :: cash_smelting79
//============================================================
-	script	::MightyHammer	-1,{
	disable_items;
	mes "[铁匠 强力锤]";
	mes "与别人不同，我是只精炼极少数物品的铁匠。";
	mes "我只精炼 ^CC0000+7 至 +9^000000 的物品。";
	next;
	mes "[铁匠 强力锤]";
	mes "我的特长是，即使精炼失败，也只会降低1级精炼度，装备不会消失。很棒吧？";
	next;
	mes "[铁匠 强力锤]";
	mes "那么我们就全力开动吧，怎么样？您想精炼哪件物品？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? callfunc("F_EquipNameCN", getequipid(.@indices[.@i])) : F_getpositionname(.@indices[.@i]) + "-[未装备]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[铁匠 强力锤]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "我是铁匠，不是理发师。";
			break;
		case EQI_ARMOR:
			mes "用我的锤子，我会让您成为天空中的明星。";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "制作义手可不是我的专长。";
			break;
		case EQI_GARMENT:
			mes "把物品拿出来吧，我来给您精炼！";
			break;
		case EQI_SHOES:
			mes "这股脚臭味是从哪来的？";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "饰品在哪里？";
			break;
		case EQI_HEAD_MID:
			mes "您想让我精炼什么？";
			break;
		case EQI_HEAD_LOW:
			mes "嗯？您想让我做什么？";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[铁匠 强力锤]";
		mes "这件物品无法精炼。";
		close;
	}
	switch( getequiprefinerycnt(.@part) ) {
	case 7:
		.@blacksmith_blessing_count = 1;
		break;
	case 8:
		.@blacksmith_blessing_count = 2;
		break;
	case 9:
		.@blacksmith_blessing_count = 4;
		break;
	default:
		mes "[铁匠 强力锤]";
		mes "我只处理精炼等级在+7到+9之间的物品。";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	mes "[铁匠 强力锤]";
	mes "要精炼您选择的装备，您需要 ^ff9999" + getitemname(.@material) + "^000000 以及20,000 Zeny的费用。";
	mes "您都准备好了吗？";
	next;
	if (select("是:否") == 2) {
		mes "[铁匠 强力锤]";
		mes "我会等您准备好。";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[铁匠 强力锤]";
		mes "这件物品看起来很可能会精炼失败。";
		mes "不过，就算失败，也只会降低1级精炼度。";
		mes "您要继续精炼吗？";
		next;
		if (countitem(6635) < .@blacksmith_blessing_count)
			setarray .@menu$[0], "", "是", "还没有";
		else {
			mes "[铁匠 强力锤]";
			mes "啊！是^0000ff铁匠的祝福^000000吗？";
			mes "有了铁匠的祝福，精炼失败时装备也不会消失！";
			next;
			mes "[铁匠 强力锤]";
			if ( .@itemtype != IT_WEAPON )
				mes "为+" + getequiprefinerycnt(.@part) + "的装备进行精炼时，使用^316AC5" + .@blacksmith_blessing_count + "个铁匠的祝福^000000即可防止装备消失。您想用它来精炼吗？";
			else
				mes "为+" + getequiprefinerycnt(.@part) + "的武器进行精炼时，使用^316AC5" + .@blacksmith_blessing_count + "个铁匠的祝福^000000即可防止装备消失。您想用它来精炼吗？";
			next;
			setarray .@menu$[0], "用它精炼", "不用它直接精炼", "暂不精炼";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[铁匠 强力锤]";
			mes "只有克服对失败的恐惧之人，才能获得杰作。";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[铁匠 强力锤]";
		mes "您刚才不是说都准备好了吗？";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[铁匠 强力锤]";
		emotion ET_FRET;
		mes "等一下……";
		mes "您以为我是傻瓜吗？！";
		mes "您趁我不注意把物品掉包了！给我出去！";
		close;
	}

	mes "[铁匠 强力锤]";
	mes "铛！铛！铛！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[铁匠 强力锤]";
		mes "每次听到这声音，都让我神清气爽。";
		mes "来，拿去吧。精炼完美成功！";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion ET_HUK;
		mes "[铁匠 强力锤]";
		mes "什么？！！";
		next;
		mes "[铁匠 强力锤]";
		mes "哎呀！我现在没脸见人了。哼哼。";
		close;
	}
	downrefitem .@part;
	next;
	emotion ET_HUK;
	mes "[铁匠 强力锤]";
	mes "糟糕！！";
	next;
	mes "[铁匠 强力锤]";
	mes "像您这样的人，肯定不会因为精炼度降低1级就怪我吧。嗯。";
	close;
}

// Basta (+10 and up) :: cash_smelting
//============================================================
-	script	::Basta	-1,{
	disable_items;
	mes "[巴斯塔]";
	mes "我是全世界最棒的铁匠，巴斯塔。";
	mes "不过我不提供普通的精炼服务。";
	mes "我只精炼 ^CC0000+10以上^000000 的装备。";
	next;
	mes "[巴斯塔]";
	mes "您想精炼哪件装备？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? callfunc("F_EquipNameCN", getequipid(.@indices[.@i])) : F_getpositionname(.@indices[.@i]) + "-[未装备]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[巴斯塔]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "您的头也算装备吗？";
			break;
		case EQI_ARMOR:
			mes "您想让我做什么？";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "制作义手可不是我的专长。";
			break;
		case EQI_GARMENT:
			mes "您到底知不知道什么是长袍？";
			break;
		case EQI_SHOES:
			mes "想精炼您的脚就别来找我，去跑马拉松吧。";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "饰品在哪里？";
			break;
		case EQI_HEAD_MID:
			mes "这个嘛……我没看到任何值得精炼的装备。";
			break;
		case EQI_HEAD_LOW:
			mes "我没办法让您变聪明。那事去找学校老师吧。";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[巴斯塔]";
		mes "这件物品连我也无法精炼。没办法。";
		close;
	}
	.@refine_count = getequiprefinerycnt(.@part);
	if (.@refine_count < 10) {
		mes "[巴斯塔]";
		mes "我不是说过了吗？我只精炼+10以上的装备。";
		close;
	}
	else if (.@refine_count == 10)
		.@blacksmith_blessing_count = 7;
	else if (.@refine_count == 11)
		.@blacksmith_blessing_count = 11;
	else if (.@refine_count == 20) {
		mes "[巴斯塔]";
		mes "这把武器堪称完美，无需再精炼了~";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	if ( .@itemtype != IT_WEAPON )
		.@type$ = "armor";
	else
		.@type$ = "weapon";
	mes "[巴斯塔]";
	mes "嗯……您想精炼的是这件吗？";
	mes "要精炼这件装备，我需要1个 ^ff9999" + getitemname(.@material) + "^000000 以及" + callfunc("F_InsertComma",.@price) + " zeny as a fee.";
	mes "您真的想精炼这件吗？";
	next;
	if (select("是:否") == 2) {
		mes "[巴斯塔]";
		mes "好吧。既然您这么想……";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[巴斯塔]";
		mes "这件" + .@type$ + "已经精炼得相当高了。";
		mes "如果再继续精炼，精炼度可能会下降。";
		next;
		mes "[巴斯塔]";
		mes "我和别处的铁匠可不一样。";
		mes "精炼度掉个3、4级什么的，那不可能……听起来太吓人了。";
		mes "在这里最多只会下降1级。";
		next;
		if (.@blacksmith_blessing_count == 0 || countitem(6635) < .@blacksmith_blessing_count) {
			mes "[巴斯塔]";
			mes "与其他铁匠相比，风险更小。";
			mes "我已经把所有注意事项都说了。您要试试吗？";
			next;
			setarray .@menu$[0], "", "是", "否";
		}
		else {
			mes "[巴斯塔]";
			mes "哇~是^316AC5铁匠的祝福^000000吗？这东西很难弄到~不过您居然有！";
			next;
			mes "[巴斯塔]";
			mes "为+" + .@refine_count + " " + .@type$ + "需要 ^316AC5" + .@blacksmith_blessing_count + "个铁匠的祝福^000000即可防止装备消失。您想用铁匠的祝福来精炼吗？";
			next;
			setarray .@menu$[0], "用铁匠的祝福精炼", "不用铁匠的祝福精炼", "暂不精炼";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[巴斯塔]";
			mes "这个嘛~";
			mes "完全不挑战，也可以说是人生的一种智慧。";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[巴斯塔]";
		mes "嗯……您没有带齐所需的材料。";
		mes "等您备齐了再来吧。";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[巴斯塔]";
		emotion ET_FRET;
		mes "等一下……";
		mes "您以为我是傻瓜吗？！";
		mes "您趁我不注意把物品掉包了！给我出去！";
		close;
	}

	mes "砰！砰！砰！砰！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[巴斯塔]";
		mes "太好了！干得漂亮！！";
		mes "我真的是全世界最棒的铁匠！";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[巴斯塔]";
		mes "啊啊啊啊啊啊！！！";
		next;
		mes "[巴斯塔]";
		mes "精炼失败了！";
		mes "像我这样世界最棒的铁匠……";
		mes "也不能保证100%成功~";
	}
	else {
		downrefitem .@part;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[巴斯塔]";
		mes "啊啊啊啊啊啊！！！";
		next;
		mes "[巴斯塔]";
		mes "可恶！";
		mes "精炼失败了，精炼度下降了！";
		mes "就算是世界最棒的铁匠，也不能保证100%成功！";
	}
	mes "真遗憾。";
	next;
	mes "[巴斯塔]";
	mes "下次我会做得更好！别担心！";
	close;
}
