//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "我可是史上最强的武器工匠！";
	mes "我不精炼普通又无趣的东西。";
	mes "我只精炼 10 级以上的物品。";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "总之，只要你的物品是 10 级以上，就可以找我。";
	mes "你想让我精炼什么？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "你身上的物品似乎都没法精炼……";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你没有装备";
		mes "任何我能精炼的";
		mes "东西。";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "看来我无法";
		mes "精炼这件物品……";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我说过，我不处理 10 级以下的物品。";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "这件物品无法";
		mes "继续精炼了。这已经是";
		mes "精炼到极限了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "嗯……" + .@article$ + " " + .@type$ + "，可以吗？";
	mes "如果你想精炼这件" + .@type$ + "，";
	mes "我需要 1 个 ^003366" + getitemname(.@material) + "^000000 以及" + callfunc("F_InsertComma",.@price) + " zeny.";
	mes "你确定要继续吗？";
	next;
	if (select("是:否") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯……如果你介意……那就算了……";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "这件"+.@type$+"已经精炼过好几次了。";
			mes "再尝试可能会被破坏。";
			mes "并非一定会损坏，但确实有很小的几率。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你可能会 ^FF0000降低升级等级^000000，这件"+.@type$+"，";
			mes "或者一旦损坏，你将失去 ^FF0000所有卡片^000000 以及附加的特殊属性。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你还要我继续精炼吗？";
			mes "我想我已经警告得够清楚了。";
			next;
			if(select("是。:否。") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "唉，不挑战也是一种选择……";
				mes "没有风险……或许也算明智。";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "嗯。你的钱似乎不够，或者缺少"+getitemname(.@material)+".";
			mes "准备好后再来吧。";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "你以为我傻吗？！";
			mes "你竟敢趁我不注意把物品掉包！给我滚出去！";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "铛！铛！铛！铛！";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "好！成功！！！";
			mes "我是最强的武器工匠。";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "铛！铛！铛！铛！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "天啊！";
				mes "升级等级下降了……";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "铛！铛！铛！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "哎呀！精炼失败了……";
				mes "我不是故意的！";
			}
			mes "就算我是史上最强的武器工匠，偶尔也会失手。";
			mes "只能说是命中注定。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我会做得更好！别担心！";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以将其精炼到安全上限，或按你想要的次数精炼。由你决定。";
		next;
		.@menu2 = select("请精炼到安全上限。","由我决定次数。","我改变主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "你想让我精炼几次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "这件物品无法精炼那么多次。";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "这将尝试精炼装备" + .@refinecheck + "次，超过安全上限。你的装备可能会被破坏……确定吗？";
			next;
			if(select("Yes...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "是你说的……那就这样吧。";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "是你说的……那就这样吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "这将花费你" + .@refinecnt + " " + getitemname(.@material) + "以及" + .@fullprice + "Zeny。可以吗？";
	next;
	if(select("是:否……") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "是你说的……那就这样吧。";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯。你的钱似乎不够，或者缺少"+getitemname(.@material)+".";
		mes "准备好后再来吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "喂……你身上什么都没装备……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "铛……不对，你以为我会那么蠢吗？！";
			mes "你把它换掉了……";
			mes "趁我还没用锤子把你敲晕，快滚！！";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "铛！铛！铛！铛！";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "铛！铛！铛！铛！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "天啊！";
				mes "升级等级下降了……";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "铛！铛！铛！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "哎呀！精炼失败了……";
				mes "我不是故意的！";
			}
			mes "就算我是史上最强的武器工匠，偶尔也会失手。";
			mes "只能说是命中注定。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我会做得更好！别担心！";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "好！成功！！！";
	mes "我是最强的铁匠。";
	close;
}

	mes "你需要 ^ff9999"+getitemname(.@material)+"^000000 以及 ^ff9999"+.@price+"^000000 Zeny。你有吗？";

	if(select("有！！:算了！！") == 2) {

		mes "[克林克]";

		mes "都会^ff0000彻底消失^000000。";

	if (countitem(.@material) == 0 || Zeny < .@price) {
