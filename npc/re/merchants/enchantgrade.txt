grademk,34,184,4	script	斯拉托斯#sratos	4_JP_GARM_H,{
	mes "[斯拉托斯]";
	mes "欢迎光临，亲爱的客人，愿今天能出现奇迹。";
	mes "有什么可以为您效劳的吗？";
	next;
	switch( select( "提升装备阶级", "兑换埃特尔道具", "什么是阶级提升？" ) ){
		case 1:
			mes "[斯拉托斯]";
			mes "客人，您是想提升装备的阶级吧。";
			mes "要转移宝石中的魔力并不容易。";
			next;
			mes "[斯拉托斯]";
			mes "魔力可能会当场爆发。";
			mes "那样继承魔力的武器也会一起爆炸！";
			next;
			switch( select( "我还是要做！", "我再考虑一下。" ) ){
				case 1:
					mes "[斯拉托斯]";
					mes "祝勇敢的客人好运！";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[斯拉托斯]";
					mes "等您有勇气挑战时，请再来。";
					close;
			}
		case 2:
			mes "[斯拉托斯]";
			mes "这些是用于提升阶级的宝石。";
			mes "埃特尔粉和作为基础的宝石，只要付一点手续费，我就为您换成埃特尔宝石。";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[斯拉托斯]";
			mes "偶尔会发现蕴含大量纯净魔力的不稳定宝石。";
			mes "我们把这些称为以太宝石，对吧？";
			next;
			mes "[斯拉托斯]";
			mes "您可以将以太宝石的魔力转移到其他武器上，当然会有一定的风险。";
			mes "它很脆弱，稍不留神就会“砰”地一声炸开。";
			next;
			mes "[斯拉托斯]";
			mes "总之，只要使用这颗以太宝石，就能强化您正在使用的武器。";
			mes "性能会变得更好。我们称之为阶级提升。";
			next;
			mes "[斯拉托斯]";
			mes "您也一样吧，心爱的武器若能更进一阶，难道不会高兴吗？";
			mes "我们就是协助此事的人。不过，稍微有点风险。啊哈哈哈。";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
