einbech,149,257,4	script	现场管理员埃尔门#extdq0	4_TOWER_16,{
	if (checkweight(25814,100) == 0 || checkweight(1201,1) == 0 || (MaxWeight - Weight) < 10000) {
		mes "- 你携带的物品过多，无法继续。 -";
		close;
	}
	mes "[埃尔门]";
	mes "你好，冒险者。";
	mes "如果你是来购买戴奈特采矿装备，或是将矿石精炼成戴奈特，那就来对地方了。";
	next;
	switch( select( "戴奈特精炼", "戴奈特商店", "戴奈特装备兑换" ) ) {
	case 1:
		npctalk "埃尔门：一次最多可精炼 100 个戴奈特。机器比较小。", "", bc_self;
		mes "[埃尔门]";
		mes "" + mesitemlink(25814) + "精炼矿石。";
		mes "" + mesitemlink(25809) + "^777777(当前" + countitem(25809) + ")^000000";
		mes "" + mesitemlink(25810) + "^777777(当前" + countitem(25810) + ")^000000";
		mes "" + mesitemlink(25811) + "^777777(当前" + countitem(25811) + ")^000000";
		mes "" + mesitemlink(25812) + "^777777(当前" + countitem(25812) + ")^000000";
		mes "" + mesitemlink(25813) + "^777777(当前" + countitem(25813) + ")^000000";
		mes "你需要五种矿石各一个。";
		next;
		input .@val;
		if (.@val > 100) {
			mes "[埃尔门]";
			mes "抱歉，冒险者。";
			mes "戴奈特一次最多只能精炼 100 个。机器太小，实在没办法。请见谅。";
			close;
		}
		if (.@val == 0) {
			mes "[埃尔门]";
			mes "取消？";
			mes "如果需要精炼戴奈特，随时来找我。除了我，可没人做得了。";
			close;
		}
		if (countitem(25809) < .@val || countitem(25810) < .@val || countitem(25811) < .@val || countitem(25812) < .@val || countitem(25813) < .@val) {
			mes "[埃尔门]";
			mes "喂！这样双方都很为难，你精炼所需的材料是不是不够了？材料！";
			close;
		}
		specialeffect EF_REPAIRWEAPON;
		progressbar_npc "FFFF00",1;
		delitem 25809,.@val;	// Ein_REDIUM
		delitem 25810,.@val;	// Ein_RINDIUM
		delitem 25811,.@val;	// Ein_ODIUM
		delitem 25812,.@val;	// Ein_PURDIUM
		delitem 25813,.@val;	// Ein_WHIDIUM
		getitem 25814,.@val;	// Ein_DYNITE
		mes "[埃尔门]";
		mes "" + .@val + "个戴奈特已精炼完成。今天状态绝佳！";
		close;
	case 2:
		mes "[埃尔门]";
		mes "这里有各种饰品和改造许可证。购买前请仔细确认所需物品。";
		close2;
		callshop "barter_ein_dun03";
		end;
	case 3:
		mes "[埃尔门]";
		mes "我可以用戴奈特兑换装备，不过装备已经装在密封严实的箱子里了。";
		mes "给我 50 个 ^4d4dff戴奈特^000000，我就为你打开一个 ^4d4dff武器箱^000000。";
		next;
		mes "[埃尔门]";
		mes "不知道会开出什么，但品质不错。要打开吗？";
		next;
		if (select( "闭上眼睛兑换。", "不兑换。" ) == 2) {
			mes "[埃尔门]";
			mes "啧，真浪费，明明是一笔好买卖。";
			close;
		}
		if (countitem(25814) < 50) {
			mes "[埃尔门]";
			mes "你的戴奈特不够啊，朋友。";
			mes "不知道会开出什么，但别处可很难用这个价格拿到这种装备。";
			close;
		}
		delitem 25814,50;
		// unknown text
		getgroupitem(IG_EIN_1HITEMBOX);
		end;
	}
}
