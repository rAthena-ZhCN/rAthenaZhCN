cmd_fild07,375,167,1	script	遗迹黑市商人#pa0829_01	4_F_JOB_HUNTER,{
	mes "[遗迹黑市商人]";
	mes "嘿，每次进出这里不觉得麻烦吗？只要 ^0000ff20,000^000000 Zeny，你就能拿到一张直接通往遗迹入口的卷轴，怎么样？";
Purchase:
	next;
	switch( select( "给我一张", "不用了" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "你携带的物品数量过多，无法继续对话。";
				mes "请整理物品后再试。";
				close;
			}
			else if (Zeny < 20000) {
				mes "[遗迹黑市商人]";
				mes "你似乎没有钱了。一张卷轴要 20,000 Zeny。";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[遗迹黑市商人]";
			mes "交易愉快。还要更多吗？";
			goto Purchase;
		case 2:
			mes "[遗迹黑市商人]";
			mes "随时再来";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	神器鉴定师#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "你携带的物品数量过多，无法继续对话。";
		mes "请整理物品后再试。";
		close;
	}
	.@stone_id = 6905;
	mes "[ 神器鉴定师 ]";
	mes "请选择你想购买的装备类型。只要你有" + getitemname(.@stone_id) + ".";
	next;
	switch (select("取消:武器:防具")) {
		case 1:
			mes "[ 神器鉴定师 ]";
			mes "随时欢迎你再来～";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "取消:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[ 神器鉴定师 ]";
			mes "随时欢迎你再来～";
			close;

		default:
			.@s--;
			mes "[ 神器鉴定师 ]";
			mes "您需要 ^0000ff" + .@price + "^000000 个" + getitemname(.@stone_id) + "，用来购买" + getitemname(.@equip_id[.@s]) + "~";
			next;
			if (select("取消:购买") == 1) {
				mes "[ 神器鉴定师 ]";
				mes "随时欢迎你再来～";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[ 神器鉴定师 ]";
				mes "你的" + getitemname(.@stone_id) + "不足，无法购买此物品。";
				close;
			}
			mes "[ 神器鉴定师 ]";
			mes "感谢你的信任。下次请再来。";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	神器强化师#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "你携带的物品数量过多，无法继续对话。";
		mes "请整理物品后再试。";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[ 神器强化师 ]";
	mes "如果你想强化在无限空间获得的装备，那就来对地方了。";
	mes "你有" + getitemname(.@stone_id) + "吗？";
	next;
	switch (select("如何给我的装备附魔？:为装备附魔。:初始化装备的附魔。")) {
		case 1:
			mes "[ 神器强化师 ]";
			mes "你将能够获得" + getitemname(.@stone_id) + "，只要你去探索帕罗斯灯塔下方的空间。";
			next;
			mes "[ 神器强化师 ]";
			mes "用那种材料，我可以强化你从无限空间获得的装备。";
			next;
			mes "[ 神器强化师 ]";
			mes "我只能附魔第 3 和第 4 个插槽。";
			next;
			mes "[ 神器强化师 ]";
			mes "附魔过程中装备不会损坏，但初始化附魔时损坏的概率很高。";
			next;
			mes "[ 神器强化师 ]";
			mes "那么想给装备附魔时，就来找我吧。";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[ 神器强化师 ]";
			mes "请选择你想附魔的装备。";
			next;
			switch (select("取消:武器:铠甲:鞋子:披风:头盔")) {
				case 1:
					mes "[ 神器强化师 ]";
					mes "那么想给装备附魔时，就来找我吧。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[ 神器强化师 ]";
				mes "我最多只能附魔到第 3 个插槽。你的装备无法再继续附魔了。";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[ 神器强化师 ]";
			mes "你可以选择 3 种附魔类型，附魔费用为" + .@fee + " " + getitemname(.@stone_id) + ".";
			mes "我会确保顺利附魔。";
			next;
			.@type = select("退出:物理:魔法:远程") - 2;
			mes "[ 神器强化师 ]";
			if (.@slot == 3)
				mes "好，开始第一次附魔。";
			else
				mes "好，进行第二次附魔。";
			next;
			if (select("我稍后再来。:请继续。") == 1) {
				mes "[ 神器强化师 ]";
				mes "如果改变主意，就来找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ 神器强化师 ]";
				mes "嗯。话说回来，你好像没在听，你的" + getitemname(.@stone_id) + ".";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[ 神器强化师 ]";
			mes "嗯。做得不错。快看看你的装备。";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[ 神器强化师 ]";
			mes "请选择你想附魔的装备。";
			next;
			switch (select("取消:武器:铠甲:鞋子:披风:头盔")) {
				case 1:
					mes "[ 神器强化师 ]";
					mes "那么想给装备附魔时，就来找我吧。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[ 神器强化师 ]";
				mes "你的装备没有任何附魔。";
				close;
			}
			mes "[ 神器强化师 ]";
			mes "初始化过程中，你的无限空间装备有可能被破坏。你还愿意继续吗？";
			next;
			if (select("退出:继续") == 1) {
				mes "[ 神器强化师 ]";
				mes "如果改变主意，就来找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ 神器强化师 ]";
				mes "嗯。话说回来，你好像没在听，你的" + getitemname(.@stone_id) + ".";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[ 神器强化师 ]";
				mes "嗯。做得不错。快看看你的装备。";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[ 神器强化师 ]";
				mes "唉，我警告过你了。运气不好吧？";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[ 神器强化师 ]";
			mes "你把装备脱下来了吗？";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[ 神器强化师 ]";
			mes "这件装备不适合附魔。别忘了，只有无限系列装备才能附魔。";
			close;
		}
		return;
	}
