moro_cav,32,73,5	script	英雄信物兑换员#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[英雄信物兑换员]";
	mes "若你曾与梦罗克交战";
	mes "并收集了";
	mes "英雄信物，";
	mes "我可以为你把它们兑换成一枚特别的戒指。";
	mes "你想看看这枚戒指吗？";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[英雄信物兑换员]";
			mes "你的背包里东西太多了，";
			mes "请整理好之后再来吧！";
			close;
		}
		switch( select( "查看戒指说明", "用戒指兑换。", "结束对话。" ) ) {
		case 1:
			mes "[英雄信物兑换员]";
			mes "这枚英雄戒指";
			mes "也许";
			mes "外表";
			mes "并不起眼，";
			next;
			mes "[英雄信物兑换员]";
			mes "但把它带去";
			mes "附魔专家伯恩斯那里，";
			mes "他就能对";
			mes "其中的全部 4 个插槽附魔";
			mes "并引出";
			mes "它潜在的能力。";
			next;
			mes "[英雄信物兑换员]";
			mes "英雄戒指";
			mes "能助你";
			mes "讨伐";
			mes "Morocc.";
			mes "还有什么疑问吗？";
			continue;
		case 2:
			mes "- 英雄戒指 -";
			mes "^006400为战士打造的戒指^000000";
			mes "^006400用于讨伐梦罗克的旅程……^000000";
			mes "^006400附魔后，^000000";
			mes "^006400会赋予持有者^000000";
			mes "^006400直面梦罗克的力量。^000000";
			mes "^006400类别：饰品／DEF：0^000000";
			mes "^006400重量：20／需求等级：160^000000";
			mes "^006400插槽：0／可装备职业：全职业^000000";
			next;
			mes "[英雄信物兑换员]";
			mes "英雄戒指需要";
			mes "30 个英雄信物";
			mes "才能兑换。";
			mes "要兑换吗？";
			next;
			if (select( "返回上一页。", "兑换 30 个英雄信物。" ) == 1) {
				mes "[英雄信物兑换员]";
				mes "再想一想";
				mes "again.";
				continue;
			}
			if (countitem(6684) < 30) {
				mes "[英雄信物兑换员]";
				mes "^006400英雄信物^000000";
				mes "会授予";
				mes "那些勇敢与梦罗克战斗的人。";
				mes "你没有";
				mes "30 个^006400英雄信物^000000，";
				mes "因此我无法帮助你。";
				continue;
			}
			mes "[英雄信物兑换员]";
			mes "我会用";
			mes "30 个^006400英雄信物^000000";
			mes "换 1 枚^006400英雄戒指^000000。";
			mes "还有别的事吗？";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
		case 3:
			mes "[英雄信物兑换员]";
			mes "总有一天";
			mes "你会发现";
			mes "自己需要英雄戒指。";
			close;
		}
	}
}

moro_cav,34,65,5	script	附魔专家伯恩斯	1_M_03,{
	disable_items;
	mes "[伯恩斯]";
	mes "你听说过英雄戒指吗？只有真正的英雄才配拥有那枚了不起的戒指，而我一直在等待它。";
	mes "我能为你做什么？";
	next;
	switch( select( "附魔英雄戒指。", "重置英雄戒指的附魔。", "Cancel." ) ) {
	case 1:
		mes "[伯恩斯]";
		mes "我只能为已装备的戒指附魔。你想为哪一枚已装备的戒指附魔？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左", "右" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[伯恩斯]";
			mes "请装备你想附魔的英雄戒指。";
			close;
		case 2981:
			break;
		default:
			mes "[伯恩斯]";
			mes "请装备你的英雄戒指。其他东西我无法附魔。";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[伯恩斯]";
			mes "已完全附魔的英雄戒指无法继续附魔。";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "Atk", "MAtk";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[伯恩斯]";
		mes "你想为第" + .@slot + "号插槽附加什么能力？";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[伯恩斯]";
		mes "附魔完成！";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[伯恩斯]";
		mes "要重置英雄戒指的附魔，我需要 3 个英雄信物。";
		next;
		if (countitem(6684) < 3) {
			mes "[伯恩斯]";
			mes "你的英雄信物不足。我无法重置你的英雄戒指。";
			close;
		}
		mes "[伯恩斯]";
		mes "我只能重置已装备的戒指。你想重置哪一枚已装备的戒指？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左", "右" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[伯恩斯]";
			mes "你必须先装备该物品，才能重置它的附魔状态。";
			close;
		case 2981:
			break;
		default:
			mes "[伯恩斯]";
			mes "请装备你的英雄戒指。其他东西我无法重置。";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[伯恩斯]";
			mes "你的英雄信物不足。我无法重置你的英雄戒指。";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[伯恩斯]";
			mes "你的英雄戒指目前处于未附魔状态。";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[伯恩斯]";
		mes "初始化完成。";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
