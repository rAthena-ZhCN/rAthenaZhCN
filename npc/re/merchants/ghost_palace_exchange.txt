dali02,51,130,4	script	次元旅行者	4_F_SHABBY,{
	if (checkweight(28100,1) == 0) {// Thanos Axe
		mes "- 因为携带的道具过多，无法继续。 -";
		close;
	}
	.@item_name_req$ = getitemname(6672);// Gray_Shard
	mes "[次元旅行者]";
	mes "冒险者朋友，" + mesitemlink( 6672 ) + "是我需要的。用我拥有的东西和您交换，如何？";
	next;
	.@type = select( "让我看看武器。", "让我看看防具。", "有什么特别的吗？", "我不需要。" ) - 1;
	mes "[次元旅行者]";
	switch(.@type) {
	case 0:
		mes "哦！您对武器感兴趣吗？幽灵宫殿能获得的武器我们都有。要不要挑一件？";
		break;
	case 1:
		mes "防具吗？幽灵宫殿能获得的防具我们都有。要不要挑一件？";
		break;
	case 2:
		mes "呵呵！看来很多人已经听说我这里有最好的东西了。请随意看看我的收藏。";
		break;
	case 3:
		mes "再见，朋友~ 再来啊！";
		close;
	}
	next;
	mes "[次元旅行者]";
	mes "啊！写在" + mesitemlink( 6672 ) + "旁边的数字就是所需数量。";
	mes "请记住这一点。";

	switch(.@type) {
	case 0:
		setarray .@items[0],
			13441, 200,		// Thanos Sword
			21009, 200, 	// Thanos Great Sword
			1438,  200,		// Thanos Spear
			1496,  200,		// Thanos Long Spear
			1669,  200,		// Thanos Staff
			2023,  200,		// Thanos Rod
			18119, 200,		// Thanos Bow
			13093, 200,		// Thanos Dagger
			28000, 200,		// Thanos Katar
			1836,  200,		// Thanos Knuckle
			16028, 200,		// Thanos Hammer
			28100, 200,		// Thanos Axe
			1933,  200,		// Thanos Violin
			1988,  200;		// Thanos Whip
		break;
	case 1:
		setarray .@items[0],
			2187,  100,		// Shield Of Gray
			15090, 100, 	// Armor Of Gray
			15091, 100,		// Gray Robe
			20721, 100,		// Cloak Of Gray
			22033, 100,		// Boots Of Gray
			18820, 100;		// Gray Helmet
		break;
	case 2:
		setarray .@items[0],
			27184, 10000,		// Knight Sakray Card
			23676, 50, 			// Gray Charcoal Melee (Melee)
			23677, 50,			// Gray Charcoal Magic (Magic)
			23678, 50;			// Gray Charcoal Range (Range)
		break;
	}
	.@size = getarraysize(.@items);
	for ( .@i = 0; .@i < .@size; .@i += 2 )
		.@menu$ += getitemname(.@items[.@i]) + " - " + .@items[.@i+1] + ":";

	while(true) {
		next;
		.@s = (select(.@menu$) - 1) * 2;
		mes "[次元旅行者]";
		if (.@type != 1)	// armor type doesn't display item info
			mes "您是说" + mesitemlink( .@items[.@s] ) + "吗？";
		mes "" + .@items[.@s+1] + "^006400" + .@item_name_req$ + "^000000个为交换所需。";
		next;
		switch( select( "我们交换吧。", "让我看看别的东西。", "我晚点再来。" ) ) {
		case 1:
			break;
		case 2:
			mes "[次元旅行者]";
			switch(.@type) {
			case 0:
				mes "我的武器已经够多了。";
				break;
			case 1:
				mes "我们有很多防具。要不要挑一件？";
				break;
			case 2:
				mes "如果您带" + .@item_name_req$ + "来，我就用它换给您任何想要的东西。";
				break;
			}
			continue;
		case 3:
			mes "[次元旅行者]";
			mes "保重，朋友~ 下次见！";
			close;
		}
		if (countitem(6672) < .@items[.@s+1]) {
			mes "[次元旅行者]";
			mes "朋友，您让我很为难。";
			mes "我需要" + .@item_name_req$ + ". I will exchange it for ^006400" + .@items[.@s+1] + " " + .@item_name_req$ + "^000000.";
			close;
		}
		mes "[次元旅行者]";
		mes "谢谢。如果还需要别的，请再来。";
		delitem 6672, .@items[.@s+1];
		getitem .@items[.@s], 1;
		close;
	}
}
