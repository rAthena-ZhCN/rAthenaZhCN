malangdo,220,167,5	script	金币兑换机 CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "您持有的物品种类过多。请减少部分物品后再继续。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "您持有的物品过重，无法继续。请减轻重量后再继续。";
		close;
	}
	mes "[硬币兑换机 CX-1]";
	mes "欢迎使用 CX-1。如果您持有硬币，可以将其兑换为更高或更低等级的硬币。";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "停止:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+"(无)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[硬币兑换机 CX-1]";
		mes "感谢您的光临。";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[硬币兑换机 CX-1]";
		mes "您选择了并未持有的硬币。";
		mes "感谢您的光临。";
		close;
	}
	mes "[硬币兑换机 CX-1]";
	mes "----兑换比率----";
	mes "向上兑换 - 30 换 10";
	mes "向下兑换 - 10 换 30";
	mes "------------------------";
	mes "^ff3333兑换时，将按当前市价收取手续费。^000000";
	next;
	setarray .@exchange_name$[0],"higher","lower";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Cannot exchange for Silvervine.
	.@menu$ = "停止:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333无法兑换为"+.@exchange_name$[.@i]+"硬币^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaa兑换为"+.@exchange_name$[.@i]+"硬币 (null)^000000:";
		else
			.@menu$ = .@menu$ + "兑换为"+.@exchange_name$[.@i]+"硬币 -"+getitemname(.@coin)+"("+.@exchange_rate[.@i]+"枚):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[硬币兑换机 CX-1]";
		mes "感谢您的光临。";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[硬币兑换机 CX-1]";
		mes "该硬币无法兑换为"+.@exchange_name$[.@i]+"硬币。";
		mes "感谢您的光临。";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[硬币兑换机 CX-1]";
		mes "您没有足够的硬币兑换为"+.@exchange_name$[.@i]+"硬币。";
		mes "感谢您的光临。";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[硬币兑换机 CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 单位的 ^003399"+getitemname(.@coin)+"^000000 已收取。";
	mes "已为您兑换成 ^003399"+.@exchange_name$[.@i]+" coins^000000, after deducting ^003399"+.@exchange_loss[(!.@i)]+"^000000 unit(s), ^003399"+.@exchange_total+"^000000 unit(s) remaining.";
	close;
}
