-	script	ZondaTeleportationServices	HIDDEN_NPC,{
	cutin "zonda_do01", 2;
	mes "[炫酷传送员]";
	mes "欢迎光临。";
	mes "我们始终努力提供超越期待的服务。";
	mes "请选择目的地。";
	next;
	// 6 个固定收费目的地 + 5 个章节传送目的地
	.@menu$ = "彩虹桥北部:埃克拉珠:辉煌领域:米德加尔特联合驻地:马努克:卡米达山麓";
	setarray .@free_map$[0], "lighthalzen", "rebel_in", "pub_cat", "sp_cor", "ba_maison";
	setarray .@free_x[0], 169, 74, 93, 107, 75;
	setarray .@free_y[0], 103, 67, 105, 129, 146;
	setarray .@free_label$[0], "里希塔乐镇", "反叛者总部", "喵星人酒吧", "柯尔特别警戒区", "霸肋门德大宅";
	setarray .@free_req$[0], "通关EP16.1解锁", "通关EP16.1解锁", "通关EP16.2解锁", "通关EP17.1解锁", "通关EP17.2解锁";
	.@ok[0] = (PRT_1QUE >= 25 || ep16_royal >= 17);
	.@ok[1] = (PRT_1QUE >= 25 || ep16_royal >= 17);
	.@ok[2] = (terra_gloria_main >= 25);
	.@ok[3] = (ep_illusion >= 47);
	.@ok[4] = (ep17_2_main >= 200);
	for (.@i = 0; .@i < 5; .@i++) {
		if (.@ok[.@i])
			.@menu$ += ":" + .@free_label$[.@i];
		else
			.@menu$ += ":^808080" + .@free_label$[.@i] + " [" + .@free_req$[.@i] + "]^000000";
	}
	.@menu$ += ":取消";
	.@sel = select(.@menu$);
	if (.@sel == 12) {
		mes "[炫酷传送员]";
		mes "感谢使用我们的服务。";
		close3;
	}
	if (.@sel > 6) {
		.@idx = .@sel - 7;
		if (!.@ok[.@idx]) {
			mes "[炫酷传送员]";
			mes "你尚未满足该目的地的进入条件。";
			mes "需要：" + .@free_req$[.@idx] + "。";
			close3;
		}
		mes "[炫酷传送员]";
		mes "正在为你传送。";
		close2;
		warp .@free_map$[.@idx], .@free_x[.@idx], .@free_y[.@idx];
		end;
	}
	switch(.@sel) {
	case 1:
		callsub( S_Zone, "彩虹桥北部", 17000,850, 7213 );
		warp "bif_fild02",160,230;
		end;
	case 2:
		callsub( S_Zone, "埃克拉珠", 17000,850, 11311 );
		warp "eclage",111,39;
		end;
	case 3:
		callsub( S_Zone, "辉煌领域", 17000,850, 10078 );
		warp "splendide",171,115;
		end;
	case 4:
		callsub( S_Zone, "米德加尔特联合驻地", 19000,950, 10078 );
		warp "mid_camp",210,288;
		end;
	case 5:
		callsub( S_Zone, "马努克", 22000,1100, 10078 );
		warp "manuk",263,177;
		end;
	case 6:
		callsub( S_Zone, "卡米达山麓", 24000,1200, 7184 );
		warp "dic_fild01",188,187;
		end;
	}
S_Zone:
	.@zeny_req = getarg(1);
	.@item_req = getarg(2);
	.@quest_req = getarg(3);
	
	if (isbegin_quest(.@quest_req) == 0) {
		mes "[炫酷传送员]";
		//mes getarg(0);
		mes "在使用传送服务前，必须先完成前往" + getarg(0) + "的通行任务。";
		close3;
	}
	mes "[炫酷传送员]";
	mes getarg(0) + "的传送费用为 ^ff0000" + .@zeny_req + " Zeny^000000，或 ^ff0000" + .@item_req + " 个绵绵岛特制罐头^000000。";
	next;
	switch( select( "支付 " + .@zeny_req + " Zeny", "使用 " + .@item_req + " 个绵绵岛特制罐头", "取消" ) ) {
	case 1:
		if (Zeny < .@zeny_req)
			break;
		Zeny -= .@zeny_req;
		return;
	case 2:
		if (countitem(12636) < .@item_req)
			break;
		delitem 12636,.@item_req; // Malang_Sp_Can
		return;
	case 3:
		mes "[炫酷传送员]";
		mes "请选择其他服务区域。";
		close3;
	}
	mes "[炫酷传送员]";
	mes "客人，您的费用不足。";
	mes "请准备好后再来。";
	close3;
}
