harboro1,356,211,5	script	向导#rockridge01	4_F_ANYA,{
	mes "[警长助手]";
	mes "欢迎来到岩脊，西部炎热干燥、冒险正在等待您的沙漠！";
	mes "我是警长助手，在这里为访客指路。您有想去的地方吗？";
	next;
	mes "[警长助手]";
	mes "告诉我您想去哪里，我会为您指出位置。";
	mes "另外，^B9062F点击我显示的地点名称^000000，即可使用^B9062F导航^000000模式。";
	while(true) {
		next;
		.@s = select( "[卡普拉职员]", "[旅馆]", "[武器/防具店]", "[道具店]", "[警长办公室]", "清除小地图。", "Cancel." );
		switch(.@s) {
		case 1:
			mes "[警长助手]";
			mes "^006400卡普拉员工^000000，好的。请点击小地图上的^0000FF方形图标^000000。";
			viewpoint 1,291,207,1,0xFFFF33;
			break;
		case 2:
			mes "[警长助手]";
			mes "我已在您的小地图上标出<NAVI>旅馆<INFO>harboro1,156,215,0,000,0</INFO></NAVI>的位置。";
			mes "点击名称即可开启导航模式。";
			viewpoint 1,156,215,2,0xAAFF00;
			break;
		case 3:
			mes "[警长助手]";
			mes "我已在您的小地图上标出<NAVI>武器/防具店<INFO>harboro1,312,193,0,000,0</INFO></NAVI>的位置。";
			mes "点击名称即可开启导航模式。";
			viewpoint 1,312,193,3,0x008080;
			break;
		case 4:
			mes "[警长助手]";
			mes "我已在您的小地图上标出<NAVI>道具店<INFO>harboro1,312,193,0,000,0</INFO></NAVI>的位置。";
			mes "点击名称即可开启导航模式。";
			viewpoint 1,312,193,4,0xFF1493;
			break;
		case 5:
			mes "[警长助手]";
			mes "我已在您的小地图上标出<NAVI>警长办公室<INFO>harboro1,241,218,0,000,0</INFO></NAVI>的位置。";
			mes "点击名称即可开启导航模式。";
			viewpoint 1,241,218,5,0xFFCCFF;
			break;
		case 6:
			mes "[警长助手]";
			mes "我来清除您小地图上的所有标记。";
			viewpoint 2,291,207,1,0xFFFF33;
			viewpoint 2,156,215,2,0xAAFF00;
			viewpoint 2,312,193,3,0x008080;
			viewpoint 2,312,193,4,0xFF1493;
			viewpoint 2,241,218,5,0xFFCCFF;
			break;
		case 7:
			mes "[警长助手]";
			mes "祝您在岩脊过得愉快！";
			close;
		}
		if (.@s == 7)
			mes "您还需要别的吗？";
		else
			mes "您还想查看其他地方吗？";
	}
}
