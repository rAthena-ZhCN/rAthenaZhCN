	mes "海上的乐园，";
	mes "美丽的波浪与温暖的阳光。";
	mes "^8B4513莫斯科比亚^000000正等着您。";
	mes "我在这里为您提供各类信息。";
	mes "若有任何疑问，尽管来问我。";
	F_Navi("[Moscovia Guide]");
	set .@str$,"Is there anything else I can do for you?";
	while (1) {
		next;
		switch(select("[ 卡普拉员工 ]:[ 王宫 ]:[ 礼拜堂 ]:[ 武器店 ]:[ 防具店 ]:[ 道具店 ]:[ 旅馆 ]:[ 返回阿尔贝塔 ]:清除小地图标记:取消")) {

			mes "[莫斯科比亚向导]";
			mes "^0000FF卡普拉员工^000000的";
			mes "位置已标记在您的小地图上，";
			mes "请查看您的小地图。";
			mes "还需要别的帮助吗？";
			viewpoint 1,223,191,0,0x0A82FF;
			break;
		case 2:
			mes "[莫斯科比亚向导]";
			mes "在"+F_Navi("王宫","moscovia,255,140","^006400")+"中，您能见到我们的沙皇陛下与其他侍从。";
			mes .@str$;
			viewpoint 1,255,140,1,0x0A82FF;
			break;
		case 3:
			mes "[莫斯科比亚向导]";
			mes "那座"+F_Navi("礼拜堂","moscovia,253,179","^008080")+"就在这里。";
			mes .@str$;
			viewpoint 1,253,179,2,0xAAFF00;
			break;
		case 4:
			mes "[莫斯科比亚向导]";
			mes "只要前往"+F_Navi("Weapon Shop","moscovia,186,191","^FF1493")+".";
			mes .@str$;
			viewpoint 1,186,191,3,0x008080;
			break;
		case 5:
			mes "[莫斯科比亚向导]";
			mes "最好的装备尽在莫斯科比亚的"+F_Navi("防具店","moscovia,204,169","^8B4513")+".";
			mes .@str$;
			viewpoint 1,204,169,4,0xB200FF;
			break;
		case 6:
			mes "[莫斯科比亚向导]";
			mes "那座"+F_Navi("道具店","moscovia,221,176","^FF1493")+"位于城镇中心以南。";
			mes "旅行所需的物品在这里都能找到。";
			mes .@str$;
			viewpoint 1,221,176,5,0xFF1493;
			break;
		case 7:
			mes "[莫斯科比亚向导]";
			mes "旅馆「"+F_Navi("黏草树","moscovia,227,206")+"」就在城镇中心以北。";
			mes "若想休息，没有比这里更好的去处。";
			mes .@str$;
			viewpoint 1,227,206,6,0x8B4513;
			break;
		case 8:
			mes "[莫斯科比亚向导]";
			mes "您可以在";
			mes F_Navi("莫斯科比亚公关官","moscovia,166,53")+"所在的码头。若您";
			mes "想返回阿尔贝塔，可以去找他。";
			viewpoint 1,166,53,7,0x00BFFF;
			break;
		case 9:
			mes "[莫斯科比亚向导]";
			mes "我会清除您小地图上的所有标记。";
			mes .@str$;
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 10:
			mes "[莫斯科比亚向导]";
			mes "祝您旅途愉快～！";
			close;
		}
	}
}
