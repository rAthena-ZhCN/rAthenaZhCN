L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[86-90任务公告板]";
		mes "你需要等待3小时才能再次接取这个任务。";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[86-90任务公告板]";
		mes "你已经猎杀完30只"+getarg(1)+getarg(5,"")+"了吗？";
		close;
	} else if (.@hunting == 2) {
		mes "[86-90任务公告板]";
		mes "你已完成猎杀。";
		mes "请收下这份奖励作为补偿。";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1
		erasequest .@quest1;
		mes "[86-90 任务板]";
		mes "你已重新接受"+getarg(1)+"的狩猎任务。";
		close;
	}*/ else {
		mes "[86-90任务公告板]";
		mes getarg(4);
		next;
		if (.@cutin$ != "") cutin .@cutin$,3;
		mes "[86-90任务公告板]";
		mes "猎杀30只"+getarg(1)+"魔物，并在此公告板上报告你的成果。";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("我去猎杀:不了，谢谢") == 1) {
			mes "[86-90任务公告板]";
			mes "完成任务后，请在此公告板上登记并领取奖励。";
			if (.@playtime > -1) erasequest .@quest2;
			setquest .@quest1;
		}
		close;
	}
