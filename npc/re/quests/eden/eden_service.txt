	mes "这是一个物品保管箱。";
	mes "纸上写着一段留言。";
	next;
	mes "+ 本保管箱专供 +";
	mes "+ 乐园团使用 +";
	mes "若要使用本";
	mes "保管箱，请确认";
	mes "以下事项：";
	next;
	mes "1.您是否为";
	mes "乐园团成员？";
	mes "2.您是否已学会";
	mes "足够的基础技能？";
	mes "3.保管箱费用为";
	mes "^4d4dff500 Zeny^000000！";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "使用保管箱需要投入 Zeny。";
		mes "费用：500 Zeny";
		mes "要使用吗？";
		next;
		switch(select("使用保管箱。:取消。")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "这个保管箱我无法使用。";
						mes "也许我还没有使用保管箱的资格。";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "我的 Zeny 不够。";
				mes "使用保管箱至少需要 500 Zeny。";
				close;
			case 2:
				mes "我下次再使用吧。";
				close;
		}
	}
	mes "使用这个保管箱需要乐园团徽章。";
	close;
}
