S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0))
		getexp getarg(2),getarg(3);
	return;
S_Erasequest:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 0:
		case 1:
			mes "[拉吉]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "关于"+ getarg(.@i+2) +"的故事就算了，我会去问别人。";
				break;
			case 2:
			case 3:
				mes "我会去问其他冒险者关于"+ getarg(.@i+2) +".";
				break;
			}
			break;
		case 2:
			mes "[拉吉]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "你听过关于"+ getarg(.@i+2) +"的故事了吗？谢谢你。我保证会写一首好歌。";
				getexp 1100000,300000;
				break;
			case 2:
				mes "梦罗克的故事够格写成歌吗？你做得很好。";
				getexp 1100000,200000;
				break;
			case 3:
				mes "你的冒险怎么样？有趣吗？哪天请让我听听你的冒险故事。";
				getexp 1100000,200000;
				break;
			}
		}
		erasequest .@quest_id;
		next;
	}
	return;
