function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "不过，我的全部力量只提供给VIP。";
		mes "不过我仍可为您开启通往"+getarg(1)+"的传送。";
		next;
		switch(select(getarg(0)+"-> 1800z:取消")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "很抱歉，您的";
				mes "zeny 不足以支付传送";
				mes "服务费用。";
				mes "传送到"+getarg(1)+"的费用为 1,800 zeny。";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}

L_Exchange:
	mes "[交易机器]";

	mes "我最多可以给您 60 个。";
	mes "输入 0 取消。";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[交易机器]";
		mes "Cancelled.";
		close;
	} else if (.@amount < 0 || .@amount > 60) {
		mes "[交易机器]";
		mes "我说了最多只能 60 个。";
		close;
	}
	mes "[交易机器]";
	mes "总计是"+.@amount+" "+getitemname(getarg(2))+".";
	mes "这样对吗？";
	next;
	if(select("好，都给我吧！:不，再等等。") == 2)
		close;
	set .@cost, getarg(1) * .@amount;
	set .@total_amount, getarg(3) * .@amount;
	if (countitem(getarg(0)) < .@cost) {
		mes "[交易机器]";
		mes "您的"+getitemname(getarg(0))+"不够，无法进行这项交易。";
		close;
	}
	delitem getarg(0),.@cost;
	getitem getarg(2),.@total_amount;
	close;
}
