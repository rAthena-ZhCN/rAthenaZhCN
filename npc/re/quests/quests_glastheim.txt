glast_01,195,273,3	script	福金的追随者#1	2_F_MAGICMASTER,{
	mes "[福金的追随者]";
	mes "我无法像福金大人那样在时间与次元之间移动，但我可以装作能做到。";
	next;
	mes "[福金的追随者]";
	mes "是什么让这座克雷斯特汉姆古城充满了如此强大的黑暗力量？";
	next;
	if( select( "我不感兴趣:这很有意思，请再多告诉我一些" ) == 2 ){
		mes "[福金的追随者]";
		mes "多亏了福金，我找到了进入这座旧克雷斯特汉姆古城的方法，这里充满了如此强大的黑暗力量。";
		next;
		mes "[福金的追随者]";
		mes "你觉得如何？想不想去那里探索？";
		next;
		if( select( "我不想探索:我很想去探索！" ) == 2 ){
			mes "[福金的追随者]";
			mes "当然，但这不是免费的服务！我必须得到一些回报才行。呵呵……";
			next;
			mes "[福金的追随者]";
			mes "我可以送你去两个地方。你想让我送你去哪里？";
			next;
			switch( select( "取消:修道院公墓（Lv100~130） - 5000z:幻象回廊2F（Lv140以上） - 10000z" ) ){
				case 1:
					mes "[福金的追随者]";
					mes "好吧，如果你改变主意，随时回来找我。我想你会有很多话想和我谈……";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[福金的追随者]";
				mes "什么？你以为这项服务是免费的吗？";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[福金的追随者]";
	mes "好吧，如果你不感兴趣，那就算了。";
	close;
}
