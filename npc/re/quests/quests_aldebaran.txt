c_tower1,232,222,5	script	贝尔杰明·巴顿#ct	968,{
	mes "[贝尔杰明·巴顿]";
	mes "这里的魔物能够操控时间。";
	if( BaseLevel < 150 ){
		mes "达到^0000FF150级^000000后再来吧。";
		close;
	}
	mes "时间、记忆、环境……在这里都可能发生扭曲。";
	next;
	mes "[贝尔杰明·巴顿]";
	mes "我求你别再追问了。那可能会让你有生命危险。";
	next;
	.@menu = select("不再追问:前往扭曲之塔2楼:前往扭曲之塔3楼");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[贝尔杰明·巴顿]";
			mes "请慎重选择。";
			next;
			mes "[贝尔杰明·巴顿]";
			mes "我需要^0000FF10,000z^000000才肯放你进去。那地方相当危险。";
			next;
			switch( select( "放弃:支付10,000z:为什么要收钱？" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[贝尔杰明·巴顿]";
						mes "真的吗？你没有钱，却要我带你去那里？";
						close;
					}
					Zeny -= 10000;
					mes "[贝尔杰明·巴顿]";
					mes "我不确定你在里面要怎么分辨时间，但请务必多加小心。";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[贝尔杰明·巴顿]";
					mes "我整天守在这里，是为了冒险者们的安全。我总会提醒他们其中的危险，所以才需要收取一些费用。";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[贝尔杰明·巴顿]";
	mes "呼……你做了个正确的选择。这地方很凶险，你最好还是回到安全的地方去。";
	close;
}
