				mes "这样才合理，不是吗？";
				mes "你不这么认为吗？";
				close;
			}
			mes "嗯，你可能不信，这个任务和一起盗窃案有关。";
			mes "我听说有一批"+ .@string$[2] +"被人从城堡里偷走了，地点是"+ .@fortress$ +".";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "小偷大概是趁攻城时段下的手。";
			mes "这是委托你寻找^4d4dff"+ .@num_items +"个以上的"+ .@string$[2] +"^000000。";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "TE地区的气氛比较轻松，";
		mes "只打攻城战也没意思吧？";
		mes "所以我带来了^4d4dff一些可以在攻城时段做的工作^000000～";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你可以在^4d4dff攻城开始前1小时起，到TE攻城战结束为止接受任务^000000，";
		mes "而且每个任务都有^4d4dff4小时的完成时限^000000。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "简单来说，就是在攻城时段从我这里接受任务，然后";
		mes "^4d4dff在时限内回来向我报告^000000。";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "试着玩玩看吧。";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "我明白了。";
		mes "改变主意的话随时再来。";
		mes "不过我不一定有空。呵呵呵。";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "看来你还没完成任务？";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "莫非你想取消？";
		next;
		switch( select( "No.","Yes." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "祝你好运！";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "如你所愿，这个任务已取消。";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "嗯，你还有剩余时间，但现在还不是攻城时段";
		mes "我觉得你不该继续。";
		mes "你应该取消这个任务。我看它已经算失败了。";
		next;
		switch( select( "我想继续。","好，我取消。" ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "我觉得不值得。那么，祝你好运！";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "如你所愿，这个任务已取消。";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "接受。","不接受。" ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dff时限为4小时。^000000";
		mes "在那段时间内完成任务，然后向我报告。";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "那就试试别的任务吧。";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "你什么也没找到。";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "你找到了“"+ getitemname(.@item_id[ .@miss_item01 ]) +"”！";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + "^4d4dff查看结果^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
