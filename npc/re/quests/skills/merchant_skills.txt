		mes "[杰森]";
		mes "纯灰色的手推车。";

		mes "[杰森]";
		mes "我可以把它改造成可爱的新款式。";

		case 1:

			mes "我想把这辆可怜的手推车彻底改造一番。";
			next;
			mes "[杰森]";
			mes "只稍微改造一下是不够的，你不觉得吗？";

			mes "[杰森]";
			mes "所以我发明了几款新设计。";

				close;

				mes "[杰森]";
				mes "手推车装饰可不容易学。";

				mes "[杰森]";

			mes "[杰森]";
			mes "你带来了可爱手推车改良券。确定要学手推车装饰吗？";
			next;
			if (select("是", "否") == 2) {
				mes "[杰森]";
				mes "等你改变主意再来吧！";
				close;
			}
			mes "[杰森]";
			mes "好！跟我来。";
			next;
			mes "[杰森]";
			mes "这个设计";
			mes "需要贴合手推车……";

			mes "这个设计";

			mes "然后加上这个……";
			mes "...";

			mes "[杰森]";

			mes "[杰森]";

			cart_remodel = 1;
			skill "MC_CARTDECORATE",1,SKILL_PERM;
			close;
		case 3:
			close;
		}
	}
	mes "[杰森]";
	mes "你装饰好手推车了吗？";

			mes "[杰森]";
			mes "你随时可以再来。";
			close;
		}
		if (countitem(6606) < 0) {
			mes "[杰森]";
			mes "你没有可爱手推车改良券吗？";

		mes "我已经把技能教给你了，现在可以试试。";

		skill "MC_CARTDECORATE", cart_remodel ,SKILL_PERM;
		close;

		mes "[杰森]";
		mes "哦？让我看看。";
		next;
		if (BaseClass != Job_Merchant) {
			mes "[杰森]";
			mes "等你能使用手推车后再来吧！";
			close;
		}
		mes "[杰森]";
		mes "你不记得我教过你什么了吗？";
		mes "我教过你这种方法。";
		next;
		mes "[杰森]";
		mes "你忘了？";
		mes "现在回想一下吧！";
		next;
		mes "[杰森]";
		mes "仔细想想，你会想起来的。";
