		mes "[可疑的猫咪]";
		mes "什么嘛～你不想听我的故事？好吧！那你在别处最好别提我的名字！";
		close;

	mes "看到木乃伊朝我走来时，我还想着“随便吧”，根本没在意。结果它一巴掌把我打倒在地！它怎么能这样对可爱的我！你这个缠满绷带的怪物……给我等着！";
	next;
	if(select("我也想去。:结束对话。") == 2)
		close;
	mes "[可疑的猫咪]";
	mes "不行！我花了那么多时间才找到那条路！！";
	next;
	mes "[可疑的猫咪]";
	mes "嗯……不过你坚持的话，就给我5000 Zeny吧。我来带路。";
	mes "什么～？别用那种眼神看我！你知道一只猫在米德加尔特生存有多难吗！";
	next;
	mes "[可疑的猫咪]";
	mes "^FF0000不过话说回来，像你这样的弱者在那里活不下去。运气好的话，也许能撑一秒？^000000";
	mes "算了，不关我的事。反正我已经警告过你了！";
	next;
	switch(select("支付5,000 Zeny出发。:算了。")) {
	case 1:
		if (Zeny < 5000) {
			mes "[可疑的猫咪]";
			mes "你想去哪儿！你穷得叮当响！";
			close;
		}
		set Zeny, Zeny-5000;
		mes "[可疑的猫咪]";
		mes "呸，这种旧钱算什么……你就没有闪闪发亮的东西吗？！";
		mes "啊，跟我来就好！";
		close2;
		warp "moc_prydn1",93,96;
		end;
	case 2:
		mes "[可疑的猫咪]";
		mes "天啊，你连5000 Zeny都没有吧？";
		close;
	}
}
