que_ng,21,72,6	script	防具名匠	4_M_RASWORD,{

	mes "你如何定义胜利？";
	mes "击倒敌人……";
	mes "然而，最重要的是……";
	mes "自己绝不能倒下！";

	mes "为此，你必须更好地保护自己。";
	mes "若连“自己”都护不住，便无法保护“他人”。";

	switch(select("狼纹护手[1]:月纹护手[1]:忍者鳞甲[1]:雾中暗影[1]:退出。")) {

		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[防具名匠]";
			mes "你能感受到狼的本性吗？";
			mes "既狂野又迷人……";
			close;
		}
		break;
	case 2:

		mes "<ITEM>月纹护手<INFO>2173</INFO></ITEM>……兼具美感与力量……二者皆不逊色。";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[防具名匠]";
			mes "何等美丽……";
			mes "仿佛全身都充盈着魔力。";
			close;
		}
		break;

		mes "<ITEM>忍者鳞甲<INFO>15054</INFO></ITEM>……据说曾有一位传说中的忍者穿过它。";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[防具名匠]";
			mes "这真是一件传说中的防具……";
			mes "即便我不相信传说……";
			close;
		}
		break;
	case 4:
		mes "[防具名匠]";
		mes "<ITEM>雾中暗影<INFO>15055</INFO></ITEM>……它是为了保护他人而造的。";

L_Select:
	.@string1$ = "查看材料。";
	.@string2$ = "查看属性。";
	while(1) {
		switch(select(.@string1$,.@string2$,"合成。","退出。")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[防具名匠]";
			mes "看来你需要更多时间考虑。";
			close;
		}
	}

L_Ingredients:
	mes "[防具名匠]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "要制作^ff0000<ITEM>狼纹护手<INFO>2172</INFO></ITEM>，需要1件<ITEM>狐纹护手<INFO>2171</INFO></ITEM>与10份<ITEM>狼血<INFO>6252</INFO></ITEM>^000000。";
		mes "你能感觉到吗？<ITEM>狼血<INFO>6252</INFO></ITEM>在搏动……";
		break;
	case 2173:	//Crescent_Armguard
		mes "要制作^ff0000<ITEM>月纹护手<INFO>2173</INFO></ITEM>，需要1件<ITEM>狐纹护手<INFO>2171</INFO></ITEM>与100个<ITEM>水晶碎片<INFO>7321</INFO></ITEM>^000000。";
		mes "其上将刻下一弯如水晶般美丽清澈的新月。";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "要制作^ff0000<ITEM>忍者鳞甲<INFO>15054</INFO></ITEM>，需要30片<ITEM>脱落的红色龙鳞<INFO>7451</INFO></ITEM>、30片<ITEM>冰块鳞片<INFO>7562</INFO></ITEM>与30片<ITEM>红鳞碎片<INFO>6091</INFO></ITEM>^000000。";
		mes "你能感受到它传说般的力量。";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "要制作^ff0000<ITEM>雾中暗影<INFO>15055</INFO></ITEM>，需要1件带洞的<ITEM>特攻忍装<INFO>15056</INFO></ITEM>与10个<ITEM>黑暗碎片<INFO>6089</INFO></ITEM>^000000。";
		mes "借由黑暗，它连存在本身都能抹去……";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "近战攻击时，有几率在5秒内攻击力+100、回避率-50。";
		mes "盾牌类，防御45，1洞";
		mes "要求等级100，仅限忍者系";
		break;
	case 2173:	//Crescent_Armguard
		mes "每强化1次，技能后延迟减少2%。";
		mes "盾牌类，防御70，1洞";
		mes "要求等级100，仅限忍者系";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%，MSP-30%";
		mes "铠甲类，防御90，1洞";
		mes "要求等级100，仅限忍者系";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "受到近战攻击时，有几率发动1级幻术-残影。";
		mes "铠甲类，防御60，1洞";
		mes "要求等级100，仅限忍者系";
		break;
	}
	next;
	return;

L_Combine:
	mes "[防具名匠]";
	mes "请务必确保^ff0000你手边备齐了正确的材料与装备^000000。";
	mes "事后再后悔也没用。";
	next;
	switch(select("合成。:退出。")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- 请等一下！！ -";
			mes "- 你无法获得物品 -";
			mes "- 因为你的负重过高。 -";
			mes "- 请减轻负重后 -";
			mes "- 减轻负重之后。 -";
			close;
		}
		return;
	case 2:
		mes "[防具名匠]";
		mes "看来你需要更多时间考虑。";
		close;
	}
	end;
}
