	while(1) {
		switch(select(.@string1$,.@string2$,"Combine.","Quit.")) {

			break;

			mes "看来你需要更多时间考虑。";

		mes "要制作^ff0000狼臂甲，需要1件狐臂甲与10份狼血^000000。";
		mes "你能感觉到吗？狼血在搏动……";
		break;

		mes "其上将刻下一弯如水晶般美丽清澈的新月。";

		mes "要求等级100，仅限忍者系";
		break;

		mes "MHP+15%，MSP-30%";
		mes "铠甲类，防御90，1洞";
		mes "要求等级100，仅限忍者系";
		break;
	case 15055:	//Tenebris_Latitantes

		mes "铠甲类，防御60，1洞";

	mes "请务必确保^ff0000你手边备齐了正确的材料与装备^000000。";
	mes "事后再后悔也没用。";

}
