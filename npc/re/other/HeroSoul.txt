	mes "[英灵之魂]";

	mes "目前可以唤醒的是跆拳系列职业的「跆拳英灵」。";

	mes "[英灵之魂]";
	mes "「跆拳英灵」持续30天，效果包括：";
	mes "跆拳踢技自动连招；";
	mes "平安/快乐休息随时激活；日月星地图自由更改；";
	mes "日月星憎恶自由更改；日月星恶魔视力恢复；日月星融合无需灵魂；";
	mes "日月星温暖自由使用；日月星平安感自由使用；";
	mes "艾斯麻无需前置要求；";
	mes "艾斯麻范围提升至3×3。";
	mes "艾斯诬取消晕眩且时长x5；";
	mes "凯阿希HP恢复升至5倍；";
	mes "灵魂积蓄速度升至3-30s/个；";
	mes "灵魂收割几率与时长翻倍；";
	mes "灵魂集结HP恢复量提升。";
	next;
	mes "[英灵之魂]";
	if (Class != JOB_TAEKWON && Class != JOB_STAR_GLADIATOR && Class != JOB_STAR_GLADIATOR2 && Class != JOB_SOUL_LINKER && Class != JOB_STAR_EMPEROR && Class != JOB_STAR_EMPEROR2) {
		mes "你的职业无法接受「跆拳英灵」。";
		close;

	if (hero_soul_expire > gettimetick(2)) {
		.@left = hero_soul_expire - gettimetick(2);
		mes "你已经拥有「跆拳英灵」。";
		mes "剩余时间：" + (.@left/86400) + "天" + ((.@left%86400)/3600) + "小时。";

		mes "你需要携带1个「英灵之魂器」。";
	mes "可在CashShop购买。";
		close;
	}
	if (select("兑换「跆拳英灵」30天","取消") == 2)
		close;
	delitem 50001,1;
	hero_soul_expire = gettimetick(2) + 2592000;
	sc_start SC_HERO_SOUL, 2592000000, 1;
	mes "[英灵之魂]";
	mes "「跆拳英灵」已觉醒。";
	mes "有效期30天，重新登录也会保留。";

-	script	HeroSoul_Login	-1,{
OnPCLoginEvent:
	if (hero_soul_expire > gettimetick(2)) {
		.@remain = hero_soul_expire - gettimetick(2);
		.@tick = .@remain * 1000;
		if (.@tick > 2592000000) .@tick = 2592000000;
		sc_start SC_HERO_SOUL, .@tick, 1;
	}
	end;
}

-	script	HeroSoul_Refresh	-1,{
OnInit:
	initnpctimer;
OnTimer43200000:
	.@n = getunits(BL_PC, .@rid);
	for ( .@i = 0; .@i < .@n; ++.@i ) {
		if (!attachrid(.@rid[.@i])) continue;
		if (hero_soul_expire > gettimetick(2)) {
			.@remain = hero_soul_expire - gettimetick(2);
			.@tick = .@remain * 1000;
			if (.@tick > 2592000000) .@tick = 2592000000;
			sc_start SC_HERO_SOUL, .@tick, 1;
		}
		if (vip_status(VIP_STATUS_ACTIVE)) {
			.@remain = vip_status(VIP_STATUS_REMAINING);
			.@tick = .@remain * 1000;
			if (.@tick > 2592000000) .@tick = 2592000000;
			sc_start SC_VIP, .@tick, 1;
		}
		detachrid;
	}
	initnpctimer;
}
