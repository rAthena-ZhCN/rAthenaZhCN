-	script	HeroSoul_Refresh	-1,{
OnInit:
	initnpctimer;
OnTimer43200000:
	.@n = getunits(BL_PC, .@rid);
	for ( .@i = 0; .@i < .@n; ++.@i ) {
		if (!attachrid(.@rid[.@i])) continue;
		if (hero_soul_expire > gettimetick(2)) {
			.@remain = hero_soul_expire - gettimetick(2);
			.@tick = .@remain * 1000;
			if (.@tick > 2592000000) .@tick = 2592000000;
			if (Class == JOB_NINJA || Class == JOB_KAGEROU || Class == JOB_OBORO)
