izlude,127,175,3	script	催眠师#novice	124,{
	mes "[催眠师]";
	mes "冒险者，你好。";
	mes "我是催眠师学院";
	mes "派来伊斯鲁得的成员";
	mes "在此提供技能重置服务";
	mes "为特定的一转职业角色提供超值价格：免费！";
	next;
	mes "[催眠师]";
	mes "虽然目前我提供无限次的";
	mes "能力值与技能重置，但我有";
	mes "两个必须满足的";
	mes "条件。第一，你必须低于";
	mes "^FF0000Base等级50^000000。";
	next;
	mes "[催眠师]";
	mes "第二，你必须是";
	mes "^FF0000剑士、服事、魔法师、";
	mes "盗贼、弓箭手、商人、";
	mes "跆拳少年或跆拳少女、";
	mes "神枪手或忍者^000000";
	mes "等职业角色才符合条件。";
	mes "那么，你有什么问题吗？";
	next;
	switch(select("^FF0000能力值/技能重置？:重置能力值:重置技能^000000:取消")) {
	case 1:
		mes "[催眠师]";
		mes "能力值/技能重置可让冒险者";
		mes "重新分配他们的技能";
		mes "点数，如果他们对";
		mes "目前的技能不满意。";
		next;
		mes "[催眠师]";
		mes "在进行";
		mes "能力值/技能重置前，你必须";
		mes "将身上所有携带物品的";
		mes "重量全部";
		mes "降为0。你可以把";
		mes "多余的物品放进卡普拉仓库。";
		next;
		mes "[催眠师]";
		mes "啊，还有一点也很重要，";
		mes "要卸下你的^FF0000手推车^000000";
		mes "如果你装备了的话。";
		mes "否则催眠不会";
		mes "生效，甚至会反噬……";
		close;
	case 2:
		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Skill";
		mes "[催眠师]";
		mes "你确定要";
		mes "进行我的";
		mes "^FF0000"+.@str$+"重置^000000服务？";
		next;
		if(select("是:取消") == 1)
			break;
	case 4:
		mes "[催眠师]";
		mes "谢谢，祝你";
		mes "冒险旅途好运。";
		mes "路上小心～";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[催眠师]";
			mes "哦！请先卸下你的";
			mes "手推车，再继续";
			mes "进行"+.@str$+"重置服务。";
			mes "谢谢配合～";
			close;
		}
		if (Weight != 0) {
			mes "[催眠师]";
			mes "如果你是为了我的"+.@str$;
			mes "重置服务而来，请";
			mes "记住，你无法";
			mes "重置你的"+.@str$+"直到";
			mes "^FF0000你背包中携带物品的重量";
			mes "降为0^000000。";
			next;
			mes "[催眠师]";
			mes "不如先把你的";
			mes "东西放进卡普拉仓库";
			mes "吧？这样你就能";
			mes "安全保管所有物品。";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[催眠师]";
			mes "抱歉，";
			mes "Base等级高于";
			mes "50的角色无法使用";
			mes .@str$+"重置服务。";
			close;
		}
		mes "[催眠师]";
		mes "感谢您使用";
		mes "我的"+.@str$+"再分配";
		mes "服务。哦，祝你";
		mes "旅途";
		mes "好运，冒险者。";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[催眠师]";
		mes "抱歉，你的";
		mes "职业不符合";
		mes "使用"+.@str$+" Reset service";
		mes "服务的资格。我只能";
		mes "提供"+.@str$+"重置给";
		mes "以下职业……";
		next;
		mes "[催眠师]";
		mes "^FF0000剑士、服事、";
		mes "魔法师、盗贼、弓箭手、";
		mes "商人、跆拳";
		mes "少年、跆拳少女、";
		mes "神枪手和忍者^000000。";
		close;
	}
}
izlude_a,127,175,3	duplicate(催眠师#novice)	催眠师#novice_a	124
