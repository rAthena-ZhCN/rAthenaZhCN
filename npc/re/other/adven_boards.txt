function	script	F_AvenBoard	{
	.@mo_board = select( "按楼层查询魔物",
					"魔物情报检索系统",
					( countitem(909) == 30000 ? "委托" : "" ),
					"取消" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[魔物情报检索系统]";
			mes "这里是魔物情报检索系统。可以查询本迷宫出现的魔物情报。搜索时请输入魔物名称，空格也需一并输入。";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Water","Earth","Fire","Wind","Poison","Holy","Shadow","Ghost","Undead";
			setarray .@size$[0],"Medium","Small","Big";
			setarray .@race$[0],"Formless","Undead","Animal","Plant","Insect","Fish","Demon","Demihuman","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[魔物情报检索系统]";
			mes "正在搜索";
			next;
			mes "[魔物情报检索系统]";
			mes "搜索完成。";
			mes "正在显示搜索结果。";
			next;
			mes "[魔物情报检索系统]";
			if (.@i >= .@size) {
				mes "没有该魔物的情报。";
				mes "请确认魔物名称。";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- 名称："+ getarg(.@i+1);
				mes "- 属性："+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- 体型："+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- 种族："+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[魔物情报检索系统]";
			mes "要搜索其他魔物的情报吗？";
			next;
			if (select( "搜索","取消" ) == 1)
				continue;
			mes "[魔物情报检索系统]";
			mes "魔物搜索系统即将关闭。";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "所需情报将会显示——只要能显示就没有问题";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "操作错误。";
		close;
	}
}

	next;
	return;
}
