//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

sec_in02,126,178,3	script	卡秋亚的秘密宝箱#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "一个没人知道里面装着什么的秘密宝箱。";
	mes "^4d4dff打开此宝箱需要消耗" + mesitemlink( .@key$ ) + ".^000000";
	next;
	switch(select("^4d4dff打开宝箱1次（1" + .@keyname$ + "）^000000","^4d4dff打开宝箱10次（10" + .@keyname$ + "）^000000")) {
		case 1:
			mes "^FF0000[公告]^000000";
			mes "^FF0000用一件随机物品交换一件" + .@keyname$ + ".^000000";
			mes "^FF0000以上交换所得的物品无法取回，也无法再兑换为" + .@keyname$ + ".^000000";
			next;
			if(select("继续。","结束对话。") == 2) {
				mes "你决定不打开宝箱。";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "数量不足：" + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dff请确认你的背包有足够的空间。^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + "已打开！";
				mes "你运气好吗？";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[公告]^000000";
			mes "^FF0000用10件随机物品交换10个" + .@keyname$ + ".^000000";
			mes "^FF0000以上交换所得的物品无法取回，也无法再兑换为" + .@keyname$ + ".^000000";
			next;
			if(select("继续。","结束对话。") == 2) {
				mes "你决定不打开宝箱。";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "您的数量不足" + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dff请确认你的背包有足够的空间。^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Custom check, just in case
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " was opened " + .@i + " times. Another one is being opened.",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + "已打开10次！";
				mes "你运气好吗？";
				close;
			}
	}
}
