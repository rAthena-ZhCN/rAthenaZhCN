	mes "[矿石鉴定师索蒂]";
	mes "你的背包里有没有^3131FF许多同种类的矿石^000000占用了大量空间？";
	mes "我可以帮你分辨矿石并立刻整理好～！";
	next;
	switch(select("整理我的矿石。:取消。")) {
	case 1:
		mes "[矿石鉴定师索蒂]";
		mes "目前我只能分辨并整理^FF0000钢铁^000000、^FF0000铁^000000、^FF0000铁矿石^000000和^FF0000神之金属原石^000000。";

		mes "那么，要我帮你整理矿石吗？";
		next;
		.@i = select("请整理钢铁。:请整理铁。:请整理铁矿石。:请整理神之金属原石。:不用了，谢谢。");
		if (.@i == 5) {
			mes "[矿石鉴定师索蒂]";
			mes "哼，我最讨厌邋遢的人了！";
			close;
		}
		setarray .@ores[0],999,998,1002,756;
		.@item = .@ores[.@i-1];
		if (countitem(.@item) == 0) {
			mes "[矿石鉴定师索蒂]";
			mes "你没有"+getitemname(.@item)+"...";
			close;
		}
		// while(1) {
			// if (countitem(.@item)) {
				// set .@count, countitem(.@item);
				// set .@total, .@total + .@count;
				// delitem .@item, .@count;
			// } else {
				// getitem .@item, .@total;
				// break;
			// }
		// }
		.@count = countitem(.@item);
		delitem .@item, .@count;
		getitem .@item, .@count;
		mes "[矿石鉴定师索蒂]";
		mes "整理好了。看起来很棒！你不觉得吗？";
		close;
	case 2:
		mes "[矿石鉴定师索蒂]";
		mes "看来你的背包很干净？呵呵！你也许是整理大师！";
