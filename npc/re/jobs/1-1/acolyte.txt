			mes "[马雷乌西斯神父]";
			mes "最后，我要衷心祝贺你坚持完成了苦修考验。";
			close;
		}

		mes "[马雷乌西斯神父]";
		mes "你想成为服事吗？";
		mes "那么，你必须完成以下条件，并把它们当作神赐予的修行。";

		close;
	}
