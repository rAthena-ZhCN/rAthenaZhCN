		sleep2 1800;
		set WH_LEAVE, 1;
		end;

		mes "现在还不能休息。";
		close;
	}
	mes "[生存日记]";
	mes "第 " + WH_DAY + " 天";
	mes "食物：" + WH_FOOD + "%";
	mes "饮水：" + WH_WATER + "%";
	mes "壁炉：" + ((WH_FIRE == 1) ? "已点燃" : "未点燃");

		set WH_STAGE, 4;
		close;
	}
	.@lost = 0;
	if (WH_FOOD < 40) {
		set WH_POINTS, WH_POINTS - 15;
		.@lost = .@lost + 15;
	}
	if (WH_WATER < 30) {
		set WH_POINTS, WH_POINTS - 15;
		.@lost = .@lost + 15;
	}
	if (WH_FIRE != 1) {
		set WH_POINTS, WH_POINTS - 10;
		.@lost = .@lost + 10;
	}
	if (WH_POINTS <= 0) {
		mes "[试炼失败]";
		mes "食物、饮水或温度长期不足，你失去了生存意志。";
		mes "你被送出了噜噜卡森林。";
		set WH_STAGE, 0;
		set WH_KEY, 0;
		set WH_DAY, 0;
		set WH_FOOD, 0;
		set WH_WATER, 0;
		set WH_WOOD, 0;
		set WH_FIRE, 0;
		set WH_POINTS, 0;
		set WH_LEAVE, 0;
		if (checkquest(5901) != -1) erasequest 5901;
		if (checkquest(5902) != -1) erasequest 5902;
		if (checkquest(5903) != -1) erasequest 5903;
		if (checkquest(5904) != -1) erasequest 5904;
		if (checkquest(5905) != -1) erasequest 5905;
		if (checkquest(5906) != -1) erasequest 5906;
		if (checkquest(5907) != -1) erasequest 5907;
		disablenpc instance_npcname("壁炉火焰#windhawk");
		warp "um_fild01",45,345;
		close;
	}
	set WH_FOOD, WH_FOOD - 40;
	if (WH_FOOD < 0) set WH_FOOD, 0;
	set WH_WATER, WH_WATER - 30;
	if (WH_WATER < 0) set WH_WATER, 0;
	set WH_FIRE, 0;
	set WH_WOOD, 0;
	disablenpc instance_npcname("壁炉火焰#windhawk");
	if (WH_PET == 1) {
		set WH_FOOD, WH_FOOD - 10;
		if (WH_FOOD < 0) set WH_FOOD, 0;
	}
	if (WH_TRAP1 == 1) {
		if (rand(100) < 50) set WH_TRAP1, 2;
		else set WH_TRAP1, 3;
	}
	if (WH_TRAP2 == 1) {
		if (rand(100) < 50) set WH_TRAP2, 2;
		else set WH_TRAP2, 3;
	}
	set WH_DAY, WH_DAY + 1;
	.@iid = instance_id(IM_PARTY);
	for ( .@i = 1; .@i <= 8; ++.@i )
		disablenpc instance_npcname("红草#" + .@i, .@iid);
	for ( .@i = 0; .@i < 8; ++.@i )
		set .@used[.@i], 0;
	.@want = rand(3, 6);
	.@got = 0;
	while ( .@got < .@want ) {
		.@idx = rand(8);
		if (.@used[.@idx] == 0) {
			set .@used[.@idx], 1;
			enablenpc instance_npcname("红草#" + (.@idx + 1), .@iid);
			.@got = .@got + 1;
		}
	}
	mes "[第" + WH_DAY + "天结算]";
	mes "生存意志损失：" + .@lost + " 点。";
	mes "食物存量：" + WH_FOOD + "%。";
	mes "饮水存量：" + WH_WATER + "%。";
	next;
	getmapxy(.@map$, .@x, .@y, BL_PC);
	warp .@map$,352,33;
	dispbottom "你睡了一觉。第" + WH_DAY + "天开始了。";
	close;
}
