		close;
	}

	mes "选择转动的方向。";
	switch(select("往东转:往西转:往南转:往北转")) {
		case 1:
			// 测试用：点击青蛙即重置进度并开启新副本
			.@old = instance_id(IM_PARTY);
			if (.@old) instance_destroy(.@old);
			set WH_STAGE, 1;
			set WH_KEY, 0;
			set WH_DAY, 0;
			set WH_FOOD, 0;
			set WH_WATER, 0;
			set WH_WOOD, 0;
			set WH_FIRE, 0;
			set WH_POINTS, 0;
			set WH_TOOL, 0;
			set WH_CARRY, 0;
			set WH_TRAP, 0;
			set WH_ANIMAL, 0;
			erasequest 5901;
			erasequest 5902;
			erasequest 5903;
			erasequest 5904;
			erasequest 5905;
			erasequest 5906;
			erasequest 5907;
			setquest 5901;
			completequest 5901;
			setquest 5902;
			completequest 5902;
			setquest 5903;
			.@iid = instance_create("Windhawk", IM_PARTY);
			if (.@iid < 0 && .@iid != -3) {
				mes "试炼暂时无法开始，请稍后再试。";
				close;
			}
			enablenpc "Windhawk_Portal#windhawk";
			mes "东边出现了一道传送门。";
			mes "入口在青蛙雕像背后。";
			close;
		default:
			mes "什么也没有发生。";
			close;
	}
}

			erasequest 5903;

			erasequest 5905;
			erasequest 5906;
			erasequest 5907;

			warp "um_fild01",45,345;
			close;
		}
		next;
	}
	set WH_FOOD, WH_FOOD - 40;
	if (WH_FOOD < 0) set WH_FOOD, 0;
	set WH_WATER, WH_WATER - 30;
	if (WH_WATER < 0) set WH_WATER, 0;

	mes "第 " + WH_DAY + " 天开始了。";
	close;
