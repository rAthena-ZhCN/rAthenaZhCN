// 噜噜卡森林：青蛙雕像
um_fild01,47,345,5	script	青蛙雕像#windhawk	4_FROG,{
	if (WH_STAGE < 1) {
		mes "一座被苔藓覆盖的青蛙雕像。";
		close;
	}
	if (getcharid(1) == 0) {
		mes "青蛙雕像的眼睛微微发亮。";
		mes "进入试炼需要组队，即使队伍里只有你一个人。";
		close;
	}
	if (getpartyleader(getcharid(1), 2) != getcharid(0)) {
		mes "只有队长才能启动青蛙雕像。";
		close;
	}
	mes "青蛙雕像发出低沉的鸣叫。";
	if (select("往东转动:离开") == 2) close;
	.@iid = instance_create("Windhawk", IM_PARTY);
	if (.@iid < 0 && .@iid != -3) {
		mes "试炼暂时无法开始，请稍后再试。";
		close;
	}
	instance_enter("Windhawk");
	close;
}

1@4win,348,19,0	script	床铺#windhawk	4_4job_silla,{
	if (WH_STAGE != 3) {
		mes "现在还不能休息。";
		close;
	}
	mes "[生存日记]";
	mes "第 " + WH_DAY + " 天";
	mes "食物：" + WH_FOOD + "%";
	mes "饮水：" + WH_WATER + "%";
	mes "壁炉：" + ((WH_FIRE == 1) ? "已点燃" : "未点燃");
	next;
	if (select("睡觉，结束这一天:再准备一下") == 2) close;
	if (WH_DAY >= 10) {
		mes "[风之猎鹰]";
		mes "十天过去了。";
		mes "南边的路已经通了，去第三片林地找帐篷的主人吧。";
		set WH_STAGE, 4;
		warp "1@4win",323,169;
		close;
	}
	if (WH_FOOD < 40 || WH_WATER < 30 || WH_FIRE != 1) {
		set WH_POINTS, WH_POINTS - 15;
		mes "食物、饮水或壁炉温度不足。";
		mes "你损失了 15 点生存意志。";
		if (WH_POINTS <= 0) {
			mes "你再也撑不住了，被送出了试炼。";
			set WH_STAGE, 0;
			set WH_KEY, 0;
			set WH_DAY, 0;
			set WH_FOOD, 0;
			set WH_WATER, 0;
			set WH_WOOD, 0;
			set WH_FIRE, 0;
			set WH_POINTS, 0;
			erasequest 70000;
			warp "payon",106,175;
			close;
		}
		next;
	}
	set WH_FOOD, WH_FOOD - 40;
	if (WH_FOOD < 0) set WH_FOOD, 0;
	set WH_WATER, WH_WATER - 30;
	if (WH_WATER < 0) set WH_WATER, 0;
	set WH_FIRE, 0;
	set WH_WOOD, 0;
	set WH_DAY, WH_DAY + 1;
	mes "你睡了一觉。";
	mes "第 " + WH_DAY + " 天开始了。";
	close;
}
