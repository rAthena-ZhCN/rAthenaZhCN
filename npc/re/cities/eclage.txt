ecl_in02,164,56,3	script	接待员#Laphine	835,{
	mes "[接待员]";
	mes "欢迎来到埃克拉珠诊所。";
	next;
	switch (select("保存位置:休息:结束对话")) {
	case 1:
		mes "[接待员]";
		mes "诊所已设为你的存档点。不过要小心哦~";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[接待员]";
		mes "在本诊所调养身心吧。这里充满了埃克拉珠纯净的能量。";
		mes "通常收费1枚辉煌领域金币，但我们对旅行者有特别优惠。一共是5,000 zeny。";
		next;
		switch (select("用辉煌领域金币支付:用Zeny支付:算了")) {
		case 1:
			if (!countitem(6081)) {
				mes "[接待员]";
				mes "你的金币似乎不够。请再确认一下你的账户好吗？";
				close;
			}
			mes "[接待员]";
			mes "请随意~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[接待员]";
				mes "你的资源似乎不足以继续。请你再确认一下账户好吗？";
				close;
			}
			mes "[接待员]";
			mes "请随意~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
