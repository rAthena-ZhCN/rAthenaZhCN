// TEMPORAL001 - 时光超越者之靴附魔
// 附魔槽位：card[3] = 第4洞强化阶段，card[2] = 第3洞特殊附魔
// 普通卡片 card[0]、card[1] 以及精炼、随机选项、词条均保留

function	script	F_TemporalBootsIdx	{
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] == 22202 && (@inventorylist_equip[.@i] & EQP_SHOES))
			return .@i;
	}
	return -1;
}

function	script	F_TemporalApply	{
	.@new_special = getarg(0);
	.@new_stage = getarg(1);
	.@idx = F_TemporalBootsIdx();
	if (.@idx < 0) return 0;

	getinventorylist;
	.@refine = @inventorylist_refine[.@idx];
	.@identify = @inventorylist_identify[.@idx];
	.@attribute = @inventorylist_attribute[.@idx];
	.@card1 = @inventorylist_card1[.@idx];
	.@card2 = @inventorylist_card2[.@idx];
	.@bound = @inventorylist_bound[.@idx];
	.@grade = @inventorylist_enchantgrade[.@idx];

	.@oid[0] = @inventorylist_option_id1[.@idx];
	.@oval[0] = @inventorylist_option_value1[.@idx];
	.@opar[0] = @inventorylist_option_parameter1[.@idx];
	.@oid[1] = @inventorylist_option_id2[.@idx];
	.@oval[1] = @inventorylist_option_value2[.@idx];
	.@opar[1] = @inventorylist_option_parameter2[.@idx];
	.@oid[2] = @inventorylist_option_id3[.@idx];
	.@oval[2] = @inventorylist_option_value3[.@idx];
	.@opar[2] = @inventorylist_option_parameter3[.@idx];
	.@oid[3] = @inventorylist_option_id4[.@idx];
	.@oval[3] = @inventorylist_option_value4[.@idx];
	.@opar[3] = @inventorylist_option_parameter4[.@idx];
	.@oid[4] = @inventorylist_option_id5[.@idx];
	.@oval[4] = @inventorylist_option_value5[.@idx];
	.@opar[4] = @inventorylist_option_parameter5[.@idx];

	.@before = countitem(22202);
	if (.@bound)
		getitembound4(22202,1,.@identify,.@refine,.@attribute,.@card1,.@card2,.@new_special,.@new_stage,.@bound,.@grade,.@oid,.@oval,.@opar);
	else
		getitem4(22202,1,.@identify,.@refine,.@attribute,.@card1,.@card2,.@new_special,.@new_stage,.@grade,.@oid,.@oval,.@opar);
	if (countitem(22202) <= .@before)
		return 0;

	unequip EQI_SHOES;
	delitemidx @inventorylist_idx[.@idx],1;
	if (countitem(22202) == 1)
		equip 22202;
	return 1;
}

function	script	F_TemporalRemove	{
	.@idx = F_TemporalBootsIdx();
	if (.@idx < 0) return 0;
	getinventorylist;
	.@inv_idx = @inventorylist_idx[.@idx];
	unequip EQI_SHOES;
	delitemidx .@inv_idx,1;
	return 1;
}

function	script	F_TemporalStage	{
	.@c = getarg(0);
	if (!.@c) return -1;

	setarray .@s1[0],4808,4832,4814,4775,4869,4785;
	setarray .@s2[0],4820,4833,4813,4776,4872,4753;
	setarray .@s3[0],4821,4834,4812,4861,4873,4754;
	setarray .@s4[0],4822,4835,4826,4862,4881,4755;

	for (.@i = 0; .@i < 6; .@i++) {
		if (.@c == .@s1[.@i]) return 10 + .@i;
		if (.@c == .@s2[.@i]) return 20 + .@i;
		if (.@c == .@s3[.@i]) return 30 + .@i;
		if (.@c == .@s4[.@i]) return 40 + .@i;
	}
	return -1;
}

function	script	F_TemporalNormal	{
	.@stage = getarg(0);
	.@series = getarg(1);
	setarray .@s1[0],4808,4832,4814,4775,4869,4785;
	setarray .@s2[0],4820,4833,4813,4776,4872,4753;
	setarray .@s3[0],4821,4834,4812,4861,4873,4754;
	setarray .@s4[0],4822,4835,4826,4862,4881,4755;
	if (.@stage == 1) return .@s1[.@series];
	if (.@stage == 2) return .@s2[.@series];
	if (.@stage == 3) return .@s3[.@series];
	return .@s4[.@series];
}

function	script	F_TemporalCrit	{
	.@series = getarg(0);
	setarray .@c0[0],4823,4824,4825;
	setarray .@c1[0],4836,4837,4838,4839;
	setarray .@c2[0],4827,4828,4829,4830;
	setarray .@c3[0],4867,4868,4900;
	setarray .@c4[0],311422;
	setarray .@c5[0],4756,4757,4758,4759;
	if (.@series == 0) return .@c0[rand(0,2)];
	if (.@series == 1) return .@c1[rand(0,3)];
	if (.@series == 2) return .@c2[rand(0,3)];
	if (.@series == 3) return .@c3[rand(0,2)];
	if (.@series == 4) return 311422;
	return .@c5[rand(0,3)];
}

function	script	F_TemporalPay	{
	.@need_p = getarg(0);
	.@need_c = getarg(1);
	.@coin = getarg(2);

	if (countitem(6755) >= .@need_p && countitem(6608) >= .@need_c) {
		if (countitem(1000620) < 1 || Zeny < 100000) {
			mes "[TEMPORAL001]";
			mes "你身上的强化材料不够哦。";
			return 0;
		}
		delitem 6755,.@need_p;
		delitem 6608,.@need_c;
		delitem 1000620,1;
		Zeny -= 100000;
		return 1;
	}

	mes "[TEMPORAL001]";
	mes "你身上的强化材料不够哦，可以改为" + .@coin + "个蹦级币 + 1个时光超越者的气息，是否使用？";
	if (select("使用" + .@coin + "个蹦级币+1个气息:取消") != 1)
		return 0;
	if (countitem(1000254) < .@coin) {
		mes "[TEMPORAL001]";
		mes "蹦级币不够。";
		return 0;
	}
	if (countitem(1000620) < 1) {
		mes "[TEMPORAL001]";
		mes "时光超越者的气息不够。";
		return 0;
	}
	delitem 1000254,.@coin;
	delitem 1000620,1;
	return 2;
}
