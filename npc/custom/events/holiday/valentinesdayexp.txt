// Stephen - Valentine Event Chocolate seller ------------------
alberta,26,243,4	script	史蒂芬#val1	58,{
	mes "[史蒂芬]";
	mes "猜我弄到了什么？";
	mes "一样在米德加尔特不容易找到的美味……";
	next;
	mes "[史蒂芬]";
	mes "巧克力！";
	mes "没错，你不喜欢巧克力吗……我可喜欢了。";
	mes "你运气不错，因为我一块只卖 5,000 zeny！";
	next;
	menu "我要买巧克力！",M_CHOCO,"不用了。",-;

	mes "[史蒂芬]";
	mes "你不要巧克力吗？";
	mes "我可提醒你！你会后悔的！";
	mes "你最好现在就买……这样的巧克力以后可再也遇不到了！";
	mes "好好想想，改天再来找我吧。";
	close;

M_CHOCO:
	mes "[史蒂芬]";
	mes "哈！";
	mes "我就知道！";
	mes "不过我一次最多只能卖你 5 块……但如果你真的需要更多……";
	mes "你可以再来一趟。";
	mes "那你要几块？";
	next;
	set @needmon,0;
	input @flag_num;
	if (@flag_num <= 0) goto L_NONE;
	if (@flag_num > 5) goto L_ERR;
	set @needmon,@flag_num*5000;
	if (Zeny < @needmon) goto L_NOTENO;
	set Zeny,Zeny - @needmon;
	getitem 558,@flag_num;
	mes "[史蒂芬]";
	mes "拿好了！";
	mes "你可以把它送给别人当礼物，也可以自己享用！";
	mes "嗯……甜甜的巧克力……";
	mes "随时来找我……！";
	close;

L_ERR:
	mes "[史蒂芬]";
	mes "很抱歉，我不能给你那么多。";
L_NONE:
	close;

L_NOTENO:
	mes "[史蒂芬]";
	mes "很抱歉，看来你买不起我的东西。";
	close;
}
