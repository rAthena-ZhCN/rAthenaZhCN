//===== Description: ========================================= 
//= A periodical quest of the Uneasy Cemetery (Kill undead / Prevent their appearance)
//= Every day, at the midnight Prontera receive a wave of Undeads.
//= They come from Uneasy Cemetery of Prontera. To protect the players
//= from the undeads terror you may either kill the enemy. Or supply Mother Mathana
//= with needed amount of Holy Water. Every citizen can take his part in the 
//= saving of Prontera city. After some days of quiet life... the Cemetery strikes back.
//===== Additional Comments: ================================= 
//= 1.1 More advanced ver. Added some bonus for the one who'd kill the last walking undead
//= 1.2 Added coords to the script to make label OmMobDead working
//= 1.2a Changed item names to item IDs. [Samuray22]

prontera,3,3,3	script	墓园异动检查	-1,{
	end;

OnHour00:
	set $UNEASY_DL,$UNEASY_DL-1;
	set $UNEASY_BL,$UNEASY_BL+30; //add need of HW for 30 bottles per day
	if ($UNEASY_BL>666) set $UNEASY_BL,666; //keep needed bottles not <=666
	if ($UNEASY_DL < 0) goto L_Start_Undead;
//The Cemetery is OK yet.
	disablenpc "玛莎娜修女";
	end;
OnInit:
	if ($UNEASY_DL >= 0) disablenpc "玛莎娜修女";
	end;

OnHour06:
	killmonsterall "prontera";	//The Sun kills undead in the morning
	end;

OnHour01:
	if ($@UNEASY_MOB > 0) mapannounce "prontera","[玛莎娜修女]: 以奥丁之名，请消灭这些四处游荡的残余亡灵！",0;
	end;

OnZombieDead:
	set $@UNEASY_MOB,$@UNEASY_MOB-1;
	if ($@UNEASY_MOB>0) end;
	set $UNEASY_DL,0;
	set $UNEASY_H$,strcharinfo(0);
	if (Sex==SEX_MALE) mapannounce "prontera","[玛莎娜修女]: 勇敢的"+$UNEASY_H$+"刚刚消灭了普隆德拉最后一只亡灵！",0;
	else mapannounce "prontera","[玛莎娜修女]: 尊贵的"+$UNEASY_H$+"刚刚消灭了普隆德拉最后一只亡灵！",0;
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	end;

L_Start_Undead:
	killmonsterall "prontera";	//kills any left monsters
	enablenpc "玛莎娜修女";
//call some monsters in the city
	set $@UNEASY_MOB, 65;
	areamonster "prontera",0,0,0,0,"Zombie",1015,30,"墓园异动检查::OnZombieDead";
//in the Cemetery
	monster "prontera",268,349,"Zombie",1015,30,"墓园异动检查::OnZombieDead";
	monster "prontera",269,350,"Ghoul",1036,5,"墓园异动检查::OnZombieDead";
//announce
	mapannounce "prontera","[玛莎娜修女]: 墓园已经不安分了！  以奥丁之名，快前往圣堂！拯救普隆德拉城！",0;
}

prontera,257,313,5	script	玛莎娜修女	79,{
	mes "[玛莎娜修女]";
	if ($UNEASY_DL <= 0) goto L_Undead_Walk;
	mes "恐怕我们那座古老的墓园出了些问题……";
	if ($UNEASY_H$==strcharinfo(0)) mes "但多亏了你，"+$UNEASY_H$+"，我们才能安睡" + $UNEASY_DL + "个夜晚！";
	if ($UNEASY_H$!=strcharinfo(0)) mes "但多亏了"+$UNEASY_H$+"的帮助，我们得到了" + $UNEASY_DL + "个安稳的夜晚！";
	emotion ET_SURPRISE;
	close;

L_Undead_Walk:
	if ($UNEASY_DL == 0) mes "它们明晚可能还会再来！";
	if ($UNEASY_DL == 0 && $UNEASY_H$==strcharinfo(0)) mes "谢谢你，"+$UNEASY_H$+"！现在我们总算能安歇到下一个午夜了！";
	if ($UNEASY_DL == 0 && $UNEASY_H$!=strcharinfo(0)) mes "但由于"+$UNEASY_H$+"的帮助，我们总算能安歇到下一个午夜了！";
	mes "要让不安分的墓园平静下来，我们得把所有坟墓都浇上圣水。但我们的兄弟姐妹已经把圣水用完了。";
	mes "你能为我们提供圣水吗？";
	next;
	menu "好，把我的圣水都拿去吧！",-, "不行，我要用。",M_NO, "我一点也没有。",M_DONT_HAVE;

	if ( countitem(523)<1 ) goto M_DONT_HAVE;
	set $UNEASY_BL,$UNEASY_BL-countitem(523);
	delitem 523,countitem(523);

	if ( $UNEASY_BL > 0 ) goto L_NEED_MORE;
//set quiet days!!! no more undead for this period!
	set $UNEASY_DL,5+((0-$UNEASY_BL)/30);
	set $UNEASY_H$,strcharinfo(0);
	mes "[玛莎娜修女]";
	mes "谢谢你，"+$UNEASY_H$+"！现在我们已经有足够的圣水了！";
	next;
	mes "[玛莎娜修女]";
	mes "把那些水浇遍墓园，我们就能得到" + $UNEASY_DL + "个平安的夜晚！";
	next;
	killmonsterall "prontera";	//kills any left monsters
	mes "[玛莎娜修女]";
	mes "看，"+ $UNEASY_H$ +"？它们现在都消失了！";
	next;
	mes "[玛莎娜修女]";
	mes "我们的教会将亲自向你致谢……";
	next;
	if (Sex==SEX_MALE) mapannounce "prontera","[玛莎娜修女]: 以奥丁之名，我们宣布英俊的"+$UNEASY_H$+"为普隆德拉的救世主！",0;
	else mapannounce "prontera","[玛莎娜修女]: 以奥丁之名，我们宣布美丽的"+$UNEASY_H$+"为普隆德拉的救世主！",0;
	mes "[玛莎娜修女]";
	mes "以奥丁之名，我们祝福你，并郑重献上一份薄礼，就出自马雷乌西斯家的酒窖。";
	getitem 505,1; //Blue_Potion
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	close;

L_NEED_MORE:
	mes "[玛莎娜修女]";
	mes "谢谢你，善良的"+strcharinfo(0)+"，但我们还需要" + $UNEASY_BL + "瓶圣水。";
	close;

M_NO:
	mes "[玛莎娜修女]";
	mes "恐怕那座古老墓园很快就要失控了……请你尽可能多为我们取得圣水。";
	close;

M_DONT_HAVE:
	mes "[玛莎娜修女]";
	mes "唉！我们还需要" + $UNEASY_BL + "瓶圣水……你为什么不去向其他人再讨些圣水呢？";
	mes "那座古老墓园很快就要失控了……";
	mes "求你了，以奥丁之名，救救普隆德拉城吧。";
	close;
}
