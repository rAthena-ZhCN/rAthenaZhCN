prt_castle,80,170,4	script	普隆德拉国王	108,{
	mes "[普隆德拉国王]";
	mes "你好，你能帮我吗？";
	next;
	if(select("当然，我能做什么？:抱歉，我很忙。")==2) {
		mes "[普隆德拉国王]";
		mes "好，没关系，我相信会有别人帮忙。";
		close; }
	mes "[普隆德拉国王]";
	mes "谢谢你。嗯，我旁边这女儿有点任性……";
	mes "如果你把这些物品给她，我相信她会安静下来。";
	mes "我会把王冠或宝剑作为报酬送给你。";
	next;
	setarray .@Items1[0],5007,742,30,741,50,753,20,732,1,710,20,629,20;
	setarray .@Items2[0],1137,2254,1,2282,1,714,1;
	switch(select("你需要什么？:我已经有物品了！:算了。")) {
		case 1:
			for(set .@i,1; .@i<3; set .@i,.@i+1) {
				mes "[普隆德拉国王]";
				mes "如果你能帮我拿到：";
				for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
					mes "~ ^0055FF"+getd(".@Items"+.@i+"["+(.@j+1)+"]")+"^000000x"+getitemname(getd(".@Items"+.@i+"["+.@j+"]"));
				mes "我会给你我的"+((.@i == 2 )?"sword.":"crown.");
				if (.@i==2) close;
				next; }
		case 2:
			mes "[普隆德拉国王]";
			mes "你想要哪件物品？";
			next;
			set .@i, select("王冠:宝剑");
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				if (countitem(getd(".@Items"+.@i+"["+.@j+"]")) < getd(".@Items"+.@i+"["+(.@j+1)+"]")) set .@nr,1;
			mes "[普隆德拉国王]";
			if (.@nr) { 
				mes "你缺了些东西。";
				mes "拿到所有物品后再回来。";
				close; }
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				delitem getd(".@Items"+.@i+"["+.@j+"]"), getd(".@Items"+.@i+"["+(.@j+1)+"]");
			getitem getd(".@Items"+.@i+"[0]"),1;
			mes "太感谢你了！！";
			mes "这一定能让她安静下来。";
			mes "如约献上我的"+((.@i==1)?"crown.":"sword.");
			close;
		case 3:
			mes "[普隆德拉国王]";
			mes "好，没关系，我相信会有别人帮忙。";
			close;
	}
}
