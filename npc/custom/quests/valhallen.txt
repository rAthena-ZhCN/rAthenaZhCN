prontera,147,171,5	script	蕾娜丝	811,{
	mes "[蕾娜斯]";
	mes "我从瓦尔哈拉而来，前来助你。";
	mes "我可以为你炼成许多瓦尔哈拉之物，";
	mes "不过制作它们需要许多稀有的素材。";
	next;
	switch(select("妙尔尼尔:斯雷普尼尔:布里希嘉曼:格莱普尼尔:梅金吉奥德:阿斯普里卡:布伦希尔德:荷尔薇:耶梦加得:穆斯佩里姆:符文精华:关闭")) {
		case 1:
			mes "[蕾娜斯]";
			mes "^3355FF妙尔尼尔^000000是托尔威力无穷的雷神之锤。";
			setarray .@Items[0],1530,1522,2,984,20,985,5,969,40,7074,2,7075,4,7078,5,7091,5,7089,5;
			break;
		case 2:
			mes "[蕾娜斯]";
			mes "^3355FF斯雷普尼尔^000000是仿照奥丁的战马制成的靴子。";
			setarray .@Items[0],2410,2406,2,969,20,985,10,7076,3,7079,5,7083,3,7086,4,7087,3;
			break;
		case 3:
			mes "[蕾娜斯]";
			mes "^3355FF布里希嘉曼^000000是美丽女神弗雷雅的神奇项链。";
			setarray .@Items[0],2630,2603,1,726,2,722,3,727,10,724,5,969,20,7073,4,7077,4,7088,3,7090,3,7092,3;
			break;
		case 4:
			mes "[蕾娜斯]";
			mes "^3355FF格莱普尼尔^000000是一根又轻又结实的绳索，制作^3355FF梅金吉奥德^000000时需要用到它";
			setarray .@Items[0],7058,7080,4,7081,5,7082,4,7084,3,7085,3;
			break;
		case 5:
			mes "[蕾娜斯]";
			mes "^3355FF梅金吉奥德^000000是托尔强大的腰带。";
			setarray .@Items[0],2629,7058,1,2627,1,969,20,726,10,984,10;
			break;
		case 6:
			mes "[蕾娜斯]";
			mes "^3355FF阿斯普里卡^000000是众神之光。";
			setarray .@Items[0],2541,7835,1,7836,1,7837,1,7838,1,2513,1,7063,100,7291,10,7293,10,985,20;
			break;
		case 7:
			mes "[蕾娜斯]";
			mes "^3355FF布伦希尔德^000000是战士的铠甲。";
			setarray .@Items[0],2383,7830,1,7831,1,7832,1,7833,1,7834,1,7510,100,2357,1,969,10,985,20;
			break;
		case 8:
			mes "[蕾娜斯]";
			mes "^3355FF荷尔薇^000000之名取自一位女武神，她总能令她的战士获得胜利。";
			setarray .@Items[0],2181,6595,2,6596,1,6594,4,6597,3,6602,1,6604,1,6605,1,2115,1;
			break;
		case 9:
			mes "[蕾娜斯]";
			mes "^3355FF耶梦加得^000000以能使敌人中毒而闻名。";
			setarray .@Items[0],2020,6603,4,6599,1,6598,1,6601,4,6600,1,6604,1,6605,1,1473,1;
			break;
		case 10:
			mes "[蕾娜斯]";
			mes "制作^3355FF耶梦加得^000000和^3355FF荷尔薇^000000需要用到^3355FF穆斯佩里姆^000000。";
			setarray .@Items[0],6605,984,50,985,50;
			break;
		case 11:
			mes "[蕾娜斯]";
			mes "制作^3355FF耶梦加得^000000和^3355FF荷尔薇^000000需要用到^3355FF符文精华^000000。";
			setarray .@Items[0],6604,12735,200,12736,100;
			break;
		default:
			close;
	}
	
	mes "^FF0000我需要的物品如下：^000000";
	set .@itemmissing,0;
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		mes "~"+.@amount+"x"+((countitem(.@id)<.@amount)?("^dd3355"):("^33dd55"))+getitemname(.@id)+((getitemslots(.@id)>0)?"["+getitemslots(.@id)+"]":"")+"^000000";
		if (countitem(.@id) < .@amount) 
			set .@itemmissing,1;
	}
	if (.@itemmissing) {
		next;
		mes "[蕾娜斯]";
		mes "去吧，把这些物品都找齐。";
		mes "我会在此等候，助你一臂之力。";
		close;
	}
	next;
	mes "[蕾娜斯]";
	mes "你想要^3355FF"+getitemname(.@Items[0])+"^000000吗？";
	next;
	if(select("否","是")==1) {
		mes "[蕾娜斯]";
		mes "若你改变主意，请再来。";
		close;
	}
	mes "[蕾娜斯]";
	mes "这是你的^3355FF"+getitemname(.@Items[0])+"^000000。";
	mes "愿它对你有所助益。";
	//second check if the player tries to bypass the first
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		if (countitem(.@id) < .@amount) 
			close;
	}
	// Delete the items _after_ the second check
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		delitem .@id, .@amount;
	}
	getitem .@Items[0],1;
	close;
}
