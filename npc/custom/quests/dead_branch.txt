//===== Description: ========================================= 
// Simple item trade-in quest, a person can get a Dead Branch
// by simply providing the items: Log x3 (7201), Wooden Heart x1 (7189), 
// Trunk x5 (1019), and Wooden Gnarl x1 (7222).
//===== Additional Comments: ================================= 
//= 1.1 Optimized, added Bloody Branch with 0.01% chance [Lupus]
//============================================================

niflheim,204,179,3	script	莫布里先生	121,{

	mes "[莫布里先生]";
	mes "你知道吗，树木是有灵魂的……";
	next;

	mes "[莫布里先生]";
	mes "其实我知道一个关于这些灵魂的秘密，或许能……在某种程度上帮到你……";
	next;

	mes "[莫布里先生]";
	mes "我可以做出一根有灵性的树枝……前提是你给我正确的材料……";
	next;
	goto L_MENU;

L_MENU:
	menu "我需要什么？",-,"这些破烂哪里找？",M_FIND,"给我做根枯树枝！",M_CREATE,"算了……",M_END;

	mes "[莫布里先生]";
	mes "嘿嘿嘿……看来你想操纵树之灵魂，对吧？好……我需要：";
	mes "^1354453 木块^000000";
	mes "^1354451 树木心脏^000000";
	mes "^1354455 树干^000000";
	mes "^1354451 木质瘤节^000000";
	next;
	goto L_MENU;

M_FIND:
	mes "[莫布里先生]";
	mes "下面这些物品可以从某些怪物身上弄到，嘿嘿……";
	mes "^135445木块可以从树精身上获得。^000000";
	mes "^135445树木心脏是树精的精髓。^000000";
	mes "^135445树干？动动脑子，跟木头有关……^000000";
	mes "^135445木质瘤节……鬼娃树……^000000";
	next;
	goto L_MENU;

L_NOLOG:
	mes "[莫布里先生]";
	mes "^135445木块^000000是躯体……我还需要更多。";
	mes "给我三块木块，我就给你做出灵魂……";
	close;

L_NOHEART:
	mes "[莫布里先生]";
	mes "^135445树木心脏^000000是灵魂的精髓。";
	mes "给我一个树木心脏，我就给你做出灵魂……";
	close;

L_NOTRUNK:
	mes "[莫布里先生]";
	mes "什么？连^1354455 树干^000000都找不到？";
	mes "去杀长老树精什么的啊……真是的……连树干都没有，哼……真没用。";
	close;

L_NOGNARL:
	mes "[莫布里先生]";
	mes "没有^135445木质瘤节^000000，我怎么做啊？";
	mes "我说的是鬼娃树……这附近到处都是……";
	close;

M_CREATE:
	mes "[莫布里先生]";
	mes "好吧……让我看看你给我带来了什么……";
	next;
	if(countitem(7201)<3) goto L_NOLOG;
	if(countitem(7189)<1) goto L_NOHEART;
	if(countitem(1019)<5) goto L_NOTRUNK;
	if(countitem(7222)<1) goto L_NOGNARL;
	delitem 7201,3;
	delitem 7189,1;
	delitem 1019,5;
	delitem 7222,1;
	mes "[莫布里先生]";
	mes "嘿……希望你跟树之灵魂玩得开心……小心点……";
	mes "嘿……小心点？你要是死了……算了，没什么。";
	if(rand(1000)==0) goto L_GIVE2;
	getitem 604,1; //Dead Branch
	close;
L_GIVE2:
	getitem 12103,1; //Bloody Branch
	close;

M_END:
	mes "[莫布里先生]";
	mes "没有灵魂给你……走吧！";
	close;
}
