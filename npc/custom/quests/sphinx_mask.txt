morocc,208,90,6	script	缠头盗贼	58,{
	mes "[缠头盗贼]";
	if(event_umbala < 2) {
		mes "你找我做什么？走开！";
		close; }
	if(sphmask_q) {
		mes "你和我没别的事了，走吧！";
		close; }
	mes "你好啊朋友，对这稀有的面具感兴趣吗？我向你保证，它的价值货真价实，朋友。真是太棒了！";
	next;
	callsub L_Menu,1,"1,000,000",1000000;
	callsub L_Menu,2,"750,000",750000;
	callsub L_Menu,3,"500,000",500000;
	next;
	mes "[缠头盗贼]";
	mes "呸！算了！我在别处能做得更好！";
	set sphmask_q,1;
	close;

L_Menu:
	mes "[缠头盗贼]";
	switch(getarg(0)) {
		case 1: mes "哇？哦呵呵呵，所以你知道这是我从那些野蛮人手里偷来的吧？我不会还回去。不过给个价钱的话……你说呢？"; break;
		case 2: mes "啊，你还是个生意人啊？好吧，那么……"; break;
		case 3: mes "嗯……你真会砍价，好吧……我的最后报价……"; break; }
	next;
	if(select("支付"+getarg(1)+"z:不成交")==2) return;
	mes "[缠头盗贼]";
	if (Zeny < getarg(2)) {
		mes "你在耍我吗？你根本没那么多钱！";
		close; }
	set Zeny, Zeny-getarg(2);
	mes "哦呵呵，那就成交！";
	getitem 7114,1;
	set sphmask_q,1;
	close;
}
