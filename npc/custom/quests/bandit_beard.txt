umbala,126,129,4	script	胡须男子#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "[大胡子]";
	mes "胡须！来买上好的胡须！";
	mes "想要一副胡须吗？";
	next;
	menu "我要胡须！",-,"不了，太恶心了！",M_NO;

	mes "[大胡子]";
	mes "呵呵～原来你想要胡须啊。";
	emotion ET_SMILE;
	mes "我倒是想给你胡须，可它们可不便宜。";
	mes "其实我只剩一副了……";
	mes "但我不卖它。";
	mes "别生气！我可以给你做一副。";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "[大胡子]";
	mes "做胡须我需要一些^8080FF线^000000。";
	mes "去伊斯鲁得找我的朋友拿些^8080FF线^000000。";
	next;

	mes "[大胡子]";
	mes "另外，我还需要你帮我找些做胡须的材料。";
	mes "它们是：";
L_LIST:
	mes "^8080FF橡皮筋^000000";
	mes "^8080FF5 动物外皮^000000";
	mes "^8080FF黑色染料^000000";
	mes "^8080FF爷爷的胡须^000000";
	mes "^8080FF100 黏稠液体^000000";
	mes "^8080FF中和剂^000000";
	mes "还有一颗^8080FF2克拉钻石^000000";
	close;

M_NO:
	mes "[大胡子]";
	mes "随你便吧！";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "[大胡子]";
	mes "抱歉，我每人只能做一副。";
	mes "胡须太多可不行，对吧？";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "[大胡子]";
	mes "来给你做胡须吧！";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080FF~只见他剪裁、缝制着胡须~^000000";
	next;
	mes "^8080FF~他把做好的胡须递给你~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "[大胡子]";
	mes "祝你过得愉快！";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "[大胡子]";
	mes "搞什么！没有材料我可做不出胡须！";
	mes "再给你看一遍清单：";
	emotion ET_HUK;
	goto L_LIST;
}

alberta,65,123,6	script	男子#bandit	85,{
	if(BEARD_QUEST == 3 && $@beardMobD) goto L_KILLED;
	if(BEARD_QUEST > 3) goto L_DONE;

	mes "[线商]";
	mes "啊啊啊！！我的货永远送不到艾尔帕兰！";
	mes "老是被一群虫子围攻。";
	mes "它们抢走了我所有的货物和^8080FF线^000000。";
	next;
	mes "[线商]";

	if(BEARD_QUEST != 3) mes "我该怎么办啊？";
	if(BEARD_QUEST != 3) close;

	if($@beardmob > 0) mes "杀！把它们全杀掉！！！";
	if($@beardmob > 0) close;
	mes "你是来帮我的吗？";
	emotion ET_QUESTION;
	next;
	menu "是",-,"否",M_NO;

	mes "[线商]";
	mes "哦，感谢神明！";
	mes "只要你能把那些虫子杀掉，我就能送货了。";
	next;
	mes "[线商]";
	mes "糟了！它们来了！！";
	emotion ET_PROFUSELY_SWEAT;
	donpcevent "BRDQ_MOBS::OnStart";
	close;

L_KILLED:
	set $@beardMobD,0;
	mes "[线商]";
	mes "谢谢你杀掉那些盗虫。";
	mes "为表谢意，这箱^8080FF线^000000送给你。";
	set BEARD_QUEST,4;
	close;

L_DONE:
	mes "[线商]";
	mes "再次感谢你！";
	emotion ET_THANKS;
	close;

M_NO:
	mes "[线商]";
	mes "该怎么办呢……";
	close;
}
