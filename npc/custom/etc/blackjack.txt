//= rAthena Project
//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	二十一点庄家	57,{
	mes "[庄家]";
	mes "你好！想玩一局 Black Jack 吗？";
M_Menu:
	next;
	menu "规则是什么？",M_0, "我要玩。",L_Play, "不用了。",M_End;

M_0:
	mes "[庄家]";
	mes "你想知道什么？";
sM_Menu0:
	next;
	menu "基本规则。",sM_0a, "胜负判定。",sM_0b, "'加倍'。",sM_0c,
	"A 的点数。",sM_0d, "Nothing.",M_Menu;

sM_0a:
	mes "[庄家]";
	mes "Black Jack 是一种纸牌游戏，目标是把点数凑到 21，或尽量";
	mes "接近 21 点，但不能超过 21。";
	next;
	mes "[庄家]";
	mes "游戏开始前，玩家必须先下注。下注之后，";
	mes "庄家和玩家各发 2 张牌，";
	mes "依手中的牌，你可以选择 ^5533FF'停牌'^000000，";
	mes "也可以选择 ^5533FF'要牌'^000000。";
	next;
	mes "[庄家]";
	mes "选择'停牌'，就是告诉庄家你不再";
	mes "需要牌了。这样庄家就可以决定是否要牌。";
	mes "要停牌，你必须有 ^FF3355至少 13 点^000000。庄家";
	mes "只有在^FF3355 17 点或以上^000000时才能停牌。";
	next;
	mes "[庄家]";
	mes "选择'要牌'，就是告诉庄家你想要";
	mes "再拿一张牌。多要牌可以提高你的";
	mes "总点数。只要你的点数 ^FF3355低于 21^000000 点，就可以";
	mes "从牌堆再要一张牌。";
	goto sM_Menu0;
sM_0b:
	mes "[庄家]";
	mes "Black Jack 有 3 种分出胜负的方式。";
	next;
	mes "[庄家]";
	mes "1.) 一局 Black Jack 结束时，如果你的点数";
	mes "比庄家多，你就赢下这一局。反之，";
	mes "点数比庄家少，你就输了。";
	next;
	mes "[庄家]";
	mes "2.) 如果要牌后总点数超过 21，";
	mes "你就自动输掉这一局。这称为 ^5533FF'爆牌'^000000。如果";
	mes "庄家爆牌，你就赢下这一局。";
	next;
	mes "[庄家]";
	mes "3.) 如果前 2 张牌的总点数是 21，";
	mes "你就自动赢下这一局。这称为 ^5533FF'Black Jack'^000000，";
	mes "发生在你拿到一张'A'和一张'10 点'牌时。如果";
	mes "庄家拿到 Black Jack，他/她就自动赢下这一局。";
	next;
	mes "[庄家]";
	mes "4.) 除输赢之外，你也可能与庄家打平。如果";
	mes "一局结束时你和庄家的总点数相同，";
	mes "这一局就是平局，没有赢家，也不会有";
	mes "金钱上的得失。";
	next;
	mes "[庄家]";
	mes "这称为与庄家 ^5533FF'平手'^000000。这同样";
	mes "适用于你和庄家同时拿到 Black Jack 的";
	mes "time.";
	goto sM_Menu0;
sM_0c:
	mes "[庄家]";
	mes "'加倍'选项让你把当前赌注翻倍，";
	mes "但代价是你只能再要";
	mes "一张牌。该选项只能在";
	mes "每局开始时使用。";
	next;
	mes "[庄家]";
	mes "加倍有用的一个例子是，";
	mes "当前 2 张牌的总点数是 11。下一张牌";
	mes "有很大机会让你达到 21 或 20 点。这手牌";
	mes "就很适合加倍。";
	goto sM_Menu0;
sM_0d:
	mes "[庄家]";
	mes "在 Black Jack 中，'A'是张特殊的牌，因为";
	mes "它有两种取值。A 可以算作 11 点，";
	mes "也可以只算 1 点。例如你有 A 和 4，";
	mes "点数就是 15 或 5。";
	next;
	mes "[庄家]";
	mes "如果你选择停牌，A 会自动算作";
	mes "11 点，总点数为 15。";
	next;
	mes "[庄家]";
	mes "如果你选择要牌并拿到 9，A 就会";
	mes "自动算作 1 点，总点数为 14";
	mes "点。如果 A 算作 11 点，你的";
	mes "总点数就会超过 21，你就输了。";
	next;
	mes "[庄家]";
	mes "正是'A'的灵活性，";
	mes "使它成为游戏中最强的牌。";
	goto sM_Menu0;
M_End:
	mes "[庄家]";
	mes "随时欢迎再来";
	close;

//================
L_Play:
	mes "[庄家]";
	mes "请下注……";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "太贵了，我可玩不起……",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[庄家]";
	mes "抱歉，你的 Zeny 不足以支付这个赌注。";
	close;

L_Cont0:
	mes "（正在发牌）";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- 以下是 ^FF5533庄家^000000 的牌：";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "庄家的点数：^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "- 以下是 ^5533FF你^000000 的牌：";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "你的点数：^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "你的点数：^5533FF" +@playTot+ "^000000，或 ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "要牌",M_Hit, "停牌",M_Stay, "加倍",M_Double;
	if(@dealerTurn == 0) menu "要牌",M_Hit, "停牌",M_Stay;

M_Stay:
	mes "[庄家]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "庄家停牌。";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "庄家要牌";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "抱歉，你的总点数不够高，不能停牌。你必须再要牌。";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[庄家]";
	mes "玩家选择了加倍。你当前的赌注将";
	mes "翻倍，并且你只能再多要 1 张牌。";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[庄家]";
	mes "恭喜，你赢了！";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[庄家]";
	mes "抱歉，你输了。";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[庄家]";
	mes "平手。你与庄家点数相同。";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "|"+@dealCard$[0]+"|   |"+@dealCard$[1]+"|";
	mes "'-----'   '-----'";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "|"+@playCard$[0]+"|   |"+@playCard$[1]+"|";
	mes "'-----'   '-----'";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "|"+@dealCard$[0]+"|   |"+@dealCard$[1]+"|   |"+@dealCard$[2]+"|";
	mes "'-----'   '-----'    '-----'";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "|"+@playCard$[0]+"|   |"+@playCard$[1]+"|   |"+@playCard$[2]+"|";
	mes "'-----'   '-----'    '-----'";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes "|"+@dealCard$[0]+"|  |"+@dealCard$[1]+"|  |"+@dealCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes "    .-----.  ";
	mes "|"+@dealCard$[3]+"|";
	mes "'-----'";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes "|"+@playCard$[0]+"|  |"+@playCard$[1]+"|  |"+@playCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes "    .-----.  ";
	mes "|"+@playCard$[3]+"|";
	mes "'-----'";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes "|"+@dealCard$[0]+"|  |"+@dealCard$[1]+"|  |"+@dealCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes "    .-----.   .-----. ";
	mes "|"+@dealCard$[3]+"|   |"+@dealCard$[4]+"|";
	mes "'-----'   '-----'";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes "|"+@playCard$[0]+"|  |"+@playCard$[1]+"|  |"+@playCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes "    .-----.   .-----. ";
	mes "|"+@playCard$[3]+"|   |"+@playCard$[4]+"|";
	mes "'-----'   '-----'";
	return;
}
