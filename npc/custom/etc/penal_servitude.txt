sec_pri,36,58,1	script	典狱长	105,{ 
	mes "[萨迪乌斯]";
	emotion ET_QUESTION;
	mes (Sex==SEX_MALE?"Mr.":"Ms.")+"囚犯 #"+BaseLevel+JobLevel+"，";
	mes "那边是什么声音？";
	next;
	menu "放我出去，我付钱！",-, "让我服劳役抵罪！",M_Q1, "没什么",M_NO_THANKS;

M_PAY:
	mes "[萨迪乌斯]";
	set .@MUSTPAY,(PRISON+1)*1000000;
	if (.@MUSTPAY<1000000) set .@MUSTPAY,1000000;
	if (.@MUSTPAY>100000000) set .@MUSTPAY,100000000;
	if (PRISON==0) mes "我对你没印象。第一次来吗？";
	if (PRISON>0) mes "又是你？有人第一次来之后还是没学乖……";
	mes "好吧，我们可以放你走，代价是 ^FF0000"+.@MUSTPAY+"z^000000.";
	next;
	menu "我付现金！",M_PAYCASH,"你们收支票吗？",M_PAYBANK,"我改主意了。",-;

	mes "[萨迪乌斯]";
	mes "你还有时间考虑……";
	close;

M_PAYCASH:
	if (.@MUSTPAY>Zeny) goto L_NOCASH;
	set Zeny,Zeny-.@MUSTPAY;
	mes "[萨迪乌斯]";
	mes "好吧，到各处去唱吧。";
	goto L_RELEASE;

L_NOCASH:
	mes "[萨迪乌斯]";
	mes "这是什么？根本不够！";
	close;

M_PAYBANK:
	if (.@MUSTPAY>#kafrabank) goto L_NOBANK;
	set #kafrabank,#kafrabank-.@MUSTPAY;
	mes "[萨迪乌斯]";
	mes "好，在支票上签字吧。再把你的名字写在我的本子上。";
	goto L_RELEASE;

L_NOBANK:
	mes "[萨迪乌斯]";
	if (#kafrabank==0) mes "你的案卷里没提到你的银行账户。";
	if (#kafrabank!=0) mes "唉，你的银行账户信息在这里…… ^FF0000"+#kafrabank+"z^000000……案卷里就这么多，不够。";
	mes "别再耍这种无聊的把戏了！";
	close;

L_RELEASE:
	set PRISON,PRISON+1;
	next;
	mes "[萨迪乌斯]";
	mes "你现在自由了！";
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

M_Q1:
	mes "[萨迪乌斯]";
	mes "我们的毒菇库存不够了。还记得你早上那碗稀粥吗？要不要到我们资助的蘑菇农场干点活？";
	next;
	menu "好，我干活抵债！",-, "我爱这座监狱！",M_NO_THANKS;

	mes "[萨迪乌斯]";
	if(checkcart() || checkfalcon() || checkriding()){
		emotion ET_SCRATCH;
		mes "抱歉，你得先解除手推车、啄木鸟或猎鹰！";
		close;
	}
	mes "去找我们的监工奥利弗谈谈。";
	next;

	nude;

	set .@rw,rand(1,4);
	if (.@rw==2) goto L_W2;
	if (.@rw==3) goto L_W3;
	if (.@rw==4) goto L_W4;

L_W1:
	savepoint "sec_in02",179,76;
	warp "sec_in02",179,76;
	close;

L_W2:
	savepoint "sec_in02",139,32;
	warp "sec_in02",139,32;
	close;

L_W3:
	savepoint "sec_in02",100,28;
	warp "sec_in02",100,28;
	close;

L_W4:
	savepoint "sec_in02",107,75;
	warp "sec_in02",107,75;
	close;

M_NO_THANKS:
	mes "[萨迪乌斯]";  
	if (rand(2)) mes "今天难道是圣诞节吗？";
	mes "现在给我闭嘴，退到一边去！";
	if (rand(2)) emotion ET_HUK;
	close;  
}

sec_in02,137,57,1	script	监工	708,{ 
	mes "[奥利弗]";

	delitem 4002,countitem(4002);//Items: Fabre_Card,
	delitem 4009,countitem(4009);//Items: Chonchon_Card,
	delitem 4022,countitem(4022);//Items: Spore_Card,
	delitem 4048,countitem(4048);//Items: Poison_Spore_Card,
	
	if (PRISON_Q <= 0 ) goto L_GET_Q;

	mes "我要你去采集 ^FF0000"+PRISON_Q+"橘色网菇^000000。";
	if ( countitem(1069)<PRISON_Q ) close;//Items: Orange_Net_Mushroom,
	mes "好……把所有的蘑菇都交给我……";
	mes "让我看看……"+countitem(1069)+"个勉强能吃的……";//Items: Orange_Net_Mushroom,
	mes "还有"+countitem(1070)+"个没用的毒菇。";//Items: Orange_Gooey_Mushroom_,
	next;
	mes "[奥利弗]";
	mes "谢谢。你自由了！";
	set PRISON_Q,0;
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

L_GET_Q:
	set PRISON,PRISON+1;
	if (PRISON > 1) mes "嗯……你已经来过了……真是的……";

	if ( countitem(1069) || countitem(1070) ) mes "把这些蘑菇丢掉！！它们有毒！";//Items: Orange_Net_Mushroom, Orange_Gooey_Mushroom_,
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,

	set PRISON_Q, PRISON*3 + BaseLevel/3;
	set PRISON_Q, PRISON_Q * (readparam(bAspd)/55 + 1);
	set PRISON_Q, PRISON_Q+rand(50,60);
	mes "好，你必须给我带来 ^FF0000"+PRISON_Q+"^000000 可食用的 ^FF0000橘色网菇^000000，毒菇……不，蘑菇。";
	if(readparam(bAspd)>=100) mes "你小子手脚挺快！那你得多采些蘑菇。";
	if (rand(10)<4) mes "那些毒菇嘛，你就留着当早饭吃吧。";
	close;
}
