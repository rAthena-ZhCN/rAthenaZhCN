//===== Description: ========================================= 
//= An account wide Banker to store Zeny
//===== Additional Comments: ================================= 
//= Syrus22 - There's an optional transaction fee at the top of
//= the script. To use it simply change the first set command
//= to set the cost variable to whatever you want the fee to be.
//= Version 2.0: Optimized and brought the script up to standard. [Jguy]
//============================================================ 

prontera,132,217,5	script	银行职员	109,{
	set @cost,500;
	mes "[银行职员]";
	mes "欢迎光临普隆德拉第一银行。今天有什么可以为您效劳？";
	next;
	switch(select("我要存款:我要取款:我想查询余额:取消")) {
	case 1:
		mes "[银行职员]";
		mes "好的……您想存入多少？单次最多可存入 999,999 Zeny。";
		next;
		if (@cost > 0) {
			mes "[银行职员]";
			mes "另外，请注意每笔交易都会收取" +@cost + "Zeny 的手续费！";
			next;
		}
		input @deposit;
		if (@deposit < 1) {
			mes "请告诉我一个实际可存入的金额。";
			close;
		}
		else if (@deposit > Zeny) {
			mes "您似乎没有要存入的那么多 Zeny！";
			close;
		}
		else if (@deposit > (Zeny - @cost)) {
			mes "您需要" + @cost + "Zeny 来支付手续费！";
			close;
		}
		else {
			set Zeny,Zeny - @deposit;
			set Zeny,Zeny - @cost;
			set #bankstorage,#bankstorage + @deposit;
			mes "[银行职员]";
			mes "非常感谢……您的 Zeny 会得到妥善保管。";
			close;
		}
	case 2:
		mes "[银行职员]";
		mes "好的……您想取出多少？单次最多可取 999,999 Zeny";
		next;
		if (@cost > 0) {
			mes "[银行职员]";
			mes "另外，请注意每笔交易都会收取" +@cost + "Zeny 的手续费！";
			next;
		}
		input @withdrawl;
		if (@withdrawl < 1) {
			mes "请不要开玩笑。我需要一个实际可取出的金额。";
			close;
		}
		else if (@withdrawl > #bankstorage) {
			mes "您的账户中只有 ^00FF00" + callfunc("F_InsertComma",#bankstorage) +"^000000 zeny in your account!";
			close;
		}
		else if ((@cost > Zeny) && ((Zeny + @withdrawl) > @cost)) {
			mes "[银行职员]";
			mes "您目前没有足够的 Zeny 支付手续费。要我从您的取款中直接扣除手续费吗？";
			next;
			switch(select("好的:不用了，谢谢")){
			case 1:
				mes "[银行职员]";
				mes "正在从取款中扣除" + @cost + "以支付存款手续费……";
				set @withdrawl,@withdrawl - @cost;
				set #bankstorage,#bankstorage - @cost;
				set @cost,0;
				next;
				set Zeny,Zeny - @cost;
				set Zeny,Zeny + @withdrawl;
				set #bankstorage,#bankstorage - @withdrawl;
				mes "[银行职员]";
				mes "这是您的 Zeny。祝您愉快。";
				close;
			case 2:
				mes "[银行职员]";
				mes "好的……欢迎下次光临。";
				close;
			}
		}
		else {
			set Zeny,Zeny - @cost;
			set Zeny,Zeny + @withdrawl;
			set #bankstorage,#bankstorage - @withdrawl;
			mes "[银行职员]";
			mes "这是您的 Zeny。祝您愉快。";
			close;
		}
	case 3:
		mes "[银行职员]";
		mes "嗯……让我查一下文件。";
		next;
		mes "*沙沙，沙沙*";
		next;
		mes "[银行职员]";
		mes "您目前账户中有" + callfunc("F_InsertComma",#bankstorage) + " Zeny in your account.";
		close;
	case 4:
		mes "[银行职员]";
		mes "好的……欢迎下次光临。";
		close;
	}
}
