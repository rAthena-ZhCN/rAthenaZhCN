function	script	duelkill	{
	if (strcharinfo(0) != $@duelist1$ && strcharinfo(0) != $@duelist2$) end;
	announce strcharinfo(0) + "赢得了决斗",1;
	set @fighting,0;
	set #heal,0;
	set @healing,0;
	end;
}

	announce "非法击杀：" + strcharinfo(0) + "已被检测到",1;
	percentheal -100,-100;
	end;
}
