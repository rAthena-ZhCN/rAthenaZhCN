		mes "不过我建议你学。";
		next;
		mes "[古德曼]";
		mes "如果你愿意，我可以";

		mes "怎么样，小子？";
		next;
		if (select("听起来不错！:以后再说吧~") == 1) {
			mes "[古德曼]";
			mes "嘿！好选择。那么，";
			mes "贪婪技能会自动";
			mes "拾取施法者周围 5*5";
			mes "格范围内的所有物品。";
			mes "从怪物尸体上捡成堆的货物时很方便。";
			next;
			mes "[古德曼]";
			mes "先说第一件事。我需要";
			mes "测试你是否有资格";

			mes "满到连一件重量 500";

			mes "任务后，回来";
			mes "找我谈。那么";

			BLACK_SK2 = 1;
			close;
		}
		mes "[古德曼]";
		mes "我明白了。忙于其他事情";
		mes "是好事，";
		mes "但记住时间也是";
		mes "一种商品，你应该";
		mes "总是把它花在最有";
		mes "价值的事情上。明白吗？";
		close;
	}
	if (BLACK_SK2 == 1) {
		if (MaxWeight - Weight >= 5000) {
			mes "[古德曼]";
			mes "嗯。你回来得";
			mes "太早了。而且你没有";
			mes "完成我给你";
			mes "定的目标。仔细听。";
			next;
			mes "[古德曼]";
			mes "去用任意物品";
			mes "把你的背包填满，直到";
			mes "满得无法放入";
			mes "一件重量 500 的物品。";
			mes "完成这项任务后，";
			mes "回来向我报告。";
			close;
		}
		mes "[古德曼]";
		mes "这么快就回来了？";
		mes "好，让我检查一下";
		mes "你身上的东西。";
		mes "我看看，背包，背包……";
		next;
		mes "[古德曼]";
		mes "干得好：你通过了。";

		mes "花在收集这些物品上的";
		mes "时间和经验，";
		mes "比它们的 Zeny 价格更有价值？";
		next;
		mes "[古德曼]";
		if (Upper == 1)
			mes "我相信，神工匠";
		else
			mes "我相信，铁匠";
		mes "首先应该是";
		mes "匠人。富有和名声";
		mes "只是随之而来的副产物。";
		if (Upper == 1)
			mes "可敬的神工匠知道，价值在于手艺……";
		else
			mes "可敬的铁匠知道，价值在于手艺……";
		next;
		if (Upper == 1) {
			mes "[古德曼]";
			mes "现在，如果你不太";
			mes "看重一件物品";
			mes "的 Zeny 价值，你就能";

			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[古德曼]";
	mes "你一直是怎么使用";
	mes "是贪婪技能吗？但愿它";
	mes "在你为所有伙伴";
	mes "锻造更优质物品的";

	mes "都带来了便利。";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
