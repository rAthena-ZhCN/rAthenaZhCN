		switch(select("嗯……不错。好吧。:它到底是怎么运作的？")) {

				mes "哈哈哈~！好好用它吧！";
				close;
			}

				mes "[铁匠]";

				mes "你没有我需要的物品。";
				next;
				mes "[铁匠]";

				close;
			}
		case 2:
			mes "[铁匠]";
			mes "这个 ^3355FF焊接面罩^000000 会用一块厚实的金属板保护你的脸。这项发明是三十年来在锤与铁砧前无尽辛劳的成果。";
			close;
		}
	case 2:
		mes "[铁匠]";
		mes "啊……！";
		mes "我猜这是你第一次见到像我这么伟大的铁匠，所以只是敬畏得说不出话来。请慢慢来，准备好了再开口。";
		next;
		mes "^3355FF这家伙似乎";
		mes "有点太";
		mes "自以为是了。^000000";
		close;
	}
}

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	妮娅#yagu	700,{
	mes "[妮娅]";
	mes "你好，我是妮娅，游戏大师";
	mes "马翁的第五名";
	mes "弟子。那么，";
	mes "要不要和我玩一局";
	mes "数字配对游戏？";
	next;
	switch(select("数字配对游戏？:练习数字配对:开始数字配对:记录保持者:取消")) {
	case 1:
		mes "[妮娅]";
		mes "啊，那个，数字配对游戏的";
		mes "目标就是";
		mes "猜出我心里想的";
		mes "数字。当然，有几点";
		mes "规则限制了我能";
		mes "选择的数字。";
		next;
		mes "[妮娅]";
		mes "我所组成的";
		mes "数字必须是三位数，我不能";
		mes "在任何一位上使用 0，";
		mes "也不能在各位上";
		mes "重复使用同一个";
		mes "数字。";
		next;
		mes "[妮娅]";
		mes "所以，举例来说，我不能";
		mes "使用 103、112、";
		mes "252 或 701 这些数字。它们";
		mes "要么含有 0，要么在各位上";
		mes "重复使用了同一个";
		mes "数字。现在明白了吗？";
		next;
		mes "[妮娅]";
		mes "接下来，你只有 5 次猜测机会，";
		mes "每次猜完，我都会告诉";
		mes "你猜对了几个数字，";
		mes "以及其中有多少个数字";
		mes "处在正确的位置上。";
		next;
		mes "[妮娅]";
		mes "举个例子，假设";
		mes "我想出的数字是 168。";
		mes "如果你的第一次猜测是 678，那么你猜对了两个数字，";
		mes "6 和 8，猜对了。不过，";
		mes "只有 8 在正确的位置上。";
		next;
		mes "[妮娅]";
		mes "我说，你最好";
		mes "先试试练习模式。";
		mes "然后，等你更";
		mes "熟练之后，再玩真正的";
		mes "数字配对游戏。别忘了速度也会计入你的成绩！";
		next;
		mes "[妮娅]";
		mes "啊，如果你";
		mes "在数字配对游戏中";
		mes "刷新了新纪录，";
		mes "你的名字就会被记录下来，";
		mes "所以别错过这个";
		mes "扬名甚至发财的机会~";
		close;
	case 2:
		mes "[妮娅]";
		mes "好，我们来玩一局练习";
		mes "数字配对游戏。请";
		mes "试着猜猜我的数字，好吗？";
		mes "如果想取消，请输入大于 1,000 的数字。";
		set .@practice,1;
		break;
	case 3:
		mes "[妮娅]";
		mes "好，我们来玩数字配对吧！";
		mes "你有 5 次机会猜出我心里想的数字。";
		mes "如果想取消，请输入大于 1,000 的数字。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[妮娅]";
		mes "^FF0000" + .@name_record$ + "^000000 是";
		mes "数字配对游戏的";
		mes "纪录保持者，成绩为";
		mes "^FF0000" + .@score_min + "分" + .@score_sec + "秒^000000。";
		close;
	case 5:
		mes "[妮娅]";
		mes "请多保重，";
		mes "冒险者。如果你";
		mes "想玩数字";
		mes "配对游戏，随时来";
		mes "找我，好吗？再见~";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[妮娅]";
				mes "你已经取消了";
				mes "数字配对游戏。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[妮娅]";
			mes "你输入的其中一个";
			mes "数字是 0，或者在各位上";
			mes "重复输入了";
			mes "相同的数值。";
			mes "请输入另一个猜测。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[妮娅]";
		mes "^FF0000"+strcharinfo(0)+"^000000，";
		mes "你的猜测是 ^0000ff"+.@input+"^000000.";
		mes "请稍等，";
		mes "让我算出你的结果。";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[妮娅]";
				mes "是的，我";
				mes "所想的数字是 ^ff0000"+.@yagu+"^000000！";
				mes "恭喜你，你";
				mes "刚刚赢得了这局练习";
				mes "游戏！现在你应该";
				mes "可以挑战真正的游戏了~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[妮娅]";
					mes "太好了！你猜中了";
					mes "我的数字，它是";
					mes "^FF0000" + .@yagu + "^000000。你用了";
					mes "^FF0000" + .@total_min + "分" + .@total_sec + "秒^000000，";
					mes "打破了先前的纪录";
					mes "，即 ^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000。";
					next;
					mes "[妮娅]";
					mes "^FF0000" + .@name_record$ + "^000000";
					mes "创下了那个旧纪录，";
					mes "但现在你才是";
					mes "数字配对游戏中";
					mes "新的挑战目标。恭喜";
					mes "你创下新纪录！";
					close;
				}
				mes "[妮娅]";
				mes "干得好，你猜中了";
				mes "我的数字！答案当然";
				mes "是 ^ff0000" + .@yagu + "^000000。你成功";
				mes "猜对了这个数字，";
				mes "用时 ^ff0000" + .@total_min + "分" + .@total_sec + "秒^000000。";
				next;
				mes "[妮娅]";
				mes "^FF0000" + .@name_record$ + "^000000 是";
				mes "目前数字配对";
				mes "游戏的纪录保持者，成绩";
				mes "为 ^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000。";
				mes "祝你好运，希望你有朝一日";
				mes "能打破这个纪录~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[妮娅]";
				mes "你的猜测中有 ^ff0000" + .@ball + "^000000 个";
				mes "数字正确。^ff0000" + .@strike + "^000000 个";
				mes "数字在你的猜测中";
				mes "处在正确的";
				mes "位置上。";
				mes "那么，再试一次吧~";
				next;
			} else {
				mes "[妮娅]";
				mes "你上一次的猜测有 ^ff0000" + .@ball + "^000000";
				mes "个数字正确，";
				mes "并且有 ^ff0000" + .@strike + "^000000 个数字";
				mes "处在正确的";
				mes "位置上。";
				switch(.@retry) {
					case 1: mes "你还有四次机会。"; break;
					case 2: mes "你还有三次机会。"; break;
					case 3: mes "你还有两次机会。"; break;
					case 4: mes "你只剩一次机会了……"; break;
					case 5: mes "答案是 ^ff0000" + .@yagu + "^000000。"; close;
				}
				next;
			}
		}
	}
}
