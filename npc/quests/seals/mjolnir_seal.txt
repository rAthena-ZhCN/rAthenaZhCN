							switch(select("伊米尔的头:天空:云之工场:高屋")) {
							default:
								set .@talk_not,1;
								break;

							}

						else if (.@talk_to == 2) {
							mes "[诺德里]";

							mes "[诺德里]";
							mes "于是托尔又问了下一个问题。众神把夜里升上天空的那颗球称作什么？";

							switch(select("循环之轮:月亮:假太阳:疾驰的陌生人")) {

								break;
							case 3:
								break;

						}
						else if (.@talk_to == 3) {
							mes "[诺德里]";
							mes "对！众神把";

							mes "阿尔维斯虽然答出了托尔的所有问题，却没注意到太阳正在升起。";

							mes "[诺德里]";

							mes "托尔故意拖长考问，好让阿尔维斯";

							next;
							mes "[诺德里]";
							mes "从这个故事里";

						}
						else {
							set .@talk_to,.@talk_to + 1;

					}
				}
				else {
					mes "[诺德里]";
					mes "Oooh...";
					mes "我渴死啦！";
					next;
					mes "[诺德里]";
					mes "嗯，我不是让你带红色药水来吗？嗓子这么干，我哪讲得了故事～";

			}
			else if (god_mjo_1 == 0) {
				mes "[诺德里]";
				mes "你找我";
				mes "有什么事吗，";
				mes "人类？";
				next;
				switch(select("没什么。:打扰了。")) {
				case 1:
					mes "[诺德里]";
					mes "真奇怪，别来烦我。";
					close;
				case 2:
					mes "[诺德里]";
					mes "奇怪。最近有太多人类想与我和我的兄弟们见面。不过，我也不能说他们的来访令人不快。";
					next;
					set god_mjo_1,1;
					mes "[诺德里]";
					mes "我有点渴了。你能给我带一瓶红色药水吗？如果你肯帮我，我就告诉你一个重要的故事。呵呵呵~";

			}

				mes "呼……呼……呼……";

		}
		else if (god_mjo_0 == 0) {

			mes "只是个玩笑。";
			mes "哈哈哈哈！";

		}
		else {
			mes "[诺德里]";
