prontera,113,294,3	script	路标#god	837,{
	mes "======== 神之封印状态 ========";
	setarray .@seal$[1], "Sleipnir","Megingjard","Brisingamen","Mjolnir";
	for(set .@i,1; .@i<=4; set .@i,.@i+1) {
		set .@val, getd("$God"+.@i);
		if (.@val == 0)
			set .@status$,"Unseen";
		else if (.@val < $@god_check1)
			set .@status$,"Active";
		else if (.@val < $@god_check2)
			set .@status$,"Appeared";
		else
			set .@status$,"Released";
		mes .@seal$[.@i]+"封印："+.@status$;
	}
	mes " ";
	mes "======= 你的封印状态 ========";
	if (god_sl_1 < 51)
		mes "^ff0000斯雷普尼尔封印：未完成^000000";
	else
		mes "^00ff00斯雷普尼尔封印：已完成^000000";
	if (god_eremes < 28)
		mes "^ff0000梅金吉奥德封印：未完成^000000";
	else
		mes "^00ff00梅金吉奥德封印：已完成^000000";
	if (god_brising < 50)
		mes "^ff0000布利辛加门封印：未完成^000000";
	else
		mes "^00ff00布利辛加门封印：已完成^000000";
	if (god_mjo_0 < 11)
		mes "^ff0000妙尔尼尔封印：未完成^000000";
	else
		mes "^00ff00妙尔尼尔封印：已完成^000000";
	close;
}
