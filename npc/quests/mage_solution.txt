pay_arche,122,100,0	script	多尔修伊	88,{
	mes "[魔法师公会成员]";
	mes "你想要溶液？嗯，给我 50 Zeny 和至少 1 个空试管。";
	next;
	if (select("好的，成交。:算了，不用了。") == 1) {
		mes "[魔法师公会成员]";
		if (Zeny < 50) {
			mes "喂！你的钱不够付我这 50 Zeny 的费用。";
			close;
		}
		if (countitem(1092) == 0) {
			mes "没有瓶子就装不了溶液！给我拿一个空试管来。";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
