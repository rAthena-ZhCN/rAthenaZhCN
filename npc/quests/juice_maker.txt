payon_in03,188,146,5	script	马克斯·汉森#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- 请稍等！-";
		mes "- 您目前携带的 -";
		mes "- 物品太多了。 -";
		mes "- 请稍后再来 -";
		mes "- 在将一些物品放入卡普拉仓库之后。-";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[商人马克斯·汉森]";
		mes "Welcome.";
		mes "你是来";
		mes "加工水果的吗？";
		next;
		switch(select("制作果汁。:谈谈水果加工的情报。:取消。")) {
		case 1:
			mes "[商人马克斯·汉森]";
			mes "您想制作什么果汁？";
			next;
			switch(select("苹果汁:香蕉汁:红萝卜汁:葡萄汁:取消")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[商人马克斯·汉森]";
				mes "那么……";
				mes "下次见。";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[商人马克斯·汉森]";
				mes "哎呀……";
				mes "您的材料不齐全。要制作"+getitemname(.@juice)+"，需要1个"+getitemname(.@fruit)+"和1个空瓶。另外还需要3 Zeny的手续费。";
				next;
				mes "[商人马克斯·汉森]";
				mes "等您准备好一切，我就把水果打碎，为您做出美味的果汁。";
				close;
			}

			mes "[商人马克斯·汉森]";
			mes "您要几瓶？";
			next;
			switch(select("能做多少就做多少。:我要指定数量。:取消。")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[商人马克斯·汉森]";
				mes "请输入100以下的数字。如果不想做，请输入'0'。最多可以制作" + countitem(.@fruit) + "瓶果汁。";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[商人马克斯·汉森]";
						mes "那么……";
						mes "欢迎再来。";
						close;
					}
					else if (.@input > 100) {
						mes "[商人马克斯·汉森]";
						mes "超过100瓶是不可能的。请选择其他数量。";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[商人马克斯·汉森]";
				mes "那么……";
				mes "欢迎再来。";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[商人马克斯·汉森]";
				mes "哎呀……";
				mes "您的材料不齐全。这种情况我也无能为力。请您先把需要的东西收集齐吧。";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[商人马克斯·汉森]";
			mes "给您！这是约好的新鲜美味果汁。应该十分清爽可口。";
			next;
			mes "[商人马克斯·汉森]";
			mes "那么……";
			mes "欢迎再来。";
			close;
		case 2:
			mes "[商人马克斯·汉森]";
			mes "在人类能够建立如此广阔的社会之前，他们靠采集树上的果实生存。果实或许是大自然赐予我们、让我们得以存续于世的恩惠。";
			next;
			mes "[商人马克斯·汉森]";
			mes "随着生活变得富足，年轻一代似乎不再吃水果了。于是我开始琢磨，怎样才能让水果更容易入口。";
			next;
			mes "[商人马克斯·汉森]";
			mes "我发现把水果做成汁，吃起来更方便，味道也好得多。";
			next;
			mes "# Fruit Juice Information #";
			mes "^CC4E5C- 苹果汁 -^000000";
			mes "苹果 x 1个，空瓶 x 1个，3 Zeny。";
			mes "^E3CF57- 香蕉汁 -^000000";
			mes "香蕉 x 1个，空瓶 x 1个，3 Zeny。";
			mes "^ED9121- 红萝卜汁 -^000000";
			mes "红萝卜 x 1个，空瓶 x 1个，3 Zeny。";
			mes "^CC00FF- 葡萄汁 -^000000";
			mes "葡萄 x 1个，空瓶 x 1个，3 Zeny。";
			close;
		case 3:
			mes "[商人马克斯·汉森]";
			mes "喂！";
			mes "如果你去拜访";
			mes "某个人，就和他说话吧！";
			close;
		}
	}
	else {
		mes "[商人马克斯·汉森]";
		mes "Welcome.";
		mes "有什么可以为您效劳的？";
		next;
		if (select("交谈:取消") == 1) {
			mes "[商人马克斯·汉森]";
			mes "在人类能够建立如此广阔的社会之前，他们靠采集树上的果实生存。果实或许是大自然赐予我们、让我们得以存续于世的恩惠。";
			next;
			mes "[商人马克斯·汉森]";
			mes "随着生活变得富足，年轻一代似乎不再吃水果了。于是我开始琢磨，怎样才能让水果更容易入口。";
			next;
			mes "[商人马克斯·汉森]";
			mes "我发现把水果做成汁，吃起来更方便，味道也好得多。";
			close;
		}
		mes "[商人马克斯·汉森]";
		mes "喂！";
		mes "如果你去拜访";
		mes "某个人，就和他说话吧！";
		close;
	}
}
