alberta,26,229,0	script	卡普拉职员#bunny	83,{
	if (BUNYBND == 1) {
		mes "[卡普拉职员]";
		mes "你好呀~！";
		mes "你要不要";
		mes "参加卡普拉";
		mes "公司的特别";
		mes "^529DFF兔耳发带活动^000000？";
		next;
		switch(select("好的，我带来了物品。:活动介绍:取消")) {
		case 1:
			mes "[卡普拉职员]";
			mes "好的~";
			mes "让我看看";
			mes "你是否带来了";
			mes "所有物品……";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉职员]";
					mes "哎呀，抱歉";
					mes "不过你需要";
					mes "至少带来";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+".";
					close;
				}
			mes "[卡普拉职员]";
			mes "太好了，看来";
			mes "你已经凑齐了";
			mes "制作兔耳发带所需的";
			mes "全部材料。";
			mes "请稍等片刻";
			mes "我这就把它组装起来……";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉职员]";
					mes "嗯？抱歉，";
					mes "其实我现在";
					mes "还做不出来。你";
					mes "还需要"+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "才能让我把这条";
					mes "兔耳发带组装起来……";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[卡普拉职员]";
			mes "啊，给你~";
			mes "完美的兔耳发带！";
			mes "那么，希望你喜欢。";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[卡普拉职员]";
			mes "感谢您";
			mes "参加这次";
			mes "特别活动，以及你";
			mes "一直以来";
			mes "对卡普拉服务的支持~";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[卡普拉职员]";
	mes "你好~！你想不想";
	mes "参加我们特别的^529DFF兔耳";
	mes "发带活动^000000，这场活动由卡普拉";
	mes "公司主办，并由";
	mes "阿尔贝塔商人公会赞助？";
	next;
	switch(select("参加活动:活动介绍:取消")) {
	case 1:
		set BUNYBND,1;
		mes "[卡普拉职员]";
		mes "太好了！感谢你";
		mes "参与！如果你";
		mes "还没听说，";
		mes "你需要收集这些";
		mes "物品，我才能为你";
		mes "组装一条兔耳发带……";
		next;
		mes "[卡普拉职员]";
		mes "100个柔毛，";
		mes "1个四叶草，";
		mes "1个猫耳发带，以及";
		mes "1颗珍珠。就这些！";
		mes "祝你好运，我会";
		mes "在这里等你~";
		close;
	case 2:
		break;
	case 3:
		close;
	}
L_Info:
	mes "[卡普拉职员]";
	mes "为感谢我们尊贵的";
	mes "顾客，卡普拉公司";
	mes "准备了一场特别活动，";
	mes "由卡普拉职员为";
	mes "带来所需物品的冒险者";
	mes "组装兔耳发带。";
	next;
	mes "[卡普拉职员]";
	mes "为了这件特别的、";
	mes "独一无二的物品，";
	mes "请带来100个柔毛，";
	mes "1个四叶草，";
	mes "1个猫耳发带，以及";
	mes "1颗珍珠。";
	next;
	mes "[卡普拉职员]";
	mes "等你准备好后，";
	mes "回来把这些";
	mes "物品交给我。";
	mes "（由";
	mes "阿尔贝塔商人公会赞助）。";
	close;
}
