	mes "[商人]";
	mes "这个神秘的男人是谁？";
	mes "我，就是那位神秘又风度翩翩的“商人”？";
	next;
	mes "[商人]";
	mes "带着他那些华丽的帽子，游走于米德加尔特大陆？没人知道……";
	next;
	mes "[商人]";
	mes "我收集并交易来自世界各地的这些帽子，究竟是为了什么？选一顶帽子，就能离揭开谜团更近一步……";
	next;
	mes "(1) ^3355FF医生头带^000000:";
	mes "1 红色头巾 + 50 铁 + 1 裂纹钻石 + 3500 Zeny";
	mes "(2)^3355FF羽毛帽^000000:";
	mes "1 浪漫绅士 + 300 鸟的羽毛 + 500 Zeny";
	mes "(3) ^3355FF歌剧魅影^000000:";
	mes "20 铁 + 1 歌唱植物 + 5000 Zeny";
	mes "(4) ^3355FF斗笠^000000:";

	next;
	switch(select("医生头带 : 羽毛帽 : 歌剧魅影 : 斗笠")) {
	case 1:
		if ((countitem(2275) > 0) && (countitem(998) > 49) && (countitem(733) > 0) && (Zeny > 3499)) {
			delitem 2275,1; //Red_Bandana
			delitem 998,50; //Iron
			delitem 733,1; //Crystal_Jewel___
			set Zeny, Zeny-3500;
			mes "[商人]";
			mes "哼！你没有行医执照吧？没关系，我听说过一个无照行医的游医，创造过医学奇迹！不过……那可能只是漫画里的情节。";
			next;
			mes "[商人]";
			mes "唉，随便吧。别被抓到就行。";
			getitem 2273,1; //Doctor_Cap
			close;
		}
		else {
			mes "[商人]";
			mes "你这家伙～";
			mes "再去看清楚我的要求。";
			mes "可你看起来不像个傻瓜。所以，老兄，现实点吧。";
			close;
		}
	case 2:
		if ((countitem(2247) > 0) && (countitem(916) > 299) && (Zeny > 499)) {
			delitem 2247,1; //Oldman's_Romance
			delitem 916,300; //Feather_Of_Birds
			set Zeny, Zeny-500;
			mes "[商人]";
			mes "哦～！你很有眼光。我知道收集这些东西不容易，但这顶帽子值得。拿去吧。现在你只差一件毛皮大衣和一根手杖了！";
			getitem 5018,1; //Feather_Bonnet
			close;
		}
		else {
			mes "[商人]";
			mes "你这家伙～";
			mes "再去看看我的要求。可你看起来不像个傻瓜。老兄，现实点吧。";
			close;
		}
	case 3:
		if ((countitem(998) > 19) && (countitem(707) > 0) && (Zeny > 4999)){ //Iron Singing_Plant 
			delitem 998,20; //Iron
			delitem 707,1; //Singing_Plant
			set Zeny, Zeny-5000;
			mes "[商人]";
			mes "这一顶？有点吓人，却带着某种浪漫气息。你觉得呢？喜欢吗？好，拿去吧，归你了！";
			getitem 2281,1; //Phantom_Of_Opera
			close;
		}
		else {
			mes "[商人]";
			mes "小丑。再去看看要求。^3355FF歌剧魅影^000000可不是轻易能得到的。所以，老兄，现实点吧。";
			close;
		}
	case 4:
		if ((countitem(1019) > 119) && (Zeny > 9999)){ //Wooden_Block 
			delitem 1019,120; //Wooden_Block
			set Zeny, Zeny-10000;
			mes "[商人]";
			mes "如果你有机会去斐扬的高地村，请去见见那位斗笠工匠。";
			next;
			mes "[商人]";
			mes "除了我，他从不把斗笠卖给别的商人，因为只有我懂得鉴赏它的品质。由于稀有，斗笠成了极其独特而卓越的商品。好！拿去吧，归你了！";
			getitem 2280,1; //Sahkkat
			close;
		}
		else {
			mes "[商人]";
			mes "笨蛋～再去看看我的要求。老兄，你看起来不像个傻瓜，现实点吧～";
			close;
		}
	}
}
