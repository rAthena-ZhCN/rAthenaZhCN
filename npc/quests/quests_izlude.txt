// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[埃德加]";
		mes "你又要去阿尔贝塔了吗？和上次一样，我给你同样的折扣，只收你250 Zeny。";
		mes "怎么样？";
		next;
		if (select("好的~！:你为什么对我这么好？！") == 1) {
			if (Zeny < 250) {
				mes "[埃德加]";
				mes "Um...";
				mes "这些钱";
				mes "不够啊。";
				mes "你再去";
				mes "多赚点钱吧？";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[埃德加]";
		mes "我天生就是这样。再说，你那调皮的笑容让我想起我那头金发的儿子，他多年前离家，去自己的船上当了水手。愿老天保佑他，无论他身在何方。";
		close;
	}
	mes "[埃德加]";
	mes "我的城镇伊斯鲁得和阿尔贝塔由西边的港口相连。我们两地往来太频繁了，我几乎都要变成阿尔贝塔人了。呵呵呵~";
	next;
	mes "[埃德加]";
	mes "我认识一个挺熟的人，叫菲力克斯，就住在阿尔贝塔。那家伙特别小气……干什么都要收钱！";
	next;
	mes "[埃德加]";
	mes "不过他其实是个好人，喜欢帮助别人。他心地善良，只要你出得起价，他就会帮你。";
	next;
	mes "[埃德加]";
	mes "最近阿尔贝塔的人都说，他其实是在真心帮大家，要杰勒比结晶只是个幌子。总之，你该去看看他能提供什么。";
	next;
	if (select("能告诉我怎么去阿尔贝塔吗？:结束对话") == 1) {
		mes "[埃德加]";
		mes "嗯？你可以用自己的两条腿走过去啊。不过你要是有钱，我建议你坐船。";
		next;
		if (select("好，知道了。:可我走腻了，而且身无分文！") == 1) {
			mes "[埃德加]";
			mes "好，路上小心~";
			close;
		}
		mes "[埃德加]";
		mes "Okay...";
		mes "你既不想走路，又没带钱，却还想去那儿？唉，真是……";
		next;
		mes "[埃德加]";
		mes "好吧好吧。我好歹是船长，可以便宜点把你送过去。250 Zeny怎么样？";
		next;
		if (select("好的~！:呸，这也太坑了！！") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[埃德加]";
				mes "Um...";
				mes "这些钱";
				mes "不够啊。";
				mes "再去";
				mes "多弄点。";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[埃德加]";
		mes "哎呀呀，";
		mes "你要是觉得";
		mes "这还算坑的话……";
		close;
	}
	mes "[埃德加]";
	mes "行吧。";
	mes "回头见~";
	close;
}
