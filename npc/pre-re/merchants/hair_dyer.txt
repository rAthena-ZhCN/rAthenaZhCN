	mes "[美发师 乔沃维奇]";
	mes "欢迎~！";
	mes "有什么可以为您效劳的？";
	next;
	mes "[美发师 乔沃维奇]";
	if (Sex == SEX_MALE) mes "哎呀，不好！您的头发受损了。看来需要专业护理。请过来坐这边。来。";
	else {
		if (rand(20) > 11) {
			mes "咦！？天啊！不行不行不行！您的头发受损太严重了！再这样下去可不行。";
			next;
			mes "[美发师 乔沃维奇]";
			mes "让我来护理您的头发吧？拜托了？";
		}
		else mes "哇！您的头发染过之后一定会很完美～要不要换个发色试试？";
	}
	next;
	while(1) {
		switch(select("染发:提示与说明:取消")) {
		case 1:
			mes "[美发师 乔沃维奇]";
			mes "好的好的，不错的选择～";
			mes "那么，这是您改头换面的机会，是您绽放美丽的机会！请选择您喜欢的颜色。";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[美发师 乔沃维奇]";
					mes "您觉得如何？想换个颜色吗？";
					next;
					if (select("是:否") == 1) {
						mes "[美发师 乔沃维奇]";
						mes "好的！请选择您喜欢的颜色。";
						next;
					}
					else  {
						mes "[美发师 乔沃维奇]";
						mes "嗯，我有点失望。我本想做得更好。不过我保证下次会做得更好。欢迎再来～";
						close;
					}
				}

				while(1) {
					switch(select("红色染料:黄色染料:紫色染料:赤色染料:绿色染料:蓝色染料:白色染料:黑色染料:还是保持现在的发色。")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[美发师 乔沃维奇]";
							mes "您一定很喜欢这个发色吧～";
							close;
						}
						mes "[美发师 乔沃维奇]";
						mes "诶～？您不染头发吗？我有点难过……";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[美发师 乔沃维奇]";
						mes "咦？可这就是您现在的发色啊。请选择不同的颜色。";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"yellow"; break;
						case 2: callsub S_NoDye,978,"violet"; break;
						case 3: callsub S_NoDye,980,"orange"; break;
						case 4: callsub S_NoDye,979,"green"; break;
						case 5: callsub S_NoDye,981,"blue"; break;
						case 6: callsub S_NoDye,982,"white"; break;
						case 7: callsub S_NoDye,983,"black"; break;
						case 8: callsub S_NoDye,975,"red"; break;
						}
						if (Zeny < 1000) {
							mes "[美发师 乔沃维奇]";
							mes "费用是 1000 Zeny。您的钱不够吗……？";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[美发师 乔沃维奇]";
			mes "心情低落的时候、被甩的时候、想给某人留下好印象的时候，甚至只是想引人注目的时候……";
			next;
			mes "[美发师 乔沃维奇]";
			mes "为了那个特别的地点和时刻，您不想拥有只属于自己的发型吗？只要您有合适的染料，我就能让您的头发变得漂亮动人。";
			next;
			mes "[美发师 乔沃维奇]";
			mes "也别太在意费用。获得美丽就和获得一切一样。呵呵～";
			next;
			mes "[美发师 乔沃维奇]";
			mes "* 费用与说明 *";
			mes "- 1 个您想要发色的染料。";
			mes "- 1000 Zeny 费用。";
			close;

		case 3:
			mes "[美发师 乔沃维奇]";
			mes "男人也好，女人也好……";
			mes "每个人都有变美的权利和义务。";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[美发师 乔沃维奇]";
		mes "咦？！可是要染发，您需要物品'"+getitemname(getarg(0))+"' 才能把头发染成"+getarg(1)+"...";
		close;
	}
	return;
