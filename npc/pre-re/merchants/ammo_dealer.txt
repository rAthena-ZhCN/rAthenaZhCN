alberta,118,160,3	script	子弹商人 托尼#alb::bdt	4_M_04,{
	mes "[托尼]";
	if (BaseClass == Job_Gunslinger) {
		mes "我是托尼，子弹商人。";
		mes "弹药不够的时候";
		mes "就来找我。只要带上";
		mes "材料，我就会为您";
		mes "制作所需的子弹。";
		next;
		switch(select("毒物子弹:火焰子弹:雷击子弹:暗盲子弹:冰冻子弹:取消")) {
		case 1: callsub S_BulletTrade,937,10,13205;// <item required>, <amount req>, <bullet id>
		case 2: callsub S_BulletTrade,7097,2,13203;
		case 3: callsub S_BulletTrade,7053,3,13204;
		case 4: callsub S_BulletTrade,1024,5,13206;
		case 5: callsub S_BulletTrade,7054,2,13207;
		case 6:
			mes "[托尼]";
			mes "改变主意了？";
			mes "那么，如果您需要";
			mes "子弹，我一直";
			mes "在这儿。需要更多";
			mes "弹药的时候，";
			mes "随时回来，神枪手。";
			close;
		}
	}
	mes "嗨，我是托尼。我负责";
	mes "为神枪手";
	mes "分发和制作";
	mes "子弹。这就是";
	mes "我们公会";
	mes "的行事方式。";
	next;
	mes "[托尼]";
	mes "抱歉，如果您是来";
	mes "买子弹的话。";
	mes "我只和正式";
	mes "的神枪手做生意。";
	close;

S_BulletTrade:
	.@required_id = getarg(0);
	.@required_amount = getarg(1);
	.@bullet_id = getarg(2);

	mes "[托尼]";
	mes "每制作一组";
	mes callfunc("F_InsertPlural",30,getitemname(.@bullet_id))+",";
	mes "您需要给我";
	mes "1 个强化武器金属 - 级数二，";
	mes "1 个强化武器金属-级数一，以及";
	mes callfunc("F_InsertPlural",.@required_amount,getitemname(.@required_id))+".";
	next;
	mes "[托尼]";
	mes "请记住，我一次最多";
	mes "只能给您 500 组";
	mes "30 发子弹。请";
	mes "输入您想要的";
	mes "子弹组数。如果想";
	mes "取消，请输入 ''0.''";
	next;
	input .@amount;
	mes "[托尼]";
	if (.@amount < 1 || .@amount > 500) {
		mes "喂，我不能给您";
		mes "那么多子弹。别忘了";
		mes "输入一个";
		mes "不超过 500 的数字";
		mes "这样才能用您的";
		mes "物品换取子弹。";
	}
	else if (countitem(1010) >= .@amount && countitem(1011) >= .@amount && countitem(.@required_id) >= (.@amount*.@required_amount)) {
		if (checkweight(.@bullet_id,.@amount * 30) == 0) {
			mes "咦？您的物品栏";
			mes "没有足够的空间";
			mes "装这么多子弹。请稍后再来";
			mes "等您腾出";
			mes "更多空间。可以试着把";
			mes "一些东西放进卡普拉仓库。";
		} else {
			mes "很好，一切";
			mes "都没问题。我收下";
			mes "这些材料，这是您的";
			mes "子弹。很高兴";
			mes "与您做生意～";
			delitem 1010,.@amount; //Phracon
			delitem 1011,.@amount; //Emveretarcon
			delitem .@required_id,.@amount * .@required_amount;
			getitem .@bullet_id,.@amount * 30;
		}
	} else {
		mes "嗯。看来您";
		mes "没有足够的材料";
		mes "制作那么多子弹。不过，";
		mes "没关系。等您";
		mes "集齐所需的";
		mes "材料再来，好吗？";
	}
	close;
}
