lou_fild01,240,354,4	script	螳螂研究者	878,{

	if (checkquest(60179) != -1) {
		if (checkquest(60179,HUNTING) != 2 ) {
			mes "[螳螂研究者]";
			mes "你已经猎杀完 50 只螳螂了吗？";
			next;
			switch(select("是:否:我想放弃")) {
			case 1:
				mes "[螳螂研究者]";
				mes "嗯，你好像还没猎杀满 50 只。";
				close;
			case 2:
				mes "[螳螂研究者]";
				mes "记住，我需要帮手猎杀螳螂，去猎杀 50 只吧。";
				close;
			case 3:
				mes "[螳螂研究者]";
				mes "你确定要停止狩猎吗？";
				mes "你目前取得的进度将被清除";
				next;
				switch(select("是:否")) {
				case 1:
					mes "[螳螂研究者]";
					mes "那好吧，如果你改变主意就回来这里。";
					erasequest 60179;
					close;
				case 2:
					mes "[螳螂研究者]";
					mes "请猎杀 50 只螳螂，让这里成为人们可以来获取经验的地方。";
					close;
				}
			}
		}
		mes "[螳螂研究者]";
		mes "太厉害了，你动作真快。";
		getexp 18818,7527;
		erasequest 60179;
		close;
	}
	if ((BaseLevel > 34) && (BaseLevel < 71)) {
		mes "[螳螂研究者]";
		mes "太好了！";
		mes "你是来帮我的吧？";
		mes "如果不是，我真不知道该怎么对付这里的怪物！";
		next;
		switch(select("好:不")) {
		case 1:
			mes "[螳螂研究者]";
			mes "太好了！我一看到你就知道可以信赖你！";
			mes "我需要你减少目前在这张地图上肆虐的怪物数量！";
			mes "这附近的螳螂太多了，低等级玩家根本没法在这里获得好的经验。";
			next;
			mes "[螳螂研究者]";
			mes "那么，你还愿意帮我吗？";
			next;
			switch(select("好:取消")) {
			case 1:
				if (BaseLevel > 70) {
					mes "[螳螂研究者]";
					mes "非常感谢你帮忙对付螳螂！";
					mes "目前情况已经稳定，我不再需要你帮忙猎杀它们了。";
					close;
				}
				mes "[螳螂研究者]";
				mes "猎杀 50 只螳螂，完成后告诉我。我会为你的付出给予报酬。";
				setquest 60179;
				close;
			case 2:
				mes "[螳螂研究者]";
				mes "如果你改变主意，请再回来。";
				close;
			}
		case 2:
			mes "[螳螂研究者]";
			mes "如果你改变主意，请再回来。";
			close;
		}
	}
	mes "[螳螂研究者]";
	mes "哎呀，哎呀！";
	next;
	mes "[螳螂研究者]";
	mes "抱歉，我现在没法跟你说话，我这里痛得要命，这些该死的螳螂要杀了我。";
	close;
}
