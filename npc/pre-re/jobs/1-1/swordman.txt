function	script	F_JobSwdStaff	{
	mes "[转职所职员]";
	mes "你要放弃吗？";
	next;
	if (select("是。:否。") == 1) {
		mapannounce "job_sword1","考生" + strcharinfo(0) + "退出了考试。",bc_map;
		warp "izlude_in",65,165;
		end;
	}
	mes "[转职所职员]";
	mes "好极了！再来一次！";
	close;
}

function	script	F_JobSwdTestStaff	{
OnTouch:
	mes "[转职所职员]";

	if (select("是。:否。") == 1) {
		mapannounce "job_sword1","考生" + strcharinfo(0) + "退出了考试。",bc_map;
		warp "izlude_in",65,165;
		end;
	}
	warp "job_sword1",getarg(0),getarg(1);
	end;
}
