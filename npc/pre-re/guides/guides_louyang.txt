			mes "武器店位于^55FF33+^000000。";

			mes "你会看到";
			mes "由洛阳经验丰富的";
			mes "铁匠锻造出的";
			mes "精良武器。";
			viewpoint 1,145,174,5,0x55FF33;
			break;
		case 5:
			mes "[士兵]";
			mes "道具店位于^3355FF+^000000。";
			next;
			mes "[士兵]";
			mes "知己知彼，";
			mes "百战不殆！";
			mes "事先做好准备总比事后后悔强。何不去看看他们的货品？";
			viewpoint 1,135,98,6,0x3355FF;
			break;

			mes "[士兵]";
			mes "旅途中感到疲倦时，我建议你去酒馆坐坐。它位于^00FF00+^000000。";
			next;
			mes "[士兵]";
			mes "酒馆是个好地方，";
			mes "可以结识其他旅客，也能听到一些有用的消息。";
			viewpoint 1,280,167,7,0x00FF00;
			break;
		case 7:
			mes "[士兵]";
			mes "如果你想移除";
			mes "迷你地图上的所有标记，请从菜单中选择'移除迷你地图所有标记'。";
		}
		break;
	case 2:
		viewpoint 2,218,255,2,0xFF00FF;
		viewpoint 2,263,94,3,0xFF00FF;
		viewpoint 2,309,80,4,0xFF00FF;
		viewpoint 2,145,174,5,0xFF00FF;
		viewpoint 2,135,98,6,0xFF00FF;
		viewpoint 2,280,167,7,0xFF00FF;
		mes "[士兵]";
		mes "好了，我已经清除你迷你地图上的所有标记。需要了解建筑位置时，随时来问我。";
		break;
	case 3:
		mes "[士兵]";
		mes "我想，偶尔";
		mes "独自去探索";
		mes "也很有趣。保重。";
	}
	close;
}
