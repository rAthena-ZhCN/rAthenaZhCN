rachel,138,146,5	script	拉赫向导	934,{
	mes "[拉赫向导]";
	mes "欢迎来到阿鲁纳佩尔兹的";
	mes "首都拉赫，这里是";
	mes "芙蕾雅女神温暖气息所及之地。";
	mes "如果您是第一次";
	mes "使用向导服务，不如";
	mes "先看看“公告”菜单如何？";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("村庄指南:移除小地图标记:公告:取消")) {
		case 1:
			mes "[拉赫向导]";
			mes "只要是拉赫境内的建筑，我都能告诉您其位置。";
			mes "那么，您想去哪里？";
			if (.@compass_check == 0) {
				mes "需要我";
				mes "在小地图上";
				mes "标记地点吗？";
				next;
				if (select("是:否") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("切斯鲁姆尼尔:旅馆:武器店:道具店:机场:取消")) {
				case 1:
					mes "[拉赫向导]";
					mes "切斯鲁姆尼尔是圣地，女神芙蕾雅的化身——教皇驻留于此。";
					mes "沿北边的道路走，就能找到那座建筑。";
					if (.@compass_check == 1) viewpoint 1,150,249,1,0xFF0000;
					break;
				case 2:
					mes "[拉赫向导]";
					mes "您可以在旅馆洗去旅途的疲惫。";
					mes "我左边的建筑就是拉赫的旅馆。";
					if (.@compass_check == 1) viewpoint 1,115,149,2,0xFF00FF;
					break;
				case 3:
					mes "[拉赫向导]";
					mes "您想看看拉赫出售的武器吗？";
					mes "武器店位于西门附近。";
					if (.@compass_check == 1) viewpoint 1,42,87,3,0x99FFFF;
					break;
				case 4:
					mes "[拉赫向导]";
					mes "拉赫的道具店出售品质最好的药水。";
					mes "它就位于西门附近。";
					if (.@compass_check == 1) viewpoint 1,83,78,4,0x0000FF;
					break;
				case 5:
					mes "[拉赫向导]";
					mes "机场位于东门之外。";
					if (.@compass_check == 1) viewpoint 1,273,125,5,0x00FF00;
					break;
				case 6:
					mes "[拉赫向导]";
					mes "如果您想清除小地图上的所有地点标记，";
					mes "再找我一次，选择“移除小地图标记”菜单即可。";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,150,249,1,0xFF0000;
			viewpoint 2,115,149,2,0xFF00FF;
			viewpoint 2,42,87,3,0x99FFFF;
			viewpoint 2,83,78,4,0x0000FF;
			viewpoint 2,273,125,5,0x00FF00;
			mes "[拉赫向导]";
			mes "好了，标记都已清除。如果您还想查询其他地点，随时告诉我。";
			break;
		case 3:
			mes "[拉赫向导]";
			mes "使用“村庄指南”菜单时，";
			mes "建筑位置会标记在屏幕右上方的小地图上，请注意查看。";
			mes "如果看不到小地图，请使用快捷键“ctrl+tab”，或按下基本信息窗口上的“地图”按钮，明白了吗？";
			mes "若无法看到村庄全图，您还可以用“-”按钮缩小地图。";
			break;
		case 4:
			mes "[拉赫向导]";
			mes "愿您在阿鲁纳佩尔兹的";
			mes "旅途愉快。";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}
